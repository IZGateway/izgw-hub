19  /**
20   * The base class for mock IIS endpoints.
21   * It checks credentials against the message headers and uses a performance simulator
22   * when the test case identifier selects one.
23   */
24  public class MockControllerBase extends SoapControllerBase implements InitializingBean {
25  	@Value("${iis.max-message-size:65536}")
26  	private int iisMaxMesageSize;
27  
28  	/**
29  	 * Make the controller and register it for access control.
30  	 * @param mshService The service that gets the expected credentials for a message header.
31  	 * @param messageNamespace The namespace of the SOAP messages.
32  	 * @param wsdl The name of the WSDL file.
33  	 * @param registry The access control registry, or null to not register the controller.
34  	 */
35  	protected MockControllerBase(IMessageHeaderService mshService, String messageNamespace, String wsdl,
36  			AccessControlRegistry registry) {
37  		super(mshService, messageNamespace, wsdl, null);
38  		if (registry != null) {
39  			registry.register(this);
40  		}
41  	}
42  
43  	public void afterPropertiesSet() {
44  		super.setMaxMessageSize(iisMaxMesageSize);
45  	}
46  
47      @Override
48      protected boolean isHubWsdl() {
49          return getDestinationService() != null;
50      }
51  
52      @Override
53  	protected void checkCredentials(HasCredentials s) throws SecurityFault {
54  		MockCredentialValidator.checkCredentials(mshService, s);
55  	}
56  
57  	@Override
58  	protected ResponseEntity<?> submitSingleMessage( // NOSONAR ? is intentional
59  		SubmitSingleMessageRequest submitSingleMessage,
60  		String destinationId
61  	) throws Fault {
62  		PerformanceSimulatorInterface p = PerformanceSimulatorMultiton.getInstance(submitSingleMessage.findTestCaseIdentifier()); 
63  		if (p != null) {
64  			ResponseEntity<?> r = p.simulateError(RequestContext.getResponse()); 
65  			if (r != null) {
66  				return r;
67  			}
68  			return new ResponseEntity<>(p.getResponse(submitSingleMessage), HttpStatus.OK);
69  		}
70  		return super.submitSingleMessage(submitSingleMessage, destinationId);
71  	}
72  }
