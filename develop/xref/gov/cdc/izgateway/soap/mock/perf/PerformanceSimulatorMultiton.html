1   package gov.cdc.izgateway.soap.mock.perf;

3   /**
4    * This class uses the Multiton pattern to return a PerformanceSimulator given the specified performanceProfileIdentifier.
5    */
6   public class PerformanceSimulatorMultiton {
7   	private PerformanceSimulatorMultiton () {}
8       static final String PERFORMANCE_PROFILE_TYPICAL = "PERF";
9       static final String PERFORMANCE_PROFILE_REPEATED_FAILURES = "PERF_REPEATED_FAILURES";
10      static final String PERFORMANCE_PROFILE_REGIONAL_OUTAGE = "PERF_REGIONAL_OUTAGE";
11      /** The test case identifier that selects the mock IIS simulator. */
12      public static final String PERFORMANCE_PROFILE_MOCK_IIS = "MOCK";
13      
14      private static final PerformanceSimulatorInterface performanceSimulatorDailyFailures = new PerformanceSimulatorDailyFailures();
15      private static final PerformanceSimulatorInterface performanceSimulatorRepeatedFailures = new PerformanceSimulatorRepeatedFailures();
16      private static final PerformanceSimulatorInterface performanceSimulatorRegionalOutage = new PerformanceSimulatorRegionalOutage();
17      private static final PerformanceSimulatorInterface mockIis = new PerformanceSimulatorMockIIS();
18      /**
19       * Get the simulator for a performance profile.
20       * @param performanceProfileIdentifier The test case identifier of the profile.
21       * @return The simulator, or null if no simulator matches the identifier.
22       */
23      public static PerformanceSimulatorInterface getInstance(String performanceProfileIdentifier) {
24      	if (performanceProfileIdentifier == null) {
25      		return null;
26      	}
27          switch (performanceProfileIdentifier) {
28          case PERFORMANCE_PROFILE_TYPICAL:
29              return performanceSimulatorDailyFailures;
30          case PERFORMANCE_PROFILE_REPEATED_FAILURES:
31              return performanceSimulatorRepeatedFailures;
32          case PERFORMANCE_PROFILE_REGIONAL_OUTAGE:
33              return performanceSimulatorRegionalOutage;
34          case PERFORMANCE_PROFILE_MOCK_IIS:
35          	return mockIis;
36          default:
37              return null;
38          }
39      }
40  }
