10  /**
11   * A simulator that the mock IIS endpoint uses to make errors, delays and responses.
12   */
13  public interface PerformanceSimulatorInterface {
14      /**
15       * Make a simulated error at random intervals.
16       * @param resp The HTTP response to write the error to.
17       * @return The error response, or null if the simulator makes no error.
18       * @throws Fault When the simulated error is a SOAP fault.
19       */
20      ResponseEntity<?> simulateError(HttpServletResponse resp) throws Fault;
21      /**
22       * Make the response to a request message.
23       * @param requestMessage The request message.
24       * @return The response message.
25       * @throws Fault When the simulator cannot make the response.
26       */
27      SubmitSingleMessageResponse getResponse(SubmitSingleMessageRequest requestMessage) throws Fault;
28  }
