16  /**
17   * Checks the credentials in a mock IIS request against the credentials of the matching message header.
18   */
19  public class MockCredentialValidator {
20  	private MockCredentialValidator() {}
21  	/**
22  	 * Run the failure action if the expected value is not empty and the actual value is different.
23  	 * @param expected The expected value. An empty value matches all values.
24  	 * @param actual The actual value.
25  	 * @param failure The action to run when the values do not match.
26  	 */
27  	public static void validateCredential(String expected, String actual, Runnable failure) {
28  		if (!StringUtils.isEmpty(expected) && !expected.equals(actual)) {
29  			failure.run();
30  		}
31  	}
32  
33  	/**
34  	 * Check the username, password and facility ID of a request against the message header
35  	 * for MSH-3 and MSH-4 of the HL7 message.
36  	 * The check passes when no message header matches or the request has no HL7 message.
37  	 * @param mshService The service that gets the message headers, or null to skip the check.
38  	 * @param s The request.
39  	 * @throws SecurityFault When one or more credentials do not match.
40  	 */
41  	public static void checkCredentials(IMessageHeaderService mshService, HasCredentials s) throws SecurityFault {
42  		String[] mshParts = null;
43  		if (mshService == null) {
44  			return;
45  		}
46  		if (s instanceof HasHL7Message hl7) {
47  			mshParts = hl7.getHl7Message().split("\\|");
48  		} else {
49  			return;
50  		}
51  		
52  		List<String> headers = Arrays.asList(mshParts.length > 2 ? mshParts[2] : "", mshParts.length > 3 ? mshParts[3] : "");
53  		List<IMessageHeader> h = mshService.getMessageHeaders(headers);
54  		if (h != null && !h.isEmpty()) {
55  			IMessageHeader hdr = h.get(0);
56  			List<String> failures = new ArrayList<>();
57  
58  			validateCredential(hdr.getUsername(), s.getUsername(),
59  					() -> failures.add("Invalid Username"));
60  			validateCredential(hdr.getPassword(), s.getPassword(),
61  					() -> failures.add("Invalid Password"));
62  			if (s instanceof HasFacilityID f) {
63  				validateCredential(hdr.getFacilityId(), f.getFacilityID(),
64  						() -> failures.add("Invalid FacilityID"));
65  			}
66  			if (!failures.isEmpty()) {
67  				throw SecurityFault.generalSecurity("User Credential Error", failures.toString(), null);
68  			}
69  		}
70  	}

72  }
