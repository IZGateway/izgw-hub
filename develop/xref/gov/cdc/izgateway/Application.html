288 	private static void checkApplication(ConfigurableApplicationContext ctx) {
289         IDestinationService destinationService = ctx.getBean(IDestinationService.class);
290         AppProperties props = ctx.getBean(AppProperties.class); 
291         serverMode = props.getServerMode();
292         serverName = props.getServerName();
293         DynamoDbRepository.setServerName(serverName);
294         
295 		initializeHealth(ctx);
296         IMessageHeaderService messageHeaderService = ctx.getBean(MessageHeaderService.class);
297         StatusCheckScheduler sc = ctx.getBean(StatusCheckScheduler.class);
298         Application app = ctx.getBean(Application.class);
299     	new Thread(() -> 
300     		// Load Mock IIS in background
301     		PerformanceSimulatorMultiton.getInstance(PerformanceSimulatorMultiton.PERFORMANCE_PROFILE_MOCK_IIS)
302     	).start();
303     	String database = HealthService.getHealth().getDatabase();
304         try {
305             // Test for database connectivity and prefetch caches.
306             List<IDestination> list = destinationService.getAllDestinations();
307 
308             if (list.isEmpty() && abortOnNoIIS) {
309             	HealthService.setHealthy(false, "No IIS Connections available in " + SystemUtils.getDestTypeAsString());
310                 log.error("No IIS Connections are available from {} in {}", database, SystemUtils.getDestTypeAsString());
311                 throw new ServiceConfigurationError(
312                 	"No IIS Connections are available from " + database + " in " + SystemUtils.getDestTypeAsString());
313             } else {
314                 // Prefetch to populate cache
315                 messageHeaderService.getAllMessageHeaders();
316                 if (app.statusCheck) {
317                 	sc.start();
318                 }
319                 HealthService.setHealthy(true, "Normal application startup");
320                 log.info("Connected to {}", database);
321             }
322         } catch (RuntimeException hex) { // NOSONAR This is handling the exception correctly
323         	HealthService.setHealthy(hex);
324             log.error(Markers2.append(hex), "Cannot get a database connection to {}: {}", database, hex.getMessage(), hex);
325             throw new ServiceConfigurationError("Cannot get a database connection to " + database, hex);
326         }
327 	}

349     /**
350      * Generate a a NIST SP 800-90A compliant secure random number
351      * generator.
352      *
353      * @return A compliant generator.
354      */
355 	@Bean
356     public static SecureRandom getSecureRandom() {
357 		/*
358          * According to NIST Special Publication 800-90A, a Nonce is
359          * A time-varying value that has at most a negligible chance of
360          * repeating, e.g., a random value that is generated anew for each
361          * use, a timestamp, a sequence number, or some combination of
362          * these.
363          *
364          * The nonce is combined with the entropy input to create the initial
365          * DRBG seed.
366          */
367     	if (secureRandom != null) {
368     		return secureRandom;
369     	}
370         byte[] nonce = ByteBuffer.allocate(8).putLong(System.nanoTime()).array();
371         EntropySourceProvider entSource = new BasicEntropySourceProvider(new SecureRandom(), true);
372         FipsDRBG.Builder drgbBldr = FipsDRBG.SHA512
373                 .fromEntropySource(entSource).setSecurityStrength(256)
374                 .setEntropyBitsRequired(256);
375         secureRandom = drgbBldr.build(nonce, true);
376         return secureRandom;
377     }

395 	// NOTE: use the ServerBuilder overload, not either List-based hook. Framework 7
396 	// deprecated configureMessageConverters(List)/extendMessageConverters(List) for
397 	// removal. WebMvcConfigurationSupport.createMessageConverters() calls
398 	// registerDefaults() before this hook, so the defaults (Jackson included) are
399 	// always present and we only add to them. The legacy list path still has the trap
400 	// that getMessageConverters() only adds defaults when the list is left empty, so
401 	// populating configureMessageConverters(List) silently dropped every default
402 	// converter app-wide.
403 	//
404 	// addCustomConverter places this ahead of the defaults. That is safe here because
405 	// SoapMessageConverter.canRead/canWrite both require
406 	// SoapMessage.class.isAssignableFrom(clazz) and exclude JSON, so it can never
407 	// claim a String or JSON payload regardless of its position.
408 	@Override
409     public void configureMessageConverters(HttpMessageConverters.ServerBuilder builder) {
410     	SoapMessageConverter smc = new SoapMessageConverter(SoapMessageConverter.INBOUND);
411     	smc.setHub(true);
412         builder.addCustomConverter(smc);
413         // Sets up SoapMessageWriter to handle \r as &#xD; if true, otherwise
414         // \r in hl7Message will be replaced with \n due to XML Parsing rules.
415         SoapMessageWriter.setFixNewLines(fixNewlines);
416     }

439      * @param http	The HttpSecurity to configure.
440      * @return	The SecurityFilterChain to use.
441      * @throws Exception If configuration fails.

443     @Bean
444     public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
445         http.authorizeHttpRequests(a -> a.requestMatchers("/**").permitAll())
446         	.x509(Customizer.withDefaults())
447         	.userDetailsService(userDetailsService());
448         if (!enableCsrf) {
449         	http.csrf(AbstractHttpConfigurer::disable);
450         }
451         return http.build();

474     /**
475      * Enable local server management port.
476      * @param additionalPort	The port to add.
477      * @return	The TomcatServletWebServerFactory to use
478      */
479 	@Bean
480 	TomcatServletWebServerFactory tomcatServletWebServerFactory(
481 			ObjectProvider<TomcatConnectorCustomizer> connectorCustomizers,
482 			ObjectProvider<TomcatContextCustomizer> contextCustomizers,
483 			ObjectProvider<TomcatProtocolHandlerCustomizer<?>> protocolHandlerCustomizers) {
484 		TomcatServletWebServerFactory factory = new TomcatServletWebServerFactory() {
485 			@Override
486 			public WebServer getWebServer(ServletContextInitializer... initializers) {
487 				log.info("Initializing Tomcat");
488 				try {
489 					return super.getWebServer(initializers);
490 				} finally {
491 					log.info("Tomcat initialized");
492 				}
493 			}
494 		};
495 		factory.getConnectorCustomizers().addAll(connectorCustomizers.orderedStream().toList());
496 		factory.getContextCustomizers().addAll(contextCustomizers.orderedStream().toList());
497 		factory.getProtocolHandlerCustomizers().addAll(protocolHandlerCustomizers.orderedStream().toList());
498         if (additionalPort < 1) {
499         	return factory;
500         }
501         Connector connector = new Connector("org.apache.coyote.http11.Http11NioProtocol");
502         connector.setScheme("http");
503         connector.setPort(additionalPort);
504         connector.setProperty("minSpareThreads", "3");  // This is for local administration, we don't need many.
505         factory.addAdditionalConnectors(connector);
506         return factory;
507 	}
