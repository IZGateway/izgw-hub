273 	 * Get the host name of this instance.
274 	 * <p>
275 	 * Every caller must use this method rather than resolving the name itself. The
276 	 * sender identity on a command and the queue a reply is read from are built from
277 	 * this value. If two callers disagreed, replies would land on a queue nobody reads
278 	 * and every remote instance would report a timeout. See IGDD-3599.
279 	 * </p>
280 	 *
281 	 * @return The host name, never null
282 	 */
283 	public static String myHostName() {
284 		// The core resolver is a strict superset of the commons-lang3 one: it reads
285 		// HOSTNAME, then COMPUTERNAME, then InetAddress, and never returns null. The
286 		// commons version returns null when HOSTNAME is unset and an empty string when
287 		// HOSTNAME is set but blank, and an empty string would silently build the queue
288 		// name izgw__refresh-requests that nobody reads.
289 		return gov.cdc.izgateway.utils.SystemUtils.getHostname();
290 	}

292 	private static void createRefreshQueues() {
293 		// Create an SQS queue named after the hostname (serverName) to receive refresh messages
294 		try {
295 			SqsClient sqsClient = getClient(region);
296 			String requestQueueName = getRequestQueueName(myHostName());
297 			String responseQueueName = getResponseQueueName(myHostName());
298 			createQueue(requestQueueName, sqsClient);
299 			createQueue(responseQueueName, sqsClient);
300 		} catch (Exception e) {
301 			log.error(Markers2.append(e), "Failed to create SQS queues: {}", e.getMessage());
302 		}
303 	}

305 	private static void createQueue(String queueName, SqsClient sqsClient) {
306 		try {
307 			sqsClient.getQueueUrl(r -> r.queueName(queueName));
308 			// This SHOULDN'T happen, but if it does, just log and move on.
309 			log.warn("SQS queue '{}' already exists", queueName);
310 		} catch (QueueDoesNotExistException e) {
311 			sqsClient.createQueue(r -> r.queueName(queueName));
312 			log.info("Created SQS queue '{}'", queueName);
313 		}
314 		createdQueues.add(queueName);
315 	}
316 
317 	private static void deleteQueues() {
318 		try {
319 		    SqsClient sqsClient = SqsClient.builder().build();
320 		    for (String queueName : createdQueues) {
321 		        deleteQueue(sqsClient, queueName);
322 		    }
323 		} catch (Exception e) {
324 		    log.error(Markers2.append(e), "Error during shutdown hook for SQS queue deletion: {}", e.getMessage());
325 		}
326 	}

328 	private static void deleteQueue(SqsClient sqsClient, String queueName) {
329 		try {
330 		    String queueUrl = sqsClient.getQueueUrl(r -> r.queueName(queueName)).queueUrl();
331 		    sqsClient.deleteQueue(DeleteQueueRequest.builder().queueUrl(queueUrl).build());
332 		    log.info("Deleted SQS queue '{}' on shutdown", queueName);
333 		} catch (Exception e) {
334 		    log.error(Markers2.append(e), "Failed to delete SQS queue '{}' on shutdown: {}", queueName, e.getMessage());
335 		}
336 	}

399     /**
400      * Send a command to every known instance other than the caller, then wait for the
401      * replies.
402      * <p>
403      * The caller identifies itself through {@code senderHost} and {@code senderRegion}
404      * on the request. Those values decide which entry is skipped, so a caller keeps its
405      * own notion of the local host name rather than inheriting one from this class.
406      * </p>
407      *
408      * @param request	The command to send. Its sender fields identify the caller
409      * @param results	The outcome map to fill in, keyed {@code region:hostname}
410      * @return the same map, with one entry per instance reached
411      */
412     public Map<String, String> distribute(RefreshRequest request, Map<String, String> results) {
413         String me = request.senderHost();
414         String myRegion = request.senderRegion();
415         Map<String, String> known;
416         try {
417             known = hostService.getHostsAndRegion();
418         } catch (HostDiscoveryException e) {
419             // Nothing was sent, and nothing is known about the rest of the fleet. The
420             // local reset still stands, so the local entry keeps its outcome, but the
421             // caller must not read the whole operation as complete. IGDD-3599.
422             log.error(Markers2.append(e, "eventType", "FLEET_DISCOVERY_FAILED",
423             		"result", DISCOVERY_FAILED),
424             		"Could not list the running instances: {}", e.getMessage());
425             results.put(FLEET_KEY, DISCOVERY_FAILED);
426             awaitRefreshResponses(request.eventId(), results);
427             return results;
428         }
429         for (Map.Entry<String, String> entry : known.entrySet()) {
430             String hostName = entry.getKey();
431             String hostRegion = entry.getValue();
432             if (hostName.equalsIgnoreCase(me) && hostRegion.equals(myRegion)) {
433                 continue;  // The caller already applied the command locally
434             }
435             sendRefreshMessage(request, results, hostName, hostRegion);
436         }
437         awaitRefreshResponses(request.eventId(), results);
438         return results;
439     }

441     /**
442      * Reduce per-instance outcomes to one status per region.
443      * <p>
444      * The result carries no host name and no instance count, so a consumer can forward
445      * it to an untrusted client without redacting it. See IGDD-3599, design decision D5.
446      * </p>
447      *
448      * @param results	Outcomes keyed {@code region:hostname}
449      * @return one of {@code complete}, {@code partial} or {@code failed} per region
450      */
451     public static Map<String, String> summarizeByRegion(Map<String, String> results) {
452         Map<String, int[]> counts = new TreeMap<>();
453         for (Map.Entry<String, String> entry : results.entrySet()) {
454             String key = entry.getKey();
455             int colon = key.indexOf(':');
456             String regionName = colon < 0 ? key : key.substring(0, colon);
457             int[] count = counts.computeIfAbsent(regionName, k -> new int[2]);
458             count[1]++;
459             if (isSuccess(entry.getValue())) {
460                 count[0]++;
461             }
462         }
463         Map<String, String> summary = new TreeMap<>();
464         for (Map.Entry<String, int[]> entry : counts.entrySet()) {
465             int ok = entry.getValue()[0];
466             int total = entry.getValue()[1];
467             String status = "partial";
468             if (ok == total) {
469                 status = "complete";
470             } else if (ok == 0) {
471                 status = "failed";
472             }
473             summary.put(entry.getKey(), status);
474         }
475         return summary;
476     }

482     /**
483      * Send a refresh request to the specified host in the specified region.
484      * @param request The refresh request to send
485      * @param results	A map to record the results of the send operation
486      * @param host	The host to send the message to
487      * @param hostRegion	The region the host is in
488      * @return results map with the status of the send operation
489      */
490     public Map<String, String> sendRefreshMessage(RefreshRequest request, Map<String, String> results, String host, String hostRegion) {
491         SqsClient sqsClient = clientFor(hostRegion);
492         String endpointName = hostRegion + ":" + host;
493         String queueName = getRequestQueueName(host);
494         try {
495             String queueUrl = sqsClient.getQueueUrl(r -> r.queueName(queueName)).queueUrl();
496             sqsClient.sendMessage(request.toSendMessageRequest(queueUrl));
497         } catch (QueueDoesNotExistException e) {
498         	// Usually a task that stopped recently. Elastic still lists it for up to
499         	// three minutes after its queue is deleted. Not the same as a real fault.
500             log.warn(Markers2.append("eventType", "SQS_SEND_FAILED", "direction", "send",
501             		"queueName", queueName, "targetHost", host, "targetRegion", hostRegion,
502             		"messageType", RefreshRequest.MESSAGE, "result", NO_QUEUE),
503             		"No SQS queue {} for {}", queueName, endpointName);
504             results.put(endpointName, NO_QUEUE);
505             return results;
506         } catch (Exception e) {
507             log.error(Markers2.append(e, "eventType", "SQS_SEND_FAILED", "direction", "send",
508             		"queueName", queueName, "targetHost", host, "targetRegion", hostRegion,
509             		"messageType", RefreshRequest.MESSAGE, "result", SEND_FAILURE),
510             		"Exception sending SQS message to {}: {}", queueName, e.getMessage());
511             results.put(endpointName, SEND_FAILURE);
512             return results;
513         }
514         log.info(Markers2.append("eventType", "SQS_SEND", "direction", "send",
515         		"queueName", queueName, "targetHost", host, "targetRegion", hostRegion,
516         		"messageType", RefreshRequest.MESSAGE, "keyId", request.jti()),
517         		"SQS send {} to {}", RefreshRequest.MESSAGE, endpointName);
518         results.put(endpointName, WAITING_FOR_RESPONSE);
519         return results;
520     }

523     /**
524      * Await responses from the refresh messages sent to other instances.
525      * @param eventId The event ID of the refresh request
526      * @param results	A map of results to update with responses
527      * @return results map with the status of the send operation
528      */
529     public Map<String, String> awaitRefreshResponses(String eventId, Map<String, String> results) {
530         if (results.values().stream().noneMatch(WAITING_FOR_RESPONSE::equals)) {
531             // Nothing was sent, so there is nothing to wait for and no queue to look up.
532             logOperationComplete(eventId, results);
533             return results;
534         }
535         SqsClient sqsClient = clientFor(region);
536         String queueName = getResponseQueueName(myHostName());
537         String queueUrl;
538         try {
539             queueUrl = sqsClient.getQueueUrl(r -> r.queueName(queueName)).queueUrl();
540         } catch (Exception e) {
541             log.error(Markers2.append(e), "Could not get response queue URL: {}", e.getMessage());
542             results.replaceAll((k, v) -> v.equals(WAITING_FOR_RESPONSE) ? INTERNAL_ERROR : v);
543             // The operation still ended, so it still gets a summary record. Without this
544             // an operator searching for SQS_OPERATION_COMPLETE finds nothing at all.
545             logOperationComplete(eventId, results);
546             return results;
547         }
548         long timeoutMillis = responseTimeoutMillis;
549         long start = System.currentTimeMillis();
550         // Replies that answer a different operation on this instance. Held so that one
551         // such message produces one log record for this wait, not one record per poll.
552         Set<String> logged = new HashSet<>();
553         boolean allResponded = results.values().stream().noneMatch(v -> v.equals(WAITING_FOR_RESPONSE));
554         while (!allResponded && System.currentTimeMillis() - start < timeoutMillis) {
555             try {
556                 collectReplies(sqsClient, queueUrl, queueName, eventId, results, logged);
557             } catch (Exception e) {
558                 // Polling or deletion failed part way through. Every reply already
559                 // collected keeps its outcome, and the rest are unknown rather than
560                 // failed on the remote instance. Throwing here instead would lose the
561                 // whole map and answer the operator with an HTTP error. IGDD-3599.
562                 log.error(Markers2.append(e, "eventType", "SQS_RECEIVE_FAILED",
563                 		"direction", "receive", "queueName", queueName,
564                 		"messageType", RefreshResponse.MESSAGE, "result", INTERNAL_ERROR),
565                 		"Could not read response queue {}: {}", queueName, e.getMessage());
566                 results.replaceAll((k, v) -> v.equals(WAITING_FOR_RESPONSE) ? INTERNAL_ERROR : v);
567                 logOperationComplete(eventId, results);
568                 return results;
569             }
570             allResponded = results.values().stream().noneMatch(v -> v.equals(WAITING_FOR_RESPONSE));
571         }
572         if (!allResponded) {
573 			results.replaceAll((k, v) -> v.equals(WAITING_FOR_RESPONSE) ? TIMED_OUT : v);
574 		}
575         logOperationComplete(eventId, results);
576         return results;
577     }

579     /**
580      * Read one batch of replies and record the ones that answer this operation.
581      *
582      * @param sqsClient	Client for this instance's own region
583      * @param queueUrl	URL of this instance's response queue
584      * @param queueName	Name of that queue, for the log record
585      * @param eventId	Correlation identifier of the running operation
586      * @param results	The outcome map to update
587      * @param logged	Identifiers of foreign replies already recorded in this wait
588      */
589     private void collectReplies(SqsClient sqsClient, String queueUrl, String queueName, String eventId,
590     		Map<String, String> results, Set<String> logged) {
591         List<Message> messages = sqsClient.receiveMessage(
592         	r -> r.queueUrl(queueUrl)
593 					.messageAttributeNames("All")
594         			.maxNumberOfMessages(10)
595         			.waitTimeSeconds(2)
596         		).messages();
597         for (Message msg : messages) {
598         	if (!RefreshResponse.MESSAGE.equals(msg.body())) {
599         		continue;
600         	}
601         	RefreshResponse response = RefreshResponse.fromMessage(msg);
602         	boolean answersThisOperation = Objects.equals(eventId, response.eventId());
603         	boolean usable = response.host() != null && response.status() != null;
604         	if (usable && !answersThisOperation) {
605         		releaseForeignReply(sqsClient, queueUrl, queueName, msg, response, logged);
606         		continue;
607         	}
608         	// Carry the reply's own eventId here as well as on a released reply, so that
609         	// every received reply reads the same way in Elasticsearch.
610         	log.info(Markers2.append("eventType", "SQS_RECEIVE", "direction", "receive",
611         			"queueName", queueName, "messageType", RefreshResponse.MESSAGE,
612         			"targetHost", response.host(), "targetRegion", response.region(),
613         			// Not "eventId": the MDC already carries the running operation's id.
614         			// A reply that answers another operation would otherwise be filed
615         			// under this one. IGDD-3599.
616         			"replyEventId", response.eventId(), "result", response.status()),
617         			"SQS receive {} from {}:{}", RefreshResponse.MESSAGE, response.region(), response.host());
618             String key = response.region() + ":" + response.host();
619             if (usable && results.get(key) != null) {
620                 results.put(key, response.status());
621                 // Delete only the messages we processed
622                 sqsClient.deleteMessage(r -> r.queueUrl(queueUrl).receiptHandle(msg.receiptHandle()));
623             } else {
624 				log.warn("Ignoring mismatched response message: {} {}", msg.body(), msg.messageAttributes());
625 			}
626         }
627     }

629     /**
630      * Make a reply that answers another operation available again.
631      * <p>
632      * Two operations on one instance share one response queue, so a poll can take a
633      * reply that the other one waits for. Every receive starts a new visibility period,
634      * so the reply must be released on every receive, not only the first. Left alone it
635      * stays hidden for the queue's visibility timeout, which is longer than the other
636      * operation's wait, and that operation reports a timeout for an instance that
637      * answered. See IGDD-3599.
638      * </p>
639      * <p>
640      * One second rather than zero. The reply is still available well inside the other
641      * operation's 15 second wait, and a reply whose operation already ended is touched
642      * about once a second instead of continuously.
643      * </p>
644      *
645      * @param sqsClient	Client for this instance's own region
646      * @param queueUrl	URL of this instance's response queue
647      * @param queueName	Name of that queue, for the log record
648      * @param msg	The message as received, carrying the current receipt handle
649      * @param response	The reply read from that message
650      * @param logged	Identifiers of foreign replies already recorded in this wait
651      */
652     private void releaseForeignReply(SqsClient sqsClient, String queueUrl, String queueName, Message msg,
653     		RefreshResponse response, Set<String> logged) {
654     	if (logged.add(msg.messageId())) {
655     		// Once per wait. One record per poll would bury the replies that do answer
656     		// this operation.
657         	log.info(Markers2.append("eventType", "SQS_RECEIVE", "direction", "receive",
658         			"queueName", queueName, "messageType", RefreshResponse.MESSAGE,
659         			"targetHost", response.host(), "targetRegion", response.region(),
660         			"replyEventId", response.eventId(), "result", response.status()),
661         			"SQS receive {} from {}:{} for another operation", RefreshResponse.MESSAGE,
662         			response.region(), response.host());
663     	}
664     	try {
665     		sqsClient.changeMessageVisibility(
666     				r -> r.queueUrl(queueUrl).receiptHandle(msg.receiptHandle()).visibilityTimeout(1));
667     	} catch (Exception e) {
668     		// Someone else's reply. Failing to release it must not end this wait.
669     		log.warn("Could not release reply for operation {}: {}", response.eventId(), e.getMessage());
670     	}
671     }

673     /**
674      * Record one summary line for a distributed operation.
675      * <p>
676      * This line answers "did every region apply the command" without a search across
677      * instances.
678      * </p>
679      *
680      * @param eventId	Correlation identifier of the operation
681      * @param results	Outcomes keyed {@code region:hostname}
682      */
683     private static void logOperationComplete(String eventId, Map<String, String> results) {
684     	// The fleet key reports whether the instance list could be read. It is not an
685     	// instance, so counting it would describe a remote instance that was never
686     	// discovered and never contacted. The region summary below still carries it, so
687     	// the discovery failure is still in this record. IGDD-3599.
688     	long total = results.keySet().stream().filter(k -> !FLEET_KEY.equals(k)).count();
689     	long ok = results.entrySet().stream()
690     			.filter(e -> !FLEET_KEY.equals(e.getKey()))
691     			.filter(e -> isSuccess(e.getValue()))
692     			.count();
693     	log.info(Markers2.append("eventType", "SQS_OPERATION_COMPLETE",
694     			"instances", total, "succeeded", ok, "failed", total - ok,
695     			"regions", summarizeByRegion(results)),
696     			"Distributed operation complete: {} of {} instances applied the command", ok, total);
697     }

700      * Start a listener thread that waits for refresh requests on the refresh queue.
701      * When a message is received, makes the appropriate API call on the DBController.
702      */
703     public void startRefreshListener() {
704         ExecutorService executor = Executors.newSingleThreadExecutor();
705         executor.submit(() -> {
706             SqsClient sqsClient = clientFor(region);
707             String queueName = getRequestQueueName(myHostName());
708             String queueUrl;
709             try {
710                 queueUrl = sqsClient.getQueueUrl(r -> r.queueName(queueName)).queueUrl();
711             } catch (Exception e) {
712                 log.error(Markers2.append(e), "Could not get refresh request queue URL: {}", e.getMessage());
713                 return;
714             }
715             log.info("Started refresh listener on queue: {}", queueName);
716             refreshLoop(sqsClient, queueUrl);
717         });
718     }

720 	private void refreshLoop(SqsClient sqsClient, String queueUrl) {
721 		while (true) {
722 		    try {
723 		        List<Message> messages = sqsClient.receiveMessage(
724 		        	r -> r.queueUrl(queueUrl)
725 		        			.messageAttributeNames("All")
726 		        			.maxNumberOfMessages(10)
727 		        			.waitTimeSeconds(10)
728 		        	).messages();
729 		        for (Message msg : messages) {
730 		        	if (RefreshRequest.MESSAGE.equals(msg.body())) {
731 		        		RefreshRequest request = RefreshRequest.fromMessage(msg);
732 		        		// Put the correlation id in the MDC, the way the request valve does for
733 		        		// an inbound HTTP request. Every record this thread writes then carries
734 		        		// it once. A log statement that appends it again produces a duplicate
735 		        		// JSON key, which is the fault commit a8f40240d fixed for "level".
736 		        		MDC.put(EventId.EVENTID_KEY, request.eventId());
737 		        		try {
738 			        		log.info(Markers2.append("eventType", "SQS_RECEIVE", "direction", "receive",
739 			        				"queueName", getRequestQueueName(myHostName()),
740 			        				"messageType", RefreshRequest.MESSAGE,
741 			        				"keyId", request.jti()),
742 			        				"SQS receive {} from {}:{}", RefreshRequest.MESSAGE,
743 			        				request.senderRegion(), request.senderHost());
744 			        		handleRefreshRequest(request);
745 			        		sqsClient.deleteMessage(r -> r.queueUrl(queueUrl).receiptHandle(msg.receiptHandle()));
746 		        		} finally {
747 		        			MDC.remove(EventId.EVENTID_KEY);
748 		        		}
749 		        	}
750 		        }
751 		    } catch (Exception e) {
752 		        log.error(Markers2.append(e), "Error processing refresh request: {}", e.getMessage());
753 		    }
754 		}
755 	}

757 	// Package-private so a test can drive one received command without starting the
758 	// listener thread. IGDD-3599.
759 	void handleRefreshRequest(RefreshRequest request) {
760 		log.info(Markers2.append("refreshRequest", request), "Refresh Request {}", request);
761 		String status = applyLocally(request);
762 		RefreshResponse response = new RefreshResponse(myHostName(), region, request.eventId(), status);
763 		sendRefreshResponse(request, response);
764 	}

766 	/**
767 	 * Apply a command on this instance, in process.
768 	 * <p>
769 	 * CAUTION: this method must never call a REST endpoint of this instance. If it
770 	 * did, and that endpoint distributes commands, every receiving instance would
771 	 * distribute again and the operation would never terminate. See IGDD-3599,
772 	 * design decision D4.
773 	 * </p>
774 	 * <p>
775 	 * Authorization is not rechecked here. The operator's request was authorized once,
776 	 * on the instance that received it, and this message arrived on a queue that only
777 	 * Hub instances can write to.
778 	 * </p>
779 	 *
780 	 * <p>
781 	 * Public because both controllers apply commands through it. The reset endpoints
782 	 * must call it for the instance that serves the request: {@link #distribute} skips
783 	 * the caller, so without this the serving instance is the one instance that never
784 	 * resets.
785 	 * </p>
786 	 *
787 	 * @param request The command to apply
788 	 * @return {@code "OK"} when every requested action succeeded, otherwise a short
789 	 *         description of the failure
790 	 */
791 	public String applyLocally(RefreshRequest request) {
792 		String result;
793 		try {
794 			if (request.jti() != null && apiKeyPrincipalProvider != null) {
795 				apiKeyPrincipalProvider.evictCredential(request.jti());
796 			}
797 			if (request.refreshConfig()) {
798 				configuration.refresh();
799 			}
800 			if (request.reset()) {
801 				resetBreakers(request.destId());
802 			}
803 			result = "OK";
804 		} catch (RuntimeException e) {
805 			log.error(Markers2.append(e), "Could not apply {}: {}", request, e.getMessage());
806 			result = Objects.toString(e.getMessage(), "NOT OK");
807 		}
808 		// Without this record the instance that serves an operator's request applies the
809 		// command silently: the remote instances log SQS_RECEIVE, and the serving one
810 		// logs nothing. An operator tracing one eventId would see every hop but the
811 		// first. This record also carries the outcome, which SQS_RECEIVE does not.
812 		log.info(Markers2.append("eventType", "COMMAND_APPLIED",
813 				"reset", request.reset(), "refreshConfig", request.refreshConfig(),
814 				"destId", request.destId(), "keyId", request.jti(), "result", result),
815 				"Applied command on this instance: reset={} destId={} result={}",
816 				request.reset(), request.destId(), result);
817 		return result;
818 	}

820 	private void resetBreakers(String destId) {
821 		if (destId == null) {
822 			endpointStatusService.resetCircuitBreakers();
823 			return;
824 		}
825 		// A null return means the repository holds no status for the destination. That
826 		// also happens when its cache is empty and the read behind it failed, so the
827 		// breaker state is unknown rather than known to be clear. The instance that
828 		// serves the operator's request answers 404 in that case, so an instance that
829 		// receives the command must not answer OK. IGDD-3599.
830 		if (endpointStatusService.resetCircuitBreakerById(destId) == null) {
831 			throw new IllegalStateException("No status for destination " + destId);
832 		}
833 	}

835     private void sendRefreshResponse(RefreshRequest request, RefreshResponse response) {
836         String responseQueueName = getResponseQueueName(request.senderHost());
837         try {
838             SqsClient sqsClient = clientFor(request.senderRegion());
839             String responseQueueUrl = sqsClient.getQueueUrl(r -> r.queueName(responseQueueName)).queueUrl();
840             sqsClient.sendMessage(response.toSendMessageRequest(responseQueueUrl));
841             log.info(Markers2.append("eventType", "SQS_SEND", "direction", "send",
842             		"queueName", responseQueueName, "messageType", RefreshResponse.MESSAGE,
843             		"targetHost", request.senderHost(), "targetRegion", request.senderRegion(),
844             		"result", response.status()),
845             		"SQS send {} to {}:{}", RefreshResponse.MESSAGE,
846             		request.senderRegion(), request.senderHost());
847         } catch (Exception e) {
848             log.error(Markers2.append(e, "eventType", "SQS_SEND_FAILED", "direction", "send",
849             		"queueName", responseQueueName, "messageType", RefreshResponse.MESSAGE),
850             		"Failed to send refresh response to {}: {}", responseQueueName, e.getMessage());
851         }
