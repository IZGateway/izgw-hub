3   /**
4    * Thrown when destination status cannot be read from Elasticsearch.
5    * <p>
6    * A reset clears a circuit breaker held in the status cache. An empty cache that was
7    * filled by a successful read holds no thrown breaker, so a reset against it succeeds
8    * and has nothing to do. An empty cache left by a failed read says nothing about the
9    * breakers, so a reset against it must not report success. See IGDD-3599.
10   * </p>
11   * <p>
12   * An instance with no Elasticsearch endpoint configured reads an empty status set
13   * successfully and does not raise this.
14   * </p>
15   *
16   * @author Audacious Inquiry
17   */
18  public class StatusUnavailableException extends RuntimeException {
19  	private static final long serialVersionUID = 1L;
20  
21  	/**
22  	 * Create the exception.
23  	 *
24  	 * @param message	What went wrong. This text becomes the outcome reported for the
25  	 *					instance, so it must read as an outcome
26  	 */
27  	public StatusUnavailableException(String message) {
28  		super(message);
29  	}
30  }
