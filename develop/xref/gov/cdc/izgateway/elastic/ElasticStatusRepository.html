89  	@Override
90  	public List<EndpointStatus> find(int maxQuarterHours, String[] include) {
91  		if (cache.isEmpty()) {
92  			refresh();
93  		}
94  		if (maxQuarterHours < 0) {
95  			maxQuarterHours = 1;
96  		} else if (maxQuarterHours > 4) {
97  			maxQuarterHours = 4;
98  		}
99  		List<EndpointStatus> l = new ArrayList<>();
100 		if (include.length == 0) {
101 			l.addAll(cache.values());
102 		} else {
103 			for (String inc: include) {
104 				EndpointStatus s = cache.get(inc);
105 				if (s != null) {
106 					l.add(s);
107 				}
108 			}
109 		}
110 		// Drop fractional part of 15 minutes from time
111 		long from = (System.currentTimeMillis() / QUARTER_HOUR.toMillis()) * QUARTER_HOUR.toMillis(); 
112 		while (--maxQuarterHours > 0) {
113 			try {
114 				l.addAll(getData(new Date(from), include).values());
115 			} catch (NoSuchAlgorithmException | IOException e) {
116 				// Ignore this, error is already logged.
117 			}
118 			from -= QUARTER_HOUR.toMillis();
119 		}
120 
121 		return l;
122 	}

133 	/**
134 	 * Make sure that the cache holds the status this instance knows about.
135 	 * <p>
136 	 * {@link #refresh()} reports a failed read by returning false and leaving the cache
137 	 * empty. A reset that ignored that cleared nothing and reported success, because an
138 	 * empty cache and a cache with no thrown breaker look the same from here. Only the
139 	 * reset methods call this: a read that returns no status is a valid answer for a
140 	 * query, and {@code findById} must still answer 404 rather than an error. See
141 	 * IGDD-3599.
142 	 * </p>
143 	 * <p>
144 	 * An instance with no Elasticsearch endpoint configured reads an empty status set
145 	 * successfully, so this does not raise for it.
146 	 * </p>
147 	 *
148 	 * @throws StatusUnavailableException If the status cannot be read
149 	 */
150 	private void requireStatus() {
151 		if (cache.isEmpty() && !refresh()) {
152 			throw new StatusUnavailableException("Could not read status from Elasticsearch");
153 		}
154 	}

158 		requireStatus();
159 		for (IEndpointStatus status: cache.values()) {
160 			if (status.isCircuitBreakerThrown()) {
161 				status.setStatus(IEndpointStatus.CONNECTED);
162 			}
163 		}

166 	@Override
167 	public EndpointStatus resetCircuitBreakerById(String id) {
168 		// Without this a failed read returns null, which the caller reports as "no status
169 		// for this destination". Two different causes would read the same way, and an
170 		// operator could not tell them apart. IGDD-3599.
171 		requireStatus();
172 		IEndpointStatus status = cache.get(id);
173 		if (status == null) {
174 			return null;
175 		}
176 		if (status.isCircuitBreakerThrown()) {
177 			status.setStatus(IEndpointStatus.CONNECTED);
178 		}
179 		if (status instanceof EndpointStatus s) {
180 			return s;
181 		}
182 		EndpointStatus s2 = new EndpointStatus(status);
183 		cache.put(id, s2);
184 		return s2;
185 	}

187 	/**
188 	 * Request status from ElasticSearch
189 	 * @return  true if refreshed, false otherwise
190 	 */
191 	public boolean refresh() {
192 		boolean refreshed = false;
193 		try {
194 			Map<String, EndpointStatus> m = getData(new Date(), INCLUDE_ALL);
195 			// Add any values for which we previously had a status, but we didn't compute one.
196 			// These at least keeps the system safe from a complete failure to track status
197 			// if it cannot reach the data in the repository (ElasticSearch).
198 			cache.values().forEach(s -> m.computeIfAbsent(s.getDestId(), (String j) -> s));
199 			// And replace the map.
200 			cache = m;
201 			refreshed = true;
202 		} catch (IOException | NoSuchAlgorithmException e) {
203 			log.error("Exception refreshing status service: {}", e.getMessage(), e);
204 		} catch (StatusUnavailableException e) {
205 			// A response that cannot be parsed is a failed read. parseResult already
206 			// wrote the detail, so this records only that the refresh did not happen.
207 			log.error("Status refresh did not complete: {}", e.getMessage());
208 		}
209 		return refreshed;
210 	}

212 	private Map<String, EndpointStatus> getData(Date from, String[] include) throws IOException, NoSuchAlgorithmException {
213 		
214 		if (config.getUrl() == null) {
215 			return new ConcurrentHashMap<>(); 
216 		}
217 		String request = getRequest(from, include);
218 		try {
219 			HttpsURLConnection con = config.getConnection();
220 			OutputStream os = con.getOutputStream();
221 			os.write(request.getBytes(StandardCharsets.UTF_8));
222 			if (con.getResponseCode() == 200) {
223 				InputStream is = con.getInputStream();
224 				String result = IOUtils.toString(is, StandardCharsets.UTF_8);
225 				return parseResult(result);
226 			} else {
227 				InputStream err = con.getErrorStream();
228 				String error = IOUtils.toString(err, StandardCharsets.UTF_8);
229 				throw new IOException(error);
230 			}
231 		} catch (IOException e) {  // NOSONAR Exception handling is OK
232 			log.error("IOException getting destination status: {}", e.getMessage());
233 			throw e;
234 		} catch (NoSuchAlgorithmException e) { // NOSONAR Exception handling is OK
235 			log.error("TLS Configuration Exception getting destination status: {}", e.getMessage());
236 			throw e;
237 		}
238 	}

244 		String destinationId;

246 		int[] count = { 0, 0 };

250 		private String url;

253 		private String errorSummary;
254 		private String errorDetail;
255 		private String destVersion;
256 		private String tag;
257 		public ParsedResponse(long histogramTime, String destId, int count, boolean hasError, long latestTxTime, long firstTxTime) {
258 			timeRange = histogramTime;
259 			destinationId = destId;
260 			this.count[hasError ? 1 : 0] = count;
261 			minTimes[hasError ? 1 : 0] = firstTxTime;
262 			maxTimes[hasError ? 1 : 0] = latestTxTime;
263 		}
264 		
265 		public boolean updateResponse(long histogramTime, int count, boolean hasError, long latestTxTime, long firstTxTime, int maxFailures) {
266 			// Accumulate errors if there aren't enough already
267 			if (histogramTime < timeRange && hasError && this.count[1] < maxFailures) {
268 				timeRange = histogramTime;
269 				this.count[1] += count;
270 				minTimes[1] = firstTxTime;
271 				maxTimes[1] = latestTxTime;
272 				return true;
273 			} else if (histogramTime > timeRange && count > 0) {
274 				timeRange = histogramTime;
275 				this.count[hasError ? 1 : 0] = count;
276 				minTimes[hasError ? 1 : 0] = firstTxTime;
277 				maxTimes[hasError ? 1 : 0] = latestTxTime;
278 				return true;
279 			}
280 			return false;
281 		}

284 		 * Compute whether the destination is available based on parsed response.
285 		 * A destination is available if its most recent positive status occurs after its most recent
286 		 * negative status.
287 		 * 
288 		 * @return	true if the destination appears to be available, false otherwise.
289 		 */
290 		public boolean isAvailable() {
291 			return maxTimes[0] > maxTimes[1];
292 		}
293 		
294 		public boolean isCircuitBroken(int threshold) {
295 			// If there are no successes, or the time of the last success is before the time of the last failure
296 			if (this.count[0] == 0 || this.maxTimes[0] < this.minTimes[1]) {
297 				return this.count[1] >= threshold;
298 			}
299 			return false;
300 		}

303 			return Math.max(maxTimes[0], maxTimes[1]);

363 	/**
364 	 * Parse aggregate status data from ElasticSearch into endpoint status.
365 	 * 
366 	 * Look at recent status to see if any destinations are failing, and if so, throw the circuit breaker on them.
367 	 * Look at older status to see if circuit breaker was thrown or should be thrown.
368 	 * 
369 	 * Anything that has had the circuit breaker thrown on it should be rechecked every few minutes, and if it's running again, the circuit breaker should be reset.
370 	 * Anything worth looking at that is failing more than three times in a row should have the circuit breaker thrown.
371 	 * @param result	The query result from ElasticSearch.  This is a JSON String in the form
372 	 * 					of a response with multiple aggregates.
373 	 * 					rawResponse.aggregations.histogram: This aggregate tracks the timestamp associated with the histogram buckets.
374 	 * 						histogram.buckets[].key is a long value giving the start of the histogram bucket.
375 	 * 					destination: This aggregate collects counts by destination identifier.
376 	 * 						histogram.buckets[].destination.buckets[].key is a string giving the destination identifier.
377 	 * 					destination.buckets[].hasProcessError: This aggregate collects counts by hasProcessError value.
378 	 * 						hasProcessError.buckets[].key_as_string is a string giving the value of hasProcessError.
379 	 * 					hasProcessError.buckets[].latestTxTime: This aggregate reports the last transaction time for the destination.
380 	 * 						latestTxTime.buckets[].key is a long value giving the time of the last transaction to this destination with the given status.
381 	 * 					hasProcessError.buckets[].firstTxTime: This aggregate reports the last transaction time for the destination.
382 	 * 						firstTxTime.buckets[].key is a long value giving the time of the last transaction to this destination with the given status.
383 	 * @return
384 	 */
385 	private Map<String, EndpointStatus> parseResult(String result) {
386 		Map<String, ParsedResponse> map = new ConcurrentHashMap<>();
387 		try {
388 			JsonNode node = getBucket(result);
389 			if (node.getNodeType() != JsonNodeType.ARRAY) {
390 				log.error("Expected array at rawResponse.aggregations.histogram.buckets");
391 				throw new IllegalStateException("Array expected at rawResponse.aggregations.histogram.buckets");
392 			} else {
393 				parseHistogram(map, node);
394 			}
395 		} catch (Exception e ) {
396 			log.error(Markers2.append(e), "Error reading status response from ElasticSearch: {}", e.getMessage());
397 			// Returning the existing cache here reported a failed read as a successful
398 			// one: refresh() set refreshed to true, so a reset of an empty cache cleared
399 			// nothing and replied OK. refresh() catches this and returns false, which
400 			// leaves the previous cache in place exactly as returning it did. Only the
401 			// reset path acts on that false, so a query still answers from whatever the
402 			// cache holds. IGDD-3599.
403 			throw new StatusUnavailableException("Could not parse the status response: " + e.getMessage());
404 		}
405 		Map<String, EndpointStatus> newCache = new HashMap<>();
406 		for (ParsedResponse r: map.values()) {
407 			EndpointStatus s = convertToStatus(r);
408             if ( s != null) {
409             	newCache.put(s.getDestId(), s);
410             }
411 		}
412 		return newCache;
413 	}

415 	private JsonNode getBucket(String result) throws JsonProcessingException {
416 		JsonNode node = mapper.readTree(result);
417 		try {
418 			node = getNode("rawResponse.aggregations.histogram.buckets", node);
419 		} catch (IllegalArgumentException e) {
420 			node = getNode("aggregations.histogram.buckets", node);
421 		}
422 		return node;
423 	}

425 	private EndpointStatus convertToStatus(ParsedResponse r) {
426 		EndpointStatus s = newEndpointStatus();
427 
428         IDestination d = destinationService.findByDestId(r.destinationId);
429         if (d == null) {
430         	// This can occur when status result from elastic has a destination in
431         	// it that is no longer active.
432             log.warn("Destination not found for {}.", r.destinationId);
433             return null;
434         }
435 
436 		s.setDestId(r.destinationId);
437 		s.setJurisdictionId(d.getJurisdictionId());
438 		if (r.isAvailable()) {
439 			s.setStatus(IEndpointStatus.CONNECTED);
440 		} else if (r.isCircuitBroken(maxFailuresBeforeCircuitBreaker)) {
441 			s.setStatus(IEndpointStatus.CIRCUIT_BREAKER_THROWN);
442 		} else {
443 			s.setStatus(r.getErrorSummary());
444 		}
445 		s.setStatusAt(new Date(r.getMaxTime()));
446 		s.setStatusBy(SystemUtils.getHostname());
447 		s.setDestUri(r.getUrl());
448 		s.setDestVersion(r.getDestVersion());
449 		s.setDetail(r.getErrorDetail());
450 		s.setDiagnostics(getDiagnostics(r.getFaultCode(), r.getFaultName()));
451 		return s;
452 	}

465 	private void parseHistogram(Map<String, ParsedResponse> map, JsonNode node) {
466 		int errors = 0;
467 		int destinationErrors = 0;
468 		for (int i = node.size() - 1; i >= 0; --i) {
469 			JsonNode histBucket = node.get(i);
470 			long histogramTime = histBucket.get("key").asLong();
471 			if (histogramTime != 0) {
472 				JsonNode dests = getNode("destination.buckets", histBucket);
473 				if (dests.getNodeType() != JsonNodeType.ARRAY) {
474 					if (errors++ == 0) {	// Report the first error of this type
475 						log.error("Expected array at destination.buckets");
476 					}
477 					continue;
478 				}
479 				destinationErrors = parseDestination(map, destinationErrors, node, histogramTime, dests);
480 			}
481 		}
482 	}

484 	private int parseDestination(Map<String, ParsedResponse> map, int errors, JsonNode node, long histogramTime,
485 			JsonNode dests) {
486 		for (JsonNode destBucket: dests) {
487 			String destId = destBucket.get("key").asText();
488 			if (!StringUtils.isEmpty(destId)) {
489 				JsonNode statii = getNode("hasProcessError.buckets", destBucket);
490 				if (node.getNodeType() != JsonNodeType.ARRAY) {
491 					if (errors++ == 0) {	// Report the first error of this type
492 						log.error("Expected array at hasProcessError.buckets");
493 					}
494 					continue;
495 				}
496 				errors = parseStatus(map, errors, histogramTime, destId, statii);
497 			}
498 		}
499 		return errors;
500 	}

502 	private int parseStatus( // NOSONAR Cognitive complexity OK
503 		Map<String, ParsedResponse> map, int errors, long histogramTime, String destId, JsonNode statii
504 	) {  
505 		for (JsonNode statusBucket: statii) {
506 			String hasErrorStr = statusBucket.get("key_as_string").asText();
507 			if (StringUtils.isEmpty(hasErrorStr)) {
508 				// Missing value
509 				if (errors++ == 0) {
510 					log.error("Missing boolean value at key_as_string");
511 				}
512 				continue;
513 			}
514 			boolean hasError = statusBucket.get("key_as_string").asBoolean();
515 			if (!hasErrorStr.equalsIgnoreCase(Boolean.toString(hasError))) {
516 				// Not really a boolean value.
517 				if (errors++ == 0) {
518 					log.error("Expected boolean at key_as_string, got {}", hasErrorStr);
519 				}
520 				continue;
521 			}
522 			int count = statusBucket.get("doc_count").asInt();
523 			if (count == 0) {
524 				// Either there are no documents (no error, no foul),
525 				// or the count field is missing.
526 				continue;
527 			}
528 			JsonNode n = getNode("latestTxTime.value", statusBucket);
529 			long latestTxTime = n.asLong();
530 			if (latestTxTime == 0 && errors++ == 0 && log.isErrorEnabled()) {
531 				log.error("latestTxTime.value invalid: {}", n.asText());
532 			}
533 			n = getNode("firstTxTime.value", statusBucket);
534 			long firstTxTime = n.asLong();
535 			if (firstTxTime == 0 && errors++ == 0 && log.isErrorEnabled()) {
536 				log.error("firstTxTime.value invalid: {}", n.asText());
537 			}
538 			
539 			ParsedResponse p = map.get(destId);
540 			// If there's no records matching this time frame, status, and destination, don't bother updating status.
541 			if (count != 0) {
542 				// If there's no original record, create one.
543 				boolean update = true;
544 				if (p == null) {
545 					p = new ParsedResponse(histogramTime, destId, count, hasError, latestTxTime, firstTxTime);
546 					map.put(destId, p);
547 				} else {
548 					// Otherwise update the existing one.
549 					update = p.updateResponse(histogramTime, count, hasError, latestTxTime, firstTxTime, maxFailuresBeforeCircuitBreaker);
550 				}
551 				if (update) {
552 					parseExtraValues(statusBucket, p);
553 				}
554 			}
555 		}
556 		return errors;
557 	}

569 		value = getTopValue("faultCode", statusBucket);

576 		}
577 		value = getTopValue("errorDetail", statusBucket);

579 			p.setErrorDetail(value);

584 		}

592 		}

599 	 * @return	The string value associated with the bucket.

601 	private String getTopValue(String key, JsonNode statusBucket) {
602 		String find = key + "-bucket." + key + "-metric.top.1.metrics";
603 		try {
604 			JsonNode node = getNode(find, statusBucket);
605 			return node.properties().iterator().next().getValue().asText();
606 		} catch (NoSuchElementException | IllegalArgumentException e) {
607 			log.warn("Could not find {}", find);
608 			return null;
609 		}
610 	}

612 	/**
613 	 * Compute the request we need to submit from the template and return it as a String.
614 	 * @param now 
615 	 * @param include 
616 	 * @return	The request that needs to be submitted to elastic.
617 	 */
618 	private String getRequest(Date now, String[] include) {
619 		Calendar cal = new GregorianCalendar();
620 		cal.setTime(now);
621 		// Look back twice as far as we need to in case the server checking before us dropped out of the rota
622 		cal.add(Calendar.MINUTE, -2 * statusCheckPeriodInMinutes);
623 		Date startTime = cal.getTime();
624 		Map<String, String> map = new HashMap<>();
625 		map.put("start", FORMATTER.format(startTime));
626 		map.put("end", FORMATTER.format(now));
627 		map.put("startLong", Long.toString(startTime.getTime()));
628 		map.put("endLong", Long.toString(now.getTime()));
629 		map.put("environment", SystemUtils.getDestTag());
630 		map.put("include", include.length == 0 ? "" : computeIncludeString(include));
631 		return populateTemplate(map);
632 	}
633 
634 	private String computeIncludeString(String[] include) {
635 		StringBuilder b = new StringBuilder();
636 		b.append(",\"include\": [ ");
637 		for (String inc: include) {
638 			b.append("\"").append(inc).append("\", ");
639 		}
640 		// Remove trailing ", " at end 
641 		b.setLength(b.length() -2);
642 		b.append("]");
643 		return b.toString();
644 	}
