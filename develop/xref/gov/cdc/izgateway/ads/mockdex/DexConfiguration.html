12  /**
13   * Configuration for DEX uploads. The DEX sender and the mock DEX endpoint use it.
14   */
15  @Data
16  @Configuration
17  @EnableScheduling
18  public class DexConfiguration {
19      @Getter
20  	@Value("${dex.tus-upload-directory:temp/upload/tus}")
21      private String tusUploadDirectory;
22  
23      @Getter
24      @Value("${dex.app-upload-directory:temp/upload/app}")
25      private String appUploadDirectory;
26  
27      @Getter
28      private final String mode;
29  
30      // Present DEX OAuth methods do not require Content-Type header, since DEX uses Query Parameters
31      @Value("${dex.oauth-in-query:false}")
32  	private boolean usingQueryParameters;
33      
34      @Value("${dex.report-progress:true}")
35  	private boolean reportingProgressEnabled;
36  
37      @Value("${dex.numRetries:2}")
38  	private int numRetries;
39      
40      /**
41       * Creates the DEX configuration.
42       * @param app The application properties that supply the server mode
43       */
44      public DexConfiguration(@Autowired AppProperties app) {
45      	mode = app.getServerMode();
46      }
47  }
