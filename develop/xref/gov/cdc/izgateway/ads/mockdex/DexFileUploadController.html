127     public DexFileUploadController(DexConfiguration dexConfig, AccessControlRegistry registry, ClientTlsSupport tlsSupport) {
128     	config = dexConfig;
129         tusUploadDirectory = Path.of(dexConfig.getTusUploadDirectory());
130         appUploadDirectory = Path.of(dexConfig.getAppUploadDirectory());
131         try {
132             Files.createDirectories(tusUploadDirectory);
133             Files.createDirectories(appUploadDirectory);
134         } catch (IOException e) {
135             log.error(Markers2.append(e), "Cannot create upload directory");
136         }
137         uploadService = new TusFileUploadService().withStoragePath(dexConfig.getTusUploadDirectory()).withUploadUri("/rest/upload/dex");//latest TUS version
138         tokenStore.setUsingQueryParameters(config.isUsingQueryParameters());
139         tokenStore.setNumRetries(dexConfig.getNumRetries());
140         tokenStore.setTlsSupport(tlsSupport);
141         //registry.register(this, "/rest");  /*Can we change this to below*/
142         registry.register(this);
143 
144         Runtime.getRuntime().addShutdownHook(new Thread()
145         {
146             @Override
147             public void run()
148             {
149                 shutdown();
150             }
151         });
152     }

164     	File folder = path.toFile();
165     	if (folder.isDirectory()) {
166     		for (File f: FileUtils.listFiles(folder, null, false)) {
167     			try {
168     				f.delete();  // NOSONAR: We don't care if delete failed

172     		}
173     	}

177 	 * Handle OAuth requests for the controller.
178 	 * 
179 	 * @param servletRequest	The servlet request
180 	 * @param username	The username
181 	 * @param password	The password
182 	 * @param refreshToken	The refresh token
183 	 * @param grantType	The grant type requested
184 	 * @param contentType	The content type of the request
185 	 * @return	The requested access token reported in json format.
186 	 * 
187 	 * @throws OAuthException If the token was not valid
188 	 */
189 	@RequestMapping(value = { "/oauth", "/oauth/refresh" }, method = { RequestMethod.POST, RequestMethod.GET})
190     @Operation(hidden=true)
191     public ResponseEntity<AccessToken> handleOauthPost(
192     	HttpServletRequest servletRequest,
193     	@RequestParam(required=false) String username,
194     	@RequestParam(required=false) String password,
195     	@RequestParam(name="refresh_token", required=false) String refreshToken,
196     	@RequestParam(name="grant_type", defaultValue="password") String grantType,
197     	@RequestHeader(HttpHeaders.CONTENT_TYPE) String contentType
198     	
199     ) throws OAuthException {
200         // Disable Logging
201     	RequestContext.disableTransactionDataLogging();
202         if (!config.isUsingQueryParameters() && !"application/x-www-form-urlencoded".equals(contentType)) {
203             throw new HttpStatusException(HttpServletResponse.SC_UNSUPPORTED_MEDIA_TYPE);
204         }
205         
206         String message = "";
207         switch (grantType) {
208         case "refresh_token":
209         	if (StringUtils.isBlank(refreshToken)) {
210         		message = "refresh_token parameter cannot be blank, empty or missing.";
211         	}
212         	break;
213         case "password":
214             if (StringUtils.isBlank(username)) {
215                 message = "username parameter cannot be blank, empty or missing.";
216             }
217             if (StringUtils.isBlank(password)) {
218             	message = "password parameter cannot be blank, empty or missing.";
219             }
220         	break;
221         default:
222             message = "grant_type must be equal to password or refresh_token.";
223         }
224         if (message.length() > 0) {
225             throw new OAuthException("invalid_request", String.format("OAuth Protocol Error: %s%nParameters: %s", message, parameterMapToString(servletRequest.getParameterMap())));
226         }
227 
228         AccessToken accessToken = null;
229 
230         // Regardless of grant type, just create a new token.
231         accessToken = tokenStore.createToken();
232         
233         log.debug("Created Token: {}", accessToken.getAccessToken());
234         
235         HttpHeaders headers = new HttpHeaders();
236         headers.add(HttpHeaders.CONTENT_TYPE, "application/json");
237         headers.add(HttpHeaders.CACHE_CONTROL, "no-store");
238         headers.add(HttpHeaders.PRAGMA, "no-cache");
239         
240         return new ResponseEntity<>(accessToken, headers, HttpServletResponse.SC_CREATED);
241     }

243     private String parameterMapToString(Map<String, String[]> parameterMap) {
244     	TreeMap<String, String> t = new TreeMap<>();
245     	for (Entry<String, String[]> e : parameterMap.entrySet()) {
246     		String value = "[Hidden]";
247     		if (!"password".equals(e.getKey()) || e.getKey().contains("token")) {
248     			String[] a = e.getValue();
249     			value = a.length > 0 ? a[0] : null;
250     		}
251     		t.put(e.getKey(), value);
252     	}
253 		return t.toString();
254 	}

263     @ExceptionHandler(AuthenticationException.class)
264     ResponseEntity<ErrorObject> handleAuthenticationException(
265         HttpServletRequest req, HttpServletResponse resp, AuthenticationException ex
266     ) {
267         HttpHeaders headers = new HttpHeaders();
268         String value = String.format(
269             "Bearer realm=%s, error=%s, error_detail=%s",
270             realm, ex.getOAuthError().getError(), ex.getOAuthError().getErrorDescription()
271         );
272         headers.add(HttpHeaders.WWW_AUTHENTICATE, value);
273         log.info("Authentication Exception {} on Token: {}", value, ex.getToken());
274         return new ResponseEntity<>(ex.getOAuthError(), headers, HttpServletResponse.SC_UNAUTHORIZED);
275     }

277     @ExceptionHandler(Exception.class)
278     ResponseEntity<? extends Object> handleException(HttpServletRequest req, HttpServletResponse resp, Exception ex) {
279         HttpHeaders headers = new HttpHeaders();
280         
281         if (ex instanceof OAuthException oaex) {
282             return handleOAuthException(req, resp, oaex);
283         }
284         if (ex instanceof AuthenticationException aex) {
285         	return handleAuthenticationException(req, resp, aex);
286         }
287         
288         ADSErrorResponse err = null;
289         String eventId = MDC.get(EventIdMdcConverter.EVENT_ID_MDC_KEY);
290 
291         if (ex instanceof FaultSupport f) {
292            err = new ADSErrorResponse(f, eventId);
293         } else if (ex instanceof ServletRequestBindingException) {
294             err = new ADSErrorResponse(new UnexpectedExceptionFault("Servlet Binding Exception", ex, null), eventId);
295             err.setRetryStrategy(RetryStrategy.CORRECT_MESSAGE);
296         } else if (ex instanceof HttpStatusException hex && hex.getStatusCode() == HttpServletResponse.SC_UNAUTHORIZED) {
297             err = new ADSErrorResponse(new UnexpectedExceptionFault("Not Authorized", ex, null), eventId);
298             headers.add(HttpHeaders.WWW_AUTHENTICATE, "Bearer");
299             err.setRetryStrategy(RetryStrategy.CORRECT_MESSAGE);
300         } else {
301             err = new ADSErrorResponse(new UnexpectedExceptionFault(ex, null), eventId);
302         }
303 
304         HttpStatus status;
305         if (ex instanceof UnknownDestinationFault) {
306             status = HttpStatus.NOT_FOUND;
307         } else {
308         	status = err.getRetryStrategy().getStatus();
309         }
310 
311         log.error(Markers2.append(ex), "{}", ex.getMessage());
312         return new ResponseEntity<>(err, headers, status);
313     }

336     /**
337      * Handle an upload request
338      * @param servletRequest	The request
339      * @param servletResponse	The response
340      * @throws Exception	If any errors occur
341      */
342     @RequestMapping(value = { "/dex", "/dex/**" }, method = { RequestMethod.POST, RequestMethod.PATCH,
343         RequestMethod.HEAD, RequestMethod.DELETE, RequestMethod.GET, RequestMethod.OPTIONS })
344     @Operation(hidden=true)
345     public void upload(HttpServletRequest servletRequest, HttpServletResponse servletResponse) throws Exception {
346         // Disable Logging
347         RequestContext.disableTransactionDataLogging();
348         
349     	// OPTIONS shouldn't require a token, it's just a status check.
350         String method = servletRequest.getMethod();
351         if (!"OPTIONS".equals(method)) {
352         	String token = null;
353             try {
354             	token = getBearerToken(servletRequest);
355                 tokenStore.getAccess(token);
356             } catch (OAuthException ex) {
357                 throw new AuthenticationException(ex.getOAuthError(), token);
358             }
359         }
360         
361         this.uploadService.process(servletRequest, servletResponse);
362 
363         String uploadURI = servletRequest.getRequestURI();
364 
365         UploadInfo uploadInfo = null;
366         try {
367             uploadInfo = this.uploadService.getUploadInfo(uploadURI);
368         } catch (IOException | TusException e) {
369             log.error(Markers2.append(e), "get upload info");
370             throw e;
371         }
372 
373         if (uploadInfo != null && !uploadInfo.isUploadInProgress()) {
374         	submissions.put(StringUtils.substringAfterLast(uploadURI, "/"), getInfo(uploadInfo));
375             try (InputStream is = this.uploadService.getUploadedBytes(uploadURI)) {
376                 Path output = this.appUploadDirectory.resolve(uploadInfo.getFileName());
377                 Files.copy(is, output, StandardCopyOption.REPLACE_EXISTING);
378             } catch (IOException | TusException e) {
379                 log.error(Markers2.append(e), "get uploaded bytes");
380                 // If an error occurs copying in mock, this is not an essential failure of the service.
381                 return;
382             }
383 
384             try {
385                 this.uploadService.deleteUpload(uploadURI);
386             } catch (IOException | TusException e) {
387                 // If an error occurs deleting data in mock, this is not an essential failure of the service.
388                 log.error(Markers2.append(e), "delete upload");
389             }
390         }
391     }

393 	private Object getInfo(UploadInfo uploadInfo) {
394     	String date = FastDateFormat
395 				.getInstance(Constants.TIMESTAMP_FORMAT)
396 				.format(uploadInfo.getCreationTimestamp());
397 		Map<String, Object> info = new TreeMap<>();
398 		info.put("manifest", uploadInfo.getMetadata());
399 		Map<String, Object> fileInfo = new TreeMap<>();
400 		fileInfo.put("size_bytes", uploadInfo.getLength());
401 		fileInfo.put("updated_at", date);
402 		info.put("file_info", fileInfo);
403 		
404 		Map<String, String> uploadStatus = new TreeMap<>();
405 		uploadStatus.put("status", "Complete");
406 		uploadStatus.put("chunk_received_at", date);
407 		info.put("upload_status", uploadStatus);
408 		
409 		Map<String, String> delivery = new TreeMap<>();
410 		delivery.put("delivered_at", date);
411 		delivery.put("status", "SUCCESS");
412 		delivery.put("name", "izgw");
413 		delivery.put("location", uploadInfo.getFileName());
414 		delivery.put("issues", null);
415 		info.put("deliveries", Collections.singletonList(delivery));
416 
417 		return info;
418 	}

420 	private String getBearerToken(HttpServletRequest servletRequest) {
421         String bearerToken = servletRequest.getHeader(HttpHeaders.AUTHORIZATION);
422         String[] parts;
423         if (bearerToken != null) {
424             bearerToken = bearerToken.trim();
425             parts = bearerToken.split("\\s+");
426             if ("Bearer".equals(parts[0]) && parts.length > 1) {
427                 return parts[1];
428             }
429         }
430         throw new HttpStatusException(HttpServletResponse.SC_UNAUTHORIZED);
431     }
