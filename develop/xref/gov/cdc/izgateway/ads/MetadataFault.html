4   import gov.cdc.izgateway.model.RetryStrategy;
5   import gov.cdc.izgateway.soap.fault.Fault;
6   import gov.cdc.izgateway.soap.fault.FaultSupport;
7   import gov.cdc.izgateway.soap.fault.MessageSupport;
8   
9   import org.apache.commons.lang3.StringUtils;
10  
11  /** A fault that reports errors in the metadata of a file submission. */
12  public class MetadataFault extends Fault implements FaultSupport {
13      private static final long serialVersionUID = 1L;
14  
15      private static final MessageSupport[] MESSAGE_TEMPLATE = {
16          new MessageSupport("MetadataFault", "70", "Error in Metadata", null,
17              "An illegal value was provided for metadata", RetryStrategy.CORRECT_MESSAGE),
18          new MessageSupport("MetadataFault", "71", "Multiple Errors in Metadata", null,
19              "Illegal values were provided for metadata", RetryStrategy.CORRECT_MESSAGE),
20      };
21      static {
22      	for (MessageSupport m: MESSAGE_TEMPLATE) {
23      		MessageSupport.registerMessageSupport(m);
24      	}
25      }
26      /** The metadata that has the errors. */
27      private final Metadata meta;
28  
29      /**
30       * Creates a metadata fault.
31       * @param meta the metadata that has the errors
32       * @param errors the error messages
33       */
34      public MetadataFault(Metadata meta, String ... errors) {
35          this(meta, null, errors);
36      }
37      /**
38       * Creates a metadata fault with a cause.
39       * @param meta the metadata that has the errors
40       * @param cause the exception that caused the fault, or null
41       * @param errors the error messages
42       */
43      public MetadataFault(Metadata meta, Throwable cause, String ... errors) {
44          super(MESSAGE_TEMPLATE[errors.length > 1 ? 1 : 0].copy().setDetail(StringUtils.join(errors, "\n")), cause);
45          this.meta = meta;
46          if (cause != null) {
47              initCause(cause);
48          }
49      }
50  
51      /**
52       * Gets the metadata that has the errors.
53       * @return the metadata
54       */
55      public Metadata getMeta() {
56          return meta;
57      }
58  
59      @Override
60      public String getSummary() {
61          return messageSupport.getSummary();
62      }
63  
64      @Override
65      public String getDetail() {
66          return messageSupport.getDetail();
67      }
68  
69      @Override
70      public String getDiagnostics() {
71          return messageSupport.getDiagnostics();
72      }
73  
74      @Override
75      public String getCode() {
76          return messageSupport.getCode();
77      }
78  
79      @Override
80      public RetryStrategy getRetry() {
81          return messageSupport.getRetry();
82      }
83  
84      @Override
85      public String getFaultName() {
86          return messageSupport.getFaultName();
87      }
88  }
