81      public MetadataImpl(Metadata resp) {
82          if (resp == null) {
83              return;
84          }
85          routeId = resp.getRouteId();
86          period = resp.getPeriod();
87          extSource = resp.getExtSource();
88          extSourceVersion = resp.getExtSourceVersion();
89          extEvent = resp.getExtEvent();
90          extEventType = resp.getExtEventType();
91          extEntity = resp.getExtEntity();
92          username = resp.getUsername(); 
93          filename = resp.getFilename();
94          extObjectKey = resp.getExtObjectKey();
95          ipAddress = resp.getIpAddress();
96          path = resp.getPath();
97          schemaVersion = resp.getSchemaVersion();
98          fileSize = resp.getFileSize();
99          uploadedDate = resp.getUploadedDate();
100         destinationId = resp.getDestinationId();
101         eventId = resp.getEventId();
102     	submissionStatus = resp.getSubmissionStatus();
103     	submissionLocation = resp.getSubmissionLocation();
104     	isTestFile = resp.isTestFile();
105 
106         if (resp instanceof MetadataImpl r2) {
107         	destination = r2.getDestination();
108         	dataStreamId = r2.getDataStreamId();
109         }
110     }

208 	private static final List<String> MONTHS = Arrays.asList("JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC");
209 	private static final List<String> QUARTERS = Arrays.asList("Q1", "Q2", "Q3", "Q4");
210 	/**
211 	 * Get the specified period in a Calendar class 
212 	 * @return	A calendar representing the reporting period date. 
213 	 */
214 	@JsonIgnore
215 	public Calendar getPeriodAsCalendar() {
216 		Calendar metaDate = Calendar.getInstance();
217 		metaDate.setTimeInMillis(System.currentTimeMillis());
218 		int year = -1;
219 		int month = -1;
220 		if (StringUtils.isNotEmpty(period)) {
221 			try {
222 				year = Integer.parseInt(StringUtils.substring(period, 0, 4));
223 			} catch (NumberFormatException ex) {
224 				// Ignore it.
225 			}
226 			if (Strings.CI.contains(getPeriod(), "Q")) {
227 				month = QUARTERS.indexOf(StringUtils.right(period.toUpperCase(), 2)) * 3;
228 			} else if (MetadataBuilder.ANNUAL.equals(MetadataBuilder.classifyPeriodFrequency(period.toUpperCase()))) {
229 				// Annual periods (bare YYYY) have no month component. Default to January
230 				// instead of leaving this Calendar's month at whatever "now" happened to
231 				// be -- callers that compare by month (rather than ignoring it, as
232 				// MetadataBuilder.checkDate() does for annual types) get a deterministic
233 				// value instead of one that varies by time of day.
234 				month = 0;
235 			} else {
236 				month = MONTHS.indexOf(StringUtils.right(period.toUpperCase(), 3));
237 			}
238 		}
239 		// Set to first of month because if you run this based on "today"
240 		// and today happens to be 7/31, but your actual month is September, when
241 		// you change month to 9, then 9/31 will roll over to 10/1.
242 		metaDate.set(Calendar.DATE, 1);
243 		if (year >= 0) {
244 			metaDate.set(Calendar.YEAR, year);
245 		}
246 		if (month >= 0) {
247 			metaDate.set(Calendar.MONTH, month);
248 		}
249 		return metaDate;
250 	}
