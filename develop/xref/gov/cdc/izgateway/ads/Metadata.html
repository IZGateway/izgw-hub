13  /**
14   * Object used to store metadata.  JSON Names are the same values as would be used
15   * in the Azure metadata.
16   * 
17   * Data Element Name		Data Element Description	Purpose	Optionality	Level 
18   * meta_destination_id	Set to - "NDLP"	Establishes that this file should be directed to CDC NDLP.	Required	File Level
19   * 
20   * meta_ext_source	Set to - "IZGW"	Establishes where data came from.	Required	File Level
21   * 
22   * meta_ext_sourceversion	V2022-12-31	Aligns with version of the specification.	Required	File Level
23   * 
24   * meta_ext_event	Set to:
25   * o    "routineImmunization" - for Routine Immunization files
26   * o    "influenzaVaccination" - for Influenza files	Determines storage container for IZGW to submit files on NDLP Storage Account	Required	File Level
27   * 
28   * meta_ext_entity	This is the three digit entity code.  See tab - Submitting Entity Codes, e.g., 
29   * o 	NYA (New York State)
30   * o 	CV1 (CVS Health Corporation)
31   * o 	DD2 (Department of Defense)	Drives metrics for vaccine admin progress at  entity grouping level.	Required	File Level
32   * 
33   * meta_username	Username of user who submitted the file to the IZGW.  If this is a system to system submission then this should be the service account.	This field is used for troubleshooting.	Required	File Level
34   * 
35   * meta_ext_objectkey	81bc2718-aef5-479b-8d67-955a2e75ac53 	This field is used to track back to the source filename / objectid.	Required	File Level
36   * 
37   * meta_ext_filename	The name of the file submitted to the IZGW.  File and extension should be named in accordance with guidelines outlined in the DUA.	Tracking the name of the file that was submitted.	Required	File Level
38   * 
39   * meta_ext_submissionperiod	This is the actual time period associated with submission for this file:
40   * o    YYYYQ#  - For Routine Immunization quarterly 
41   *       submissions (e.g., 2022Q1)
42   * o    YYYY-MON - for Influenza Vaccination monthly 
43   *       submissions (e.g., 2023-FEB)	Tracks frequency associated with file submission, as well as the actual file submission timeframe based on the frequency requirements associated with the program (i.e., Quarterly for Routine Immunizations, and Monthly for Influenza Vaccinations)	Required	File Level
44   *       
45   * meta_schema_version	This is an optional metadata field used for DEX Uploads.  It represents the version of the metadata  associated with the file being uploaded to DEX. If the optional meta_schema_version metadata field is not provided, then the earliest version found in the metadata config file for the given meta_destination_id will be used.	Indicates the version of metadata for the file being uploaded to DEX. 	Optional	File Level
46   * 
47   *  
48   */

89  	/**
90  	 * Gets the DEX metadata version (version). DEX requires this field.
91  	 * @return "1.0" for {@link #DEX_VERSION1}, else "2.0"
92  	 */
93  	@JsonProperty("version")
94  	default String getVersion() {
95  		switch (StringUtils.defaultString(getExtSourceVersion())) {
96  		case DEX_VERSION1:
97  			return "1.0";
98  		case DEX_VERSION2:
99  		default:
100 			return "2.0";
101 		}
102 	}

428             equals(this.getExtEntity(), that.getExtEntity()) &&

430             equals(this.getExtEventType(), that.getExtEventType()) &&
431             equals(this.getExtObjectKey(), that.getExtObjectKey()) &&
432             equals(this.getExtSource(), that.getExtSource()) &&
433             equals(this.getExtSourceVersion(), that.getExtSourceVersion()) &&
434             equals(this.getFilename(), that.getFilename()) &&
435             this.getFileSize() == that.getFileSize() &&
436             equals(this.getUploadedDate(), that.getUploadedDate()) &&
437             equals(this.getUsername(), that.getUsername()) &&
438             equals(this.getIpAddress(), that.getIpAddress()) &&
439             equals(this.getFilename(), that.getExtEntity()) &&
