34       * Noise words stripped from submitted reportType values when performing
35       * fuzzy matching against the registry. This allows legacy submissions such
36       * as {@code "farmerFlu"} to match the canonical entry {@code "farmerFluVaccination"}.
37       */
38      protected static final List<String> NOISE_WORDS =
39              List.of("vaccination", "immunization", "prevention", "monthly", "quarterly");

42      }

44      /**
45       * Strip all occurrences of {@link #NOISE_WORDS} from {@code s} (case-insensitively)
46       * and return the result in lower-case.
47       *
48       * @param s the input string
49       * @return lower-cased string with noise words removed
50       */
51      public static String stripNoiseWords(String s) {
52          String result = s.toLowerCase();
53          for (String noise : NOISE_WORDS) {
54              result = result.replace(noise, "");
55          }
56          return result;
57      }

59      /**
60       * Find a matching event type from the registry for the submitted reportType using
61       * three-tier matching: exact, case-insensitive, then noise-word stripped.
62       *
63       * @param reportType the submitted reportType value
64       * @param eventTypes the registered event type names
65       * @return the matched canonical event type, or empty if no match found
66       */
67      public static Optional<String> matchReportType(String reportType, Collection<String> eventTypes) {
68          // Tier 1: exact match
69          if (eventTypes.contains(reportType)) {
70              return Optional.of(reportType);
71          }
72          // Tier 2: case-insensitive match
73          Optional<String> match = eventTypes.stream()
74                  .filter(e -> e.equalsIgnoreCase(reportType))
75                  .findFirst();
76          if (match.isPresent()) {
77              return match;
78          }
79          // Tier 3: noise-word stripped match
80          String strippedSubmitted = stripNoiseWords(reportType);
81          return eventTypes.stream()
82                  .filter(e -> stripNoiseWords(e).equals(strippedSubmitted))
83                  .findFirst();
84      }

87       * Given a Facility Identifier, get it's associated principal identifier.
88       * @param facilityId	The facility ID
89       * @param principal	The name of the principal
90       * @return	A string identifying the facility.
91       */
92      public static String mapFacilityId(String facilityId, String principal) {
93          // If there is an entity id, then no mapping is needed, just validate it
94          if (facilityId != null) {
95              // Normalize the value
96              return facilityId.trim().toUpperCase();
97          } 
98          // Otherwise map from source data from certificate.
99          return mapFromOrganization(principal);
100     }

128     private static final char[] DISALLOWED = {

133     /**
134      * Check filenames for invalid characters
135      * @param filename	The filename
136      * @return	True if no invalid characters are found.
137      */
138     public static boolean validateFilename(String filename) {
139         if (StringUtils.isEmpty(filename) || filename.length() >= 1024) {
140             return false;
141         }
142         return filename.codePoints().allMatch(c -> c > 0x1f && c < 0xd7ff && Arrays.binarySearch(DISALLOWED, (char)c) < 0);
143     }

156     /**
157      * Construct the file path to a blob from it's metadata.
158      * NOTE: This is the original NDLP v1 Construction, this might change 
159      * for most recent code base
160      *  
161      * @param base	The base URI
162      * @param meta	The metadata for the submission
163      * @return	The path to use for the blob
164      */
165     public static String getPath(URI base, Metadata meta) {
166         long now = System.currentTimeMillis();
167         if (meta.getExtSourceVersion().equals(Metadata.DEX_VERSION2)) {
168         	return String.format("%s/%s/%s/%tY/%tm/%td/%s", 
169     			base.getPath(), 
170     			meta.getDataStreamId(), 
171     			meta.getExtEntity(), 
172     			now, now, now,
173     			meta.getFilename()
174         	);
175         }
176         return String.format(
177             "%s/%s/%s/%tY%tm%td/%s", 
178             base.getPath(),
179             meta.getPeriod(),
180             meta.getExtEntity(),
181             now, now, now,
182             meta.getFilename()
183         );
184     }

186     /**
187      * Azure tokens may be locked to an IP Address. For dev and test environments, we have TWO
188      * different IP addresses requests can come from. This method will extract the password
189      * based on the IP egress address of the system that is currently running.
190      * 
191      * This is a bit of a hack, but is ONLY needed for dev and test environments that emerge
192      * from multiple regions or zones.  If more than two egress addresses are used, the DB
193      * might need to be updated to support longer password strings.
194      * 
195      * @param password A comma-separated list of SAS tokens
196      * @return	The appropriate token to use for an Azure blob store.
197      * @throws SecurityFault If no token matches the egress IP address of this system
198      */
199 	public static String getAzureToken(String password) throws SecurityFault {
200 		String[] tokens = password.split(",");
201 		for (String token: tokens) {
202 			String ipAddress = StringUtils.substringBetween(token, "sip=", "&");
203 			if (StringUtils.isEmpty(ipAddress)) {
204 				// This token is NOT locked to an IP Address 
205 				log.debug("Did not find a token matching {}", MY_IP_ADDRESS);
206 				return token;
207 			}
208 			if (ipAddress.equals(MY_IP_ADDRESS) || ipAddress.contains("-") && myIpInRange(ipAddress)) {
209 				// This token is locked to my IP Address or to a range of IP addresses that my IP Address is in. 
210 				log.debug("Found token for {} with parameters {}", MY_IP_ADDRESS, StringUtils.substringBefore(token, "&sig="));
211 				return token;
212 			}
213 		}
214 		throw SecurityFault.generalSecurity("No token found for IP Address " + MY_IP_ADDRESS, null, null);
215 	}

217 	/**
218 	 * Verify that MY_IP_ADDRESS is within the range specified by ipAddress.
219 	 * 
220 	 * @param ipRange	The range
221 	 * @return true if within the range
222 	 */
223 	private static boolean myIpInRange(String ipRange) {
224 		String[] range = ipRange.split("-");
225 		if (range.length != 2) {
226 			throw new IllegalArgumentException("Not an IP Address Range: " + ipRange);
227 		}
228 		try {
229 			InetAddress low = InetAddress.getByName(range[0]);
230 			InetAddress high = InetAddress.getByName(range[1]);
231 			InetAddress myAddr = InetAddress.getByName(MY_IP_ADDRESS);
232 			byte[] mine = myAddr.getAddress();
233 			if (low.getClass() == high.getClass() && low.getClass() == myAddr.getClass()) {
234 				return 
235 					Arrays.compareUnsigned(low.getAddress(), mine) <= 0 &&
236 					Arrays.compareUnsigned(high.getAddress(), mine) >= 0;
237 			}
238 			return false; // Not same class of IP address, my address is v4 and range is v6 or vice versa
239 		} catch (UnknownHostException e) {
240 			throw new IllegalArgumentException("Unknown host exception checking " + ipRange, e);
241 		}
242 	}

244 	/**
245 	 * Call a web service to get this server's egress IP Address.
246 	 * Tries checkip.amazonaws.com and ipv4.icanhazip.com, both of which should be highly reliable.
247 	 * 
248 	 * @return The egress IP Address of this system.
249 	 */
250     public static String getMyIpAddress() {
251     	String[] urlStrings = { "http://checkip.amazonaws.com/", "http://ipv4.icanhazip.com/" };
252     	Throwable ex = null;
253     	for (String urlString: urlStrings) {
254 	    	try (BufferedReader br = new BufferedReader(new InputStreamReader(new URL(urlString).openStream()))) {
255 	    	    return br.readLine();
256 	    	} catch (Exception e) {
257 	    		if (ex == null) {
258 	    			ex = e;
259 	    		}
260 	    	}
261     	}
262     	throw new ServiceConfigurationError("Cannot get IP Address", ex);
263 	}
