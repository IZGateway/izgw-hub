18  /**
19   * Contract for Blob Storage sender interface to enable reuse of this
20   * with other blob stores (e.g., Azure, S3 bucket).
21   */
22  public interface FileSender {
23      /**
24       * Sends a file to the destination.
25       * @param route the destination
26       * @param dataHandler the file data
27       * @param meta the metadata of the file
28       * @return the connection that sent the file
29       * @throws MessageTooLargeFault if the file is larger than the maximum upload size
30       * @throws HubClientFault if the destination returns an error or the request is not valid
31       * @throws DestinationConnectionFault if the hub cannot connect to the destination
32       * @throws MetadataFault if the metadata is not valid
33       * @throws SecurityFault if a security error occurs
34       */
35      HttpURLConnection sendFile(IDestination route, DataHandler dataHandler, Metadata meta) throws MessageTooLargeFault, HubClientFault, DestinationConnectionFault, MetadataFault, SecurityFault;
36      /**
37       * Checks the connection to the destination.
38       * @param r the destination
39       * @return the URI of the destination
40       * @throws HubClientFault if the destination returns an error or the request is not valid
41       * @throws MetadataFault if the metadata is not valid
42       * @throws DestinationConnectionFault if the hub cannot connect to the destination
43       * @throws SecurityFault if a security error occurs
44       */
45      String getStatus(IDestination r) throws HubClientFault, MetadataFault, DestinationConnectionFault, SecurityFault;
46      /**
47       * Gets a file from the destination.
48       * @param r the destination
49       * @param meta the metadata that identifies the file
50       * @return the file content and the HTTP response headers
51       * @throws MetadataFault if the metadata is not valid
52       * @throws HubClientFault if the destination returns an error or the request is not valid
53       * @throws DestinationConnectionFault if the hub cannot connect to the destination
54       * @throws SecurityFault if a security error occurs
55       */
56      Pair<InputStream, Map<String, List<String>>> getFile(IDestination r, Metadata meta) throws MetadataFault, HubClientFault, DestinationConnectionFault, SecurityFault;
57      /**
58       * Deletes a file at the destination.
59       * @param route the destination
60       * @param meta the metadata that identifies the file
61       * @return the result of the delete operation
62       * @throws MetadataFault if the metadata is not valid
63       * @throws HubClientFault if the destination returns an error or the request is not valid
64       * @throws DestinationConnectionFault if the hub cannot connect to the destination
65       * @throws SecurityFault if a security error occurs
66       */
67      String deleteFile(IDestination route, Metadata meta) throws MetadataFault, HubClientFault, DestinationConnectionFault, SecurityFault;
68  	/**
69  	 * Gets the status of a file submission from the destination.
70  	 * @param r the destination
71  	 * @param m the metadata that identifies the submission
72  	 * @return the submission status
73  	 * @throws DestinationConnectionFault if the hub cannot connect to the destination
74  	 * @throws MetadataFault if the metadata is not valid
75  	 * @throws HubClientFault if the destination returns an error or the request is not valid
76  	 * @throws SecurityFault if a security error occurs
77  	 */
78  	String getSubmissionStatus(IDestination r, Metadata m) throws DestinationConnectionFault, MetadataFault, HubClientFault, SecurityFault;
79  }
