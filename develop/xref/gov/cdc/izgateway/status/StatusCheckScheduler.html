25  /**
26   * This class is used to check the status of existing connections.
27   * It wakes up every 5 minutes and checks anything that it hasn't checked
28   * whose age is older than frequencyInMinutes, and everything that hasn't been
29   * checked in 5 minutes + frequencyInMinutes.
30   */
31  @Slf4j
32  @Component
33  @Lazy(false)
34  public class StatusCheckScheduler {
35      /** How often should an instance resync with other instances on the schedule */
36      private static final Duration RESYNC_DURATION = Duration.ofMinutes(5); // Every 5 minutes
37      private static final int RESYNCS_PER_CYCLE = 3;
38      /** How long should it take to check everything, must be a multiple of RESYNC_DURATION */
39      private static final Duration CYCLE_DURATION = Duration.ofMinutes(RESYNC_DURATION.toMinutes() * RESYNCS_PER_CYCLE); // Every 15 minutes
40      /** How long should it take for all nodes to start up */
41      private static final Duration STARTUP_ALLOWANCE = Duration.ofMinutes(3); // Three minutes
42      private static final FastDateFormat TIMESTAMP_FORMATTER = FastDateFormat.getInstance(Constants.TIMESTAMP_FORMAT);
43      private static final boolean DISABLED = false;  // Set to true to disable for debugging purposes.
44      private final StatusCheckerService statusCheckerService;
45      private final EndpointStatusService endpointStatusService;
46      private final DestinationService dests;
47  	private final ScheduledExecutorService scheduler;
48  
49      /**
50       * Make the scheduler. Spring supplies the services.
51       * @param statusCheckerService The service that checks destinations and supplies the scheduler.
52       * @param endpointStatusService The service that keeps the endpoint status.
53       * @param dests The service that gets the destinations.
54       */
55      @Autowired
56      public StatusCheckScheduler(StatusCheckerService statusCheckerService, EndpointStatusService endpointStatusService, DestinationService dests) {
57          this.statusCheckerService = statusCheckerService;
58          this.endpointStatusService = endpointStatusService;
59          this.dests = dests;
60          this.scheduler = statusCheckerService.getScheduler();
61      }

63      /**
64       *	Bridges the diagnostic lookup in the repository.
65       * @param faultName The name of the fault
66       * @param faultCode The code of the fault
67       * @return The diagnostic string for the fault
68       */
69      public String getDiagnostics(String faultName, String faultCode) {
70          FaultSupport s = MessageSupport.getTemplate(faultCode, faultName);
71          return s == null ? null : s.getDiagnostics();
72      }

74      /**
75       * Start the status checks. Testing endpoints are exempt from the checks.
76       * The method loads the endpoint status, checks the dev destination, and schedules
77       * a resync every 5 minutes.
78       * @throws ServiceConfigurationError When the endpoint status cannot be loaded.
79       */
80      public void start() {
81          log.info("Starting Status Checker");
82          // Exempt testing endpoints from status checks, a) don't waste time, these will fail, and don't circuit break them
83          // because then we'll need to reset the breaker, and the only way to do that is to get a good connection, which can
84          // never happen with these.
85          dests.getAllDestinations().stream().filter(this::isTestingEndpoint).forEach(d -> statusCheckerService.addExemption(d.getDestId()));
86          // Initialize status from Elastic.  If this fails (e.g., b/c of a configuration error), ensure IZGW won't start
87          // properly.
88          if (!endpointStatusService.refresh()) {
89              log.error("Could not initialize Status Service");
90              throw new ServiceConfigurationError("Could not initialize Status Service");
91          }
92      	statusCheckerService.checkDestination(dests.findByDestId("dev"));
93  
94          scheduler.scheduleAtFixedRate(this::setSchedule, getNextSyncTime(), RESYNC_DURATION.toMillis(), TimeUnit.MILLISECONDS);
95          log.info("Status Checker started");
96      }
97  
98      /**
99       * The StatusChecker is going to run at least every 5 minutes, and all operating nodes will sync up
100      * every five minutes on the schedule.
101      *
102      * @return	The next time to sync up, which is every 5 minutes starting at the hour, but not less than two minutes into the future
103      */
104     private long getNextSyncTime() {
105         long resyncTime = RESYNC_DURATION.toMillis();
106         // next = next five minute interval that is no less than 3 minutes into the future, but no more than 8 minutes.
107         long nextSyncTime = resyncTime - (System.currentTimeMillis() % resyncTime);
108         if (nextSyncTime < STARTUP_ALLOWANCE.toMillis()) {
109             nextSyncTime += RESYNC_DURATION.toMillis();
110         }
111         return nextSyncTime;
112     }
113 
114     private void setSchedule() {
115         Map<String, String> map = StatusCheckerService.setupMDC();
116         try {
117             // First thing to do is refresh status and destinations.
118             log.debug("Updating schedule");
119             endpointStatusService.refresh();
120             dests.refresh();
121 
122             List<IDestination> dList = Arrays.asList(
123                     dests.getAllDestinations().stream()
124                             .filter(this::isInteresting).toArray(len -> new IDestination[len])
125             );
126             IEndpointStatus[] sList = endpointStatusService.findAll().toArray(new IEndpointStatus[0]);
127             List<String> hosts = statusCheckerService.getHosts();
128 
129             // OK, at two minutes into the future, all nodes have generally agreed on:
130             // which destinations need to be checked.
131             // what the current status of those destinations are.
132 
133             int myInstanceOffset = 0;
134             for (IEndpointStatus s: sList) {
135                 if (!hosts.contains(s.getStatusBy())) {
136                     if (s.getStatusBy().equalsIgnoreCase(SystemUtils.getHostname())) {
137                         myInstanceOffset = hosts.size();
138                     }
139                     hosts.add(s.getStatusBy());
140                 }
141             }
142             // Hosts will take a position in sorted order by name.
143             hosts.sort(String::compareTo);
144             dList.sort((ths,that) -> ths.getDestId().compareTo(that.getDestId()));
145 
146             buildMySchedule(dList, hosts, myInstanceOffset);
147         } finally {
148         	StatusCheckerService.restoreMDC(map);
149         }
150     }

152     /**
153      * Report on destinations that a status check will serve some purpose for.
154      * - Nodes that aren't under maintenace,
155      * - operated by THIS code
156      * - or specifically exempt b/c they don't support that API call yet
157      * @param d	The destination to check
158      * @return
159      */
160     private boolean isInteresting(IDestination d) {
161         boolean isUnderMaintenance = d.isUnderMaintenance();
162         boolean isWorthChecking = isNotLocalOrInvalid(d);
163         boolean isExempt = isExempt(d.getDestId());
164         return !isUnderMaintenance && isWorthChecking && !isExempt;
165     }

167     /**
168      * Build the schedule of status checks for this instance.
169      * Assumptions:  All instances are up and running, and have agreed upon the dList and hosts available.
170      * @param dList
171      * @param hosts
172      * @param myInstanceOffset
173      */
174     private void buildMySchedule(List<IDestination> dList, List<String> hosts, int myInstanceOffset) {
175         log.debug("Building status check schedule for {}", SystemUtils.getHostname());
176         long now = System.currentTimeMillis();
177         int numberOfDestsToCheckPerSync = (dList.size() + RESYNCS_PER_CYCLE - 1) / RESYNCS_PER_CYCLE;
178         long periodBetweenChecks = RESYNC_DURATION.toMillis() / numberOfDestsToCheckPerSync;
179         /** whatPartOfListToCheck will be an integer between 0 and RESYNCS_PER_CYCLE - 1 */
180         int whatPartOfListToCheck = (int) ((now % CYCLE_DURATION.toMillis()) / RESYNC_DURATION.toMillis());
181         int startingDestOffset = numberOfDestsToCheckPerSync * whatPartOfListToCheck;
182 
183         for (int i = 0; i < numberOfDestsToCheckPerSync; i += hosts.size()) {
184             int index = i + myInstanceOffset + startingDestOffset;
185             if (index >= dList.size()) {
186                 break;
187             }
188             scheduler.schedule(
189                     () -> checkDestination(dList.get(index).getDestId()),
190                     (index+1) * periodBetweenChecks, TimeUnit.MILLISECONDS
191             );
192 
193             log.debug(
194                     "Status Check on {} scheduled for {} by {}",
195                     dList.get(index).getDestId(),
196                     TIMESTAMP_FORMATTER.format(now + (index+1) * periodBetweenChecks),
197                     SystemUtils.getHostname()
198             );
199         }
200     }

202     private boolean isExempt(String destId) {
203         return statusCheckerService.getConfig().getExempt().contains(destId) ||
204         	   statusCheckerService.getConfig().getTestingEndpoints().contains(destId);
205     }
206 
207     private IEndpointStatus checkDestination(String destId) {
208         long minRecentCheck = System.currentTimeMillis() - RESYNC_DURATION.toMillis();
209         // Get the most current configuration.
210         IDestination dest = dests.findByDestId(destId);
211         // And the most recent status.
212         IEndpointStatus status = endpointStatusService.findById(destId);
213 
214         if (DISABLED) {
215         	return status;
216         }
217         // If there is a status, and it is connected, and it is recent
218         if (status != null && status.isConnected() && status.getStatusAt().getTime() > minRecentCheck) {
219             // Just return the current status, we don't need a check for endpoints that are in frequent use and working.
220             return status;
221         }
222 
223         // But, if we don't know the status, or the last status was not recent, or it wasn't connected, check it.
224         status = statusCheckerService.checkDestination(dest);
225 
226         return status;
227     }

229     private static final String[] NOT_WORTH_CHECKING_PREFIXES = {
230         "/", // Local URI
231         "https://localhost", // Also a Local URI
232         "https://192.0", // A test URI
233         "https://iis.invalid" // A test URI
234     };
235 
236     private boolean isTestingEndpoint(IDestination d) {
237         return statusCheckerService.getConfig().getTestingEndpoints().contains(d.getDestId());
238     }
239 
240     private boolean isNotLocalOrInvalid(IDestination d) {
241         if (d == null) {
242             return false;
243         }
244         // We need not check endpoints which are local to this server.
245         String uri = d.getDestUri();
246         for (String prefix: NOT_WORTH_CHECKING_PREFIXES) {
247             if (Strings.CS.startsWith(uri, prefix)) {
248                 return false;
249             }
250         }
251         return true;
252     }
