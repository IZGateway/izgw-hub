14  /**
15   * DynamoDB entity for an API key credential. The {@code jti} identifies the credential, and the
16   * record holds its status, its permitted environments and use-types, and its lifecycle timestamps.
17   */
18  // javadoc reads the source without Lombok, so it does not see the constructor that
19  // Lombok generates (@NoArgsConstructor). It then reports "use of default constructor, which does
20  // not provide a comment". The real constructor exists. This suppression hides only
21  // missing-comment warnings in this class (IGDD-2974). Other javadoc checks still apply.
22  @SuppressWarnings("doclint:missing")
23  @Data
24  @EqualsAndHashCode(callSuper = false)
25  @NoArgsConstructor
26  @DynamoDbBean
27  public class ApiKeyCredential extends DynamoDbAudit implements DynamoDbEntity {
28  
29      private String jti;
30      // Server-side set of numeric environment IDs (values 1-6 per the IZG Environment enumeration) in
31      // which this credential is valid. Written by Config Console (IGDD-3140); read by Hub at routing time.
32      // Environment authorization is NOT carried in the JWT — Hub looks the credential up by jti and checks
33      // that its own env (SystemUtils.getDestType()) is contained in this set. The set can change without
34      // rewriting the sort key. Standard credentials contain exactly one ID; admin/operational credentials
35      // MAY contain several.
36      //
37      // Persisted as a DynamoDB Number Set (NS). DynamoDB cannot store an empty set, so "no environments"
38      // is represented by the attribute being absent, which deserializes to null here — null and empty are
39      // therefore equivalent and both deny. See ApiKeyPrincipalProvider#lookupAndCacheCredential.
40      private Set<Integer> environments;
41      // Server-side set of use-types (PATIENT / PROVIDER / PUBLIC_HEALTH per gov.cdc.izgateway.hub.security.UseType)
42      // this credential may submit. Written by Config Console (IGDD-3140), which validates the values against the
43      // enumeration; read by Hub at routing time and intersected with the destination jurisdiction's
44      // allowedUseTypes (IGDD-3257). Like `environments`, it is NOT carried in the JWT, so policy can change
45      // without reissuing the credential.
46      //
47      // Persisted as a DynamoDB String Set (SS). As with `environments`, an absent attribute deserializes to
48      // null, and null and empty are equivalent: a credential with no useTypes is authorized for nothing.
49      private Set<String> useTypes;
50      private String status;
51      private String jurisdictionId;
52      // Serialized as ISO-8601 UTC strings (e.g. 2025-06-04T00:00:00Z) by the AWS SDK Enhanced Client's
53      // default InstantAsStringAttributeConverter. Distinct from the Date fields in DynamoDbAudit.
54      private Instant issuedAt;
55      private Instant expiresAt;
56      private Instant revokedAt;
57      private String revokedBy;
58      // Written by Hub's grace-period sweep (IGDD-3167) when a grace-expired credential reached its own
59      // expiresAt on/before graceExpiresAt. Mutually exclusive with revokedAt/revokedBy: a credential
60      // transitions to exactly one terminal status, so only the matching pair is ever populated.
61      private Instant expiredAt;
62      private String expiredBy;
63      // Written by Config Console on renewal (IGDD-2707) onto the superseded (old) credential, which is
64      // moved to status "grace_period"; read by Hub's grace-period revocation job (IGDD-2711).
65      // graceExpiresAt is the instant after which the grace-period credential becomes eligible for
66      // automated revocation; supersededBy is the jti of the renewed credential that replaced it. Both
67      // are null on credentials that have never been renewed. graceExpiresAt serializes as an ISO-8601
68      // UTC string like the other Instants.
69      private Instant graceExpiresAt;
70      private String supersededBy;
71  
72      @Override
73      public String getPrimaryId() {
74          // Sort key is the jti alone (no environment prefix); permitted environments are the server-side
75          // `environments` set rather than being encoded in the key (IGDD-3140).
76          return jti;
77      }
78  }
