23  /**
24   * Supports Event Tracking
25   *
26   * @author Audacious Inquiry
27   */
28  @SuppressWarnings("serial")
29  @Data
30  @EqualsAndHashCode(callSuper=true)
31  @AllArgsConstructor
32  @NoArgsConstructor
33  @DynamoDbBean
34  public class Event extends DynamoDbAudit implements DynamoDbEntity, Serializable {
35  	// Well known events.
36  	/** The Migration event */
37  	public static final String MIGRATION = "Migration";
38  	/** The Startup event */
39  	public static final String STARTUP = "Startup";
40  	/** The Shutdown event */
41  	public static final String SHUTDOWN = "Shutdown";
42  	/** The Global Table Creation event */
43  	public static final String CREATED = "Created";
44  	
45  	/**
46  	 * Provides easy access to the Map class for Swagger documentation
47  	 */
48  	public static class Map extends MappableEntity<Event>{
49  		/**
50  		 * Create an empty map.
51  		 */
52  		public Map() {
53  			// Nothing to initialize.
54  		}
55  	}
56  
57      /** The event name. */
58      @Schema(description = "The event name")
59      private String name;
60  
61      /** The target of the event. */
62      @Schema(description = "The target of the event")
63      private String target;
64      
65      /** The start timestamp. */
66      @Schema(description = "Start timestamp")
67      private Date started = getCreatedOn();
68      
69      /** The completed timestamp. */
70      @Schema(description = "Completed timestamp")
71      private Date completed;
72      
73      /** The system that reports the event. */
74      @Schema(description = "System reporting the event")
75      private String reportedBy = getCreatedBy();
76      
77      /** The unique identifier of the event. */
78      @Schema(description = "The Unique Id of the event")
79      private String eventId = UUID.randomUUID().toString();
80      
81      /**
82       * Create a new event.
83       * @param name	The name of the event (e.g., Migration)
84       */
85      public Event(String name) {
86      	this(name, null);
87      }
88      
89      /**
90       * Create a new event.
91       * @param name	The name of the event (e.g., Migration)
92       * @param target The target of the event.
93       */
94      public Event(String name, String target) {
95      	this.name = name;
96      	this.target = target;
97      }    
98      
99  	@Override
100 	public String getPrimaryId() {
101 		if (StringUtils.isEmpty(name)) {
102 			throw new IllegalArgumentException("Event name cannot be null or empty");
103 		}
104 		if (started == null) {
105 			started = new Date();
106 		}
107 		return name + "#" + StringUtils.defaultString(target) + "#" + DateConverter.convert(started);
108 	}
109 	
110     /**
111      * Get the start time of the event as a date
112      * @return	The start time of the event
113      */
114     @DynamoDbConvertedBy(DateConverter.class)
115     public Date getStarted() {
116     	return started;
117     }
118     
119     /**
120      * Get the completion time of the event as a date
121      * @return	The completion time of the event
122      */
123     @DynamoDbConvertedBy(DateConverter.class)
124     public Date getCompleted() {
125     	return completed;
126     }
127 }
