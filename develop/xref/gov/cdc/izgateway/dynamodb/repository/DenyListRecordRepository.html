15  /**

18   */
19  public class DenyListRecordRepository extends DynamoDbRepository<DenyListRecord> implements IDenyListRecordRepository<DenyListRecord> {
20      /**
21       * Constructs a new DenyListRecordRepository with the given DynamoDB client and table name.
22       * @param client the DynamoDB enhanced client
23       * @param tableName the name of the DynamoDB table
24       */
25      public DenyListRecordRepository(@Autowired DynamoDbEnhancedClient client, String tableName) {
26          super(DenyListRecord.class, client, tableName);
27      }
28  
29      /**
30       * Stores the given deny list record in DynamoDB.
31       * @param denyListRecord the deny list record to store
32       * @return the stored deny list record
33       */
34      @Override
35      public DenyListRecord store(DenyListRecord denyListRecord) {
36          return super.saveAndFlush(denyListRecord);
37      }
38  
39  	/**
40  	 * Migrate access controls to deny list records.
41  	 * @param list	The list of access controls
42  	 * @param who	The originator of the original access control record
43  	 * @param when	When that happened
44  	 */
45  	public void migrateAccessControls(List<? extends IAccessControl> list, String who, Date when) {
46  		for (IAccessControl ac : list) {
47  			if (!ac.getCategory().equals(IAccessControlService.GROUP_CATEGORY) || !ac.getName().equals("blacklist")) {
48  				continue;
49  			}
50  			DenyListRecord dlr = new DenyListRecord(ac, who, when);
51  			super.saveAndFlush(dlr);
52  		}
53  		
54  	}
55  }
