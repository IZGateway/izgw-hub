6   import org.springframework.boot.autoconfigure.condition.ConditionalOnExpression;

8   import org.springframework.context.annotation.Bean;
9   import org.springframework.context.annotation.Configuration;
10  import org.springframework.core.env.Environment;
11  import software.amazon.awssdk.services.secretsmanager.SecretsManagerClient;
12  
13  import java.time.Duration;
14  import java.util.Arrays;
15  import java.util.ServiceConfigurationError;
16  
17  /**
18   * Configuration properties for API key (JWT) authentication, bound from the {@code jwt} prefix.
19   */
20  @Configuration
21  @ConfigurationProperties(prefix = "jwt")
22  @Data
23  public class JwtConfig {
24  
25      /**
26       * Create the configuration. Spring sets the fields.
27       */
28      public JwtConfig() {
29          // Nothing to initialize.
30      }
31  
32      private String issuer;
33      private String secretsManagerSecretName;
34      private Duration secretCacheTtl = Duration.ofHours(1);
35      private Duration credentialCacheTtl = Duration.ofMinutes(5);
36      /** Local dev only — bypasses Secrets Manager. Must not be set in any deployed (non-local) profile. */
37      private String testSecret;
38  
39      // Not a config property — injected by Spring. @Data generates a setter but @ConfigurationProperties
40      // will not bind to it because there is no "spring-environment" property key.
41      @Autowired
42      private Environment springEnvironment;
43  
44      @PostConstruct
45      void validateConfig() {
46          if (issuer == null || issuer.isBlank()) {
47              throw new ServiceConfigurationError(
48                      "jwt.issuer must be configured — set the JWT_ISSUER environment variable");
49          }
50          if (testSecret != null && !testSecret.isBlank()) {
51              boolean hasLocalProfile = Arrays.stream(springEnvironment.getActiveProfiles())
52                      .anyMatch(p -> p.startsWith("local"));
53              if (!hasLocalProfile) {
54                  throw new ServiceConfigurationError(
55                          "jwt.test-secret must not be set outside of a local Spring profile — " +
56                          "remove JWT_TEST_SECRET from the deployed task definition");
57              }
58          }
59      }
60  
61      /**
62       * Provide a SecretsManagerClient bean only when jwt.test-secret is absent or blank.
63       * When test-secret is present (local dev), SM is not needed; skipping this bean avoids
64       * AWS credential resolution failures in local environments.
65       *
66       * Note: the previous @ConditionalOnProperty(havingValue="") was a no-op — Spring treats
67       * an empty havingValue as "match any non-false value", so the bean was always created.
68       * {@code @ConditionalOnExpression} correctly evaluates the actual property value at refresh time.
69       * @return The Secrets Manager client
70       */
71      @Bean
72      @ConditionalOnExpression("'${jwt.test-secret:}'.isEmpty()")
73      public SecretsManagerClient secretsManagerClient() {
74          return SecretsManagerClient.create();
75      }
76  }
