29  /**
30   * Authenticates a request that has a Bearer-scheme API key (a signed JWT) and creates the
31   * {@link ApiKeyPrincipal} for it. It caches signing secrets and credential lookups.
32   */
33  @Component
34  @Slf4j
35  public class ApiKeyPrincipalProvider {
36  
37      private static final Duration MAX_TOKEN_LIFETIME = Duration.ofDays(366);
38      private static final long CLOCK_SKEW_SECONDS = 30;
39  
40      private final JwtConfig config;
41      private final ApiKeyCredentialRepository credentialRepository;
42      private final JwtTokenExtractor jwtTokenExtractor;
43      private final SecretsManagerClient secretsManagerClient;
44  
45      private static final Duration NEGATIVE_SECRET_CACHE_TTL = Duration.ofSeconds(60);
46  
47      private final Cache<String, String> secretCache;
48      private final Cache<String, Boolean> negativeSecretCache;
49      private final Cache<String, ApiKeyPrincipal> credentialCache;
50      private final Cache<String, Boolean> revokedCache;
51      private final Cache<String, Boolean> absentCache;
52  
53      /**
54       * Create the provider. Spring supplies the parameters.
55       * @param config	The JWT configuration
56       * @param credentialRepository	The repository that stores API key credentials
57       * @param jwtTokenExtractor	The extractor that gets the Bearer token from the request
58       * @param secretsManagerClient	The Secrets Manager client for signing secrets, or null if not available
59       */
60      @Autowired
61      public ApiKeyPrincipalProvider(
62              JwtConfig config,
63              ApiKeyCredentialRepository credentialRepository,
64              JwtTokenExtractor jwtTokenExtractor,
65              @Autowired(required = false) SecretsManagerClient secretsManagerClient
66      ) {
67          this.config = config;
68          this.credentialRepository = credentialRepository;
69          this.jwtTokenExtractor = jwtTokenExtractor;
70          this.secretsManagerClient = secretsManagerClient;
71  
72          this.secretCache = Caffeine.newBuilder()
73                  .expireAfterWrite(config.getSecretCacheTtl())
74                  .build();
75  
76          this.negativeSecretCache = Caffeine.newBuilder()
77                  .expireAfterWrite(NEGATIVE_SECRET_CACHE_TTL)
78                  .build();
79  
80          this.credentialCache = Caffeine.newBuilder()
81                  .expireAfterWrite(config.getCredentialCacheTtl())
82                  .build();
83  
84          this.revokedCache = Caffeine.newBuilder()
85                  .expireAfterWrite(MAX_TOKEN_LIFETIME)
86                  .build();
87  
88          this.absentCache = Caffeine.newBuilder()
89                  .expireAfterWrite(config.getCredentialCacheTtl())
90                  .build();
91      }

93      /**
94       * Get the principal for the API key in the request.
95       * @param request	The HTTP request
96       * @return The principal, or null if the request has no Bearer token
97       * @throws ApiKeyAuthenticationException	If the request has a Bearer token that fails authentication
98       */
99      public IzgPrincipal getPrincipal(HttpServletRequest request) {
100         // Step 1: Extract Bearer token — return null if absent (fallback to cert auth is still allowed;
101         // no API key was presented at all).
102         String token;
103         try {
104             token = jwtTokenExtractor.extractToken(request);
105         } catch (MissingJwtTokenException e) {
106             return null;
107         }
108 
109         // From here on, a Bearer-scheme Authorization header WAS presented. Every failure below throws
110         // ApiKeyAuthenticationException instead of returning null: a caller that presents an API key and
111         // fails authentication must not be silently retried against the client certificate.
112 
113         // Step 2: Parse JWT header only
114         SignedJWT signedJwt;
115         try {
116             signedJwt = SignedJWT.parse(token);
117         } catch (ParseException e) {
118             log.warn("JWT rejected: failed to parse token: {}", e.getMessage());
119             throw new ApiKeyAuthenticationException("Failed to parse JWT token: " + e.getMessage());
120         }
121 
122         // Step 3: Pre-check alg and iss before expensive operations
123         if (!JWSAlgorithm.HS256.equals(signedJwt.getHeader().getAlgorithm())) {
124             log.warn("JWT rejected: unsupported algorithm={}", signedJwt.getHeader().getAlgorithm());
125             throw new ApiKeyAuthenticationException("Unsupported algorithm=" + signedJwt.getHeader().getAlgorithm());
126         }
127 
128         JWTClaimsSet unverifiedClaims;
129         try {
130             unverifiedClaims = signedJwt.getJWTClaimsSet();
131         } catch (ParseException e) {
132             log.warn("JWT rejected: failed to parse claims: {}", e.getMessage());
133             throw new ApiKeyAuthenticationException("Failed to parse JWT claims: " + e.getMessage());
134         }
135 
136         if (!config.getIssuer().equals(unverifiedClaims.getIssuer())) {
137             log.warn("JWT rejected: issuer mismatch (expected={}, got={})", config.getIssuer(), unverifiedClaims.getIssuer());
138             throw new ApiKeyAuthenticationException("Issuer mismatch (expected=" + config.getIssuer() + ", got=" + unverifiedClaims.getIssuer() + ")");
139         }
140 
141         // Step 4: Resolve signing secret by kid
142         String kid = signedJwt.getHeader().getKeyID();
143         if (kid == null || kid.isBlank()) {
144             log.warn("JWT rejected: missing kid header");
145             throw new ApiKeyAuthenticationException("Missing kid header");
146         }
147         byte[] secretBytes = resolveSecret(kid);
148         if (secretBytes == null) {
149             log.warn("JWT rejected: unable to resolve signing secret for kid={}", kid);
150             throw new ApiKeyAuthenticationException("Unable to resolve signing secret for kid=" + kid);
151         }
152 
153         // Step 5: Verify HS256 signature
154         try {
155             if (!signedJwt.verify(new MACVerifier(secretBytes))) {
156                 log.warn("JWT signature verification failed for kid={}", kid);
157                 throw new ApiKeyAuthenticationException("Signature verification failed for kid=" + kid);
158             }
159         } catch (JOSEException e) {
160             log.warn("JWT verification exception for kid={}: {}", kid, e.getMessage());
161             throw new ApiKeyAuthenticationException("Verification exception for kid=" + kid + ": " + e.getMessage());
162         }
163 
164         // Step 6: Validate claims against verified payload
165         JWTClaimsSet claims = unverifiedClaims;
166         Date expiry = claims.getExpirationTime();
167         if (expiry == null || expiry.toInstant().plusSeconds(CLOCK_SKEW_SECONDS).isBefore(Instant.now())) {
168             log.warn("JWT rejected: expired exp={}", expiry);
169             throw new ApiKeyAuthenticationException("Expired exp=" + expiry);
170         }
171 
172         Date notBefore = claims.getNotBeforeTime();
173         if (notBefore != null && notBefore.toInstant().isAfter(Instant.now().plusSeconds(CLOCK_SKEW_SECONDS))) {
174             log.warn("JWT rejected: not yet valid nbf={}", notBefore);
175             throw new ApiKeyAuthenticationException("Not yet valid nbf=" + notBefore);
176         }
177 
178         // Step 7: Credential cache lookup by jti
179         // Note: the token carries no `env` claim — environment authorization is a server-side property
180         // of the credential and is enforced against its `environments` list after the DynamoDB lookup
181         // (see lookupAndCacheCredential).
182         String jti = claims.getJWTID();
183         if (jti == null) {
184             log.warn("JWT rejected: missing jti claim");
185             throw new ApiKeyAuthenticationException("Missing jti claim");
186         }
187 
188         if (revokedCache.getIfPresent(jti) != null) {
189             log.warn("JWT rejected: jti={} is in REVOKED cache", jti);
190             throw new ApiKeyAuthenticationException("jti=" + jti + " is in REVOKED cache");
191         }
192 
193         if (absentCache.getIfPresent(jti) != null) {
194             log.debug("JWT rejected: jti={} is in ABSENT cache", jti);
195             throw new ApiKeyAuthenticationException("jti=" + jti + " is in ABSENT cache");
196         }
197 
198         ApiKeyPrincipal cached = credentialCache.getIfPresent(jti);
199         if (cached != null) {
200             return cached;
201         }
202 
203         // Step 8: DynamoDB lookup on cache miss
204         return lookupAndCacheCredential(claims, jti);
205     }

207     private IzgPrincipal lookupAndCacheCredential(JWTClaimsSet claims, String jti) {
208         var credentialOpt = credentialRepository.findByJti(jti);
209 
210         if (credentialOpt.isPresent() && isUsableStatus(credentialOpt.get().getStatus())) {
211             ApiKeyCredential credential = credentialOpt.get();
212 
213             // Environment authorization: the request's target environment must be in the credential's
214             // server-side `environments` set (a DynamoDB Number Set). A mismatch is cached in the short-lived
215             // absentCache (not the 366-day revoked sentinel) so an `environments` edit takes effect within the
216             // credential TTL.
217             //
218             // DynamoDB cannot store an empty set, so an absent attribute arrives here as null; null and empty
219             // both mean "valid in no environment" and are handled by the same deny below.
220             Integer targetEnv = SystemUtils.getDestType();
221             if (credential.getEnvironments() == null || !credential.getEnvironments().contains(targetEnv)) {
222                 absentCache.put(jti, Boolean.TRUE);
223                 log.warn("JWT rejected: jti={} not valid for target env={} (environments={})",
224                         jti, targetEnv, credential.getEnvironments());
225                 throw new ApiKeyAuthenticationException("jti=" + jti + " not valid for target env=" + targetEnv);
226             }
227 
228             String sub = claims.getSubject();
229             String upn = (String) claims.getClaim("upn");
230             if (upn == null || upn.isBlank()) {
231                 log.warn("JWT rejected: missing or blank upn claim for jti={}", jti);
232                 throw new ApiKeyAuthenticationException("Missing or blank upn claim for jti=" + jti);
233             }
234 
235             // useTypes is carried on the principal so the routing-time intersection check (IGDD-3257) uses the
236             // credential looked up by jti without a second DynamoDB read per message; it refreshes on the
237             // credential cache TTL, the same window that governs an `environments` edit.
238             ApiKeyPrincipal principal = new ApiKeyPrincipal(
239                     sub,
240                     jti,
241                     upn,
242                     config.getIssuer(),
243                     credential.getUseTypes()
244             );
245             credentialCache.put(jti, principal);
246             log.debug("Authenticated ApiKeyPrincipal jti={} org={} upn={}", jti, sub, upn);
247             return principal;
248         }
249 
250         if (credentialOpt.isEmpty()) {
251             absentCache.put(jti, Boolean.TRUE);
252             log.warn("JWT rejected: jti={} not found in DynamoDB", jti);
253             throw new ApiKeyAuthenticationException("jti=" + jti + " not found in DynamoDB");
254         }
255 
256         // Credential found but not in a usable state (e.g. validated, expired, revoked) — cache for full token lifetime
257         revokedCache.put(jti, Boolean.TRUE);
258         log.warn("JWT rejected: jti={} has non-usable status={}, cached in REVOKED cache", jti, credentialOpt.get().getStatus());
259         throw new ApiKeyAuthenticationException("jti=" + jti + " has non-usable status=" + credentialOpt.get().getStatus());
260     }

262     private static boolean isUsableStatus(String status) {
263         return "active".equals(status) || "grace_period".equals(status);
264     }

266     private byte[] resolveSecret(String kid) {
267         if (config.getTestSecret() != null && !config.getTestSecret().isEmpty()) {
268             return config.getTestSecret().getBytes(StandardCharsets.UTF_8);
269         }
270 
271         String cached = secretCache.getIfPresent(kid);
272         if (cached != null) {
273             return cached.getBytes(StandardCharsets.UTF_8);
274         }
275 
276         if (negativeSecretCache.getIfPresent(kid) != null) {
277             log.debug("Secrets Manager: kid={} in negative cache, skipping lookup", kid);
278             return null;
279         }
280 
281         if (secretsManagerClient == null) {
282             log.error("SecretsManagerClient not available and jwt.test-secret not set");
283             return null;
284         }
285 
286         try {
287             String secretValue = secretsManagerClient.getSecretValue(
288                     req -> req.secretId(config.getSecretsManagerSecretName()).versionId(kid)
289             ).secretString();
290             secretCache.put(kid, secretValue);
291             return secretValue.getBytes(StandardCharsets.UTF_8);
292         } catch (ResourceNotFoundException e) {
293             log.warn("Secrets Manager: no version found for kid={}, caching negative result", kid);
294             negativeSecretCache.put(kid, Boolean.TRUE);
295             return null;
296         } catch (Exception e) {
297             log.error("Secrets Manager lookup failed for kid={}: {}", kid, e.getMessage());
298             return null;
299         }
300     }

302     /**
303      * Remove a credential from the caches and mark it as revoked, so that later requests with it fail.
304      * @param jti	The JWT identifier of the credential
305      */
306     public void evictCredential(String jti) {
307         credentialCache.invalidate(jti);
308         absentCache.invalidate(jti);
309         revokedCache.put(jti, Boolean.TRUE);
310         log.info("Evicted credential cache entries for jti={}", jti);
311     }
