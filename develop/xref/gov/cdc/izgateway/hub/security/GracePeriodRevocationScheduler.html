18  /**
19   * Scheduled job that terminates superseded API-key credentials after their grace period expires
20   * (IGDD-2711, User Story 10), recording the correct terminal status (IGDD-3167).
21   *
22   * <p>When Config Console renews an API key (IGDD-2707) it issues a new credential and moves the old
23   * one to status {@code grace_period} with a {@code graceExpiresAt}; it keeps authenticating alongside
24   * the new key during the grace window. Once {@code graceExpiresAt} passes, the old key must be
25   * terminated so renewed keys do not accumulate as indefinitely-valid credentials. Per the credential
26   * state model the effective grace end is {@code min(graceExpiresAt, expiresAt)}: a key whose own
27   * {@code expiresAt} was reached first is recorded {@code expired}; a key cut off before its own expiry
28   * is recorded {@code revoked} ({@link ApiKeyCredentialRepository#resolveTerminalStatus}). This job
29   * performs that sweep inside Hub, reusing the credential repository and audit logger.</p>
30   *
31   * <p>The job is gated on {@code apikey.grace-revocation.enabled} (off by default) so it can be enabled
32   * per environment. Spring's scheduling infrastructure is enabled globally by {@code @EnableScheduling}
33   * on the application class, independent of this bean's conditional.</p>
34   *
35   * <p><b>Multi-instance safety (design D5):</b> every enabled instance runs the cycle; correctness does
36   * not depend on electing a single runner. Each candidate is terminated via a <em>conditional</em> write
37   * ({@link ApiKeyCredentialRepository#terminateIfGracePeriod}) that only succeeds while the key is still
38   * {@code grace_period}, so a given key is terminated — and audited — exactly once even under concurrent
39   * runs. (An earlier host-ordering election was removed: the host registry can retain stale hosts, which
40   * could make a lone live instance defer to a ghost and never run.) Cross-instance cache propagation is
41   * intentionally not broadcast — see {@link #evictLocalCache(String)}.</p>
42   */
43  @Component
44  @Slf4j
45  @ConditionalOnProperty(prefix = "apikey.grace-revocation", name = "enabled", havingValue = "true")
46  public class GracePeriodRevocationScheduler {
47  
48      private final ApiKeyCredentialRepository credentialRepository;
49      private final ApiKeyAuditLogger auditLogger;
50      private final ApiKeyPrincipalProvider apiKeyPrincipalProvider;
51  
52      /**
53       * Create the scheduler. Spring supplies the parameters.
54       * @param credentialRepository	The repository that stores API key credentials
55       * @param auditLogger	The logger for API key audit events
56       * @param apiKeyPrincipalProvider	The provider whose credential cache this job clears
57       */
58      @Autowired
59      public GracePeriodRevocationScheduler(
60              ApiKeyCredentialRepository credentialRepository,
61              ApiKeyAuditLogger auditLogger,
62              ApiKeyPrincipalProvider apiKeyPrincipalProvider
63      ) {
64          this.credentialRepository = credentialRepository;
65          this.auditLogger = auditLogger;
66          this.apiKeyPrincipalProvider = apiKeyPrincipalProvider;
67      }
68  
69      /**
70       * Scheduled entry point. Runs at a fixed delay after the previous completion. Wraps the cycle
71       * so that any failure is logged (for CloudWatch alerting per AC #3) and never escapes to the
72       * scheduler thread.
73       */
74      @Scheduled(
75              fixedDelayString = "${apikey.grace-revocation.interval:PT1H}",
76              initialDelayString = "${apikey.grace-revocation.initial-delay:PT5M}"
77      )
78      public void scheduledRun() {
79          String previousEventId = MDC.get(EventId.EVENTID_KEY);
80          MDC.put(EventId.EVENTID_KEY, EventId.DEFAULT_TX_ID);
81          try {
82              runRevocationCycle();
83          } catch (Exception e) {  // NOSONAR — a scheduled job must never let a failure escape its thread
84              // Structured eventType so a CloudWatch Logs metric filter can match on the JSON field
85              // ({ $.eventType = "GRACE_REVOCATION_FAILED" }) rather than a brittle message substring (AC #3).
86              log.error(Markers2.append("eventType", "GRACE_REVOCATION_FAILED").and(Markers2.append(e)),
87                      "Grace-period revocation cycle failed: {}", e.getMessage());
88          } finally {
89              if (previousEventId != null) {
90                  MDC.put(EventId.EVENTID_KEY, previousEventId);
91              } else {
92                  MDC.remove(EventId.EVENTID_KEY);
93              }
94          }
95      }
96  
97      /**
98       * Execute one revocation cycle: find every superseded credential whose grace period has expired,
99       * terminate each to its correct status ({@code expired} or {@code revoked}, IGDD-3167), emit the
100      * matching audit event, and evict the local cache. The sweep is NOT scoped to this Hub's
101      * environment -- see {@link #runRevocationCycle()} for why -- so the {@code environment} and
102      * {@code serverName} fields on the log and audit events identify the instance that performed a
103      * termination, not the set of credentials considered. Emits a
104      * {@code GRACE_REVOCATION_STARTED} log at the beginning and a {@code GRACE_REVOCATION_RUN} log on
105      * successful completion (with the number of credentials evaluated, expired, and revoked); a
106      * failure is logged as {@code GRACE_REVOCATION_FAILED} by {@link #scheduledRun()}. These three
107      * events let operations see that a run started and whether it succeeded or failed (AC #4; alarms
108      * may be layered on these log events later).
109      */
110     CycleResult runRevocationCycle() {
111         // The credential sort key is {jti} with no environment prefix (IGDD-3140), so there is no prefix
112         // left to scope a query by: the sweep evaluates every ApiKeyCredential record. A key's permitted
113         // environments are a server-side `environments` set, not part of the key.
114         //
115         // The DynamoDB table is shared across environments (dev and test both use izgateway-dev-test),
116         // so any enabled Hub may terminate a credential belonging to another environment. That is
117         // harmless -- grace expiry is environment-independent and the conditional write keeps it
118         // exactly-once -- but it does mean a termination cannot be attributed without recording who did
119         // it, hence `environment`/`serverName` below and on the audit events (IGDD-3257 review).
120         log.info(Markers2.append("eventType", "GRACE_REVOCATION_STARTED",
121                         "environment", SystemUtils.getDestTag(),
122                         "serverName", SystemUtils.getHostname()),
123                 "Grace-period revocation cycle started");
124 
125         List<ApiKeyCredential> candidates = credentialRepository.findGraceRevocationCandidates();
126 
127         int evaluated = candidates.size();
128         int expired = 0;
129         int revoked = 0;
130         for (ApiKeyCredential credential : candidates) {
131             try {
132                 String terminalStatus = terminateCredential(credential);
133                 if (ApiKeyCredentialRepository.STATUS_EXPIRED.equals(terminalStatus)) {
134                     expired++;
135                 } else if (ApiKeyCredentialRepository.STATUS_REVOKED.equals(terminalStatus)) {
136                     revoked++;
137                 }
138             } catch (Exception e) {  // NOSONAR — one bad credential must not abort the rest of the sweep
139                 log.warn(Markers2.append("eventType", "GRACE_REVOCATION_ERROR", "keyId", credential.getJti())
140                                 .and(Markers2.append(e)),
141                         "Failed to terminate credential {}: {}", credential.getJti(), e.getMessage());
142             }
143         }
144 
145         log.info(Markers2.append(
146                 "eventType", "GRACE_REVOCATION_RUN",
147                 "environment", SystemUtils.getDestTag(),
148                 "serverName", SystemUtils.getHostname(),
149                 "evaluated", evaluated,
150                 "expired", expired,
151                 "revoked", revoked
152         ), "Grace-period revocation cycle succeeded: evaluated={}, expired={}, revoked={}",
153                 evaluated, expired, revoked);
154         return new CycleResult(evaluated, expired, revoked);
155     }
156 
157     /** Outcome of one revocation cycle: candidates evaluated, and how many were marked expired vs revoked. */
158     record CycleResult(int evaluated, int expired, int revoked) {
159     }
160 
161     /**
162      * Terminate a single superseded credential via a conditional write (terminate only if still
163      * {@code grace_period}). The terminal status ({@code expired} vs {@code revoked}) is resolved from
164      * the credential's own {@code expiresAt}/{@code graceExpiresAt} up front (IGDD-3167), so the actor
165      * value written and the audit event emitted match the write the repository performs. On the write
166      * that actually performs the termination, emit the matching audit event and evict the credential
167      * from this instance's local cache. If another instance already terminated it (conditional write
168      * fails), do nothing — so the audit event fires exactly once across the fleet.
169      *
170      * @param credential the grace-expired candidate to terminate
171      * @return the terminal status applied ({@link ApiKeyCredentialRepository#STATUS_EXPIRED} or
172      *         {@link ApiKeyCredentialRepository#STATUS_REVOKED}) if this call performed the
173      *         termination; {@code null} if the conditional write lost
174      */
175     private String terminateCredential(ApiKeyCredential credential) {
176         String terminalStatus = ApiKeyCredentialRepository.resolveTerminalStatus(credential);
177         boolean expired = ApiKeyCredentialRepository.STATUS_EXPIRED.equals(terminalStatus);
178         String actor = expired ? ApiKeyAuditLogger.SYSTEM_GRACE_EXPIRATION : ApiKeyAuditLogger.SYSTEM_GRACE_REVOCATION;
179 
180         boolean won = credentialRepository.terminateIfGracePeriod(credential, Instant.now(), actor);
181         if (!won) {
182             return null;
183         }
184 
185         String jti = credential.getJti();
186         if (expired) {
187             auditLogger.apiKeyExpired(jti, credential.getJurisdictionId(), actor, credential.getSupersededBy());
188         } else {
189             auditLogger.apiKeyRevoked(jti, credential.getJurisdictionId(), actor, credential.getSupersededBy());
190         }
191         evictLocalCache(jti);
192         return terminalStatus;
193     }
194 
195     /**
196      * Evict the revoked credential from this instance's local cache so it stops serving the key
197      * immediately.
198      *
199      * <p>Cross-instance broadcast is intentionally <em>not</em> performed here. Grace-period revocation
200      * is non-urgent (the key has been winding down for the whole grace window), and every other instance
201      * re-validates against DynamoDB when its credential-cache entry expires ({@code jwt.credential-cache-ttl},
202      * default 5 minutes), at which point the now-{@code revoked} status takes effect — so the revocation
203      * converges across the fleet within the cache TTL without any broadcast. Immediate all-instance
204      * propagation is the concern of Config Console's manual revoke (IGDD-2707) via {@code /rest/refresh},
205      * not of this scheduled sweep. See design D6.</p>
206      *
207      * @param jti the revoked credential identifier to evict locally
208      */
209     private void evictLocalCache(String jti) {
210         apiKeyPrincipalProvider.evictCredential(jti);
211     }
