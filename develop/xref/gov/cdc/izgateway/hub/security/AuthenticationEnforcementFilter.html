16  import java.io.IOException;
17  import java.util.Set;

19  /**
20   * Returns 401 for unauthenticated callers when server.ssl.client-auth=want.
21   * Required when Hub accepts JWT-only clients (no TLS cert) — without this filter,
22   * anonymous connections would reach business logic as UnauthenticatedPrincipal.
23   */
24  @Component
25  @ConditionalOnProperty(name = "server.ssl.client-auth", havingValue = "want")
26  @Order(Ordered.HIGHEST_PRECEDENCE)
27  public class AuthenticationEnforcementFilter extends OncePerRequestFilter {
28  
29      private final PrincipalService principalService;
30  
31      /**
32       * Create the filter. Spring supplies the parameter.
33       * @param principalService	The service that gets the principal for a request
34       */
35      @Autowired
36      public AuthenticationEnforcementFilter(PrincipalService principalService) {
37          this.principalService = principalService;
38      }
39  
40      /**
41       * Health check endpoints the ALB target group and the container probe reach without a
42       * credential. Matched exactly, for the same reason as
43       * {@code AccessControlValve.HEALTH_CHECK_PATHS}: a prefix match hands this exemption to
44       * anything else that happens to start with {@code /rest/health}, which is more than the
45       * probes need.
46       */
47      private static final Set<String> HEALTH_CHECK_PATHS = Set.of("/rest/health", "/rest/healthy");
48  
49      @Override
50      protected boolean shouldNotFilter(HttpServletRequest request) {
51          return HEALTH_CHECK_PATHS.contains(request.getServletPath());
52      }
53  
54      @Override
55      protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
56              throws ServletException, IOException {
57          if (principalService.getPrincipal(request) instanceof UnauthenticatedPrincipal) {
58              response.sendError(HttpServletResponse.SC_UNAUTHORIZED, "Authentication required");
59              return;
60          }
61          filterChain.doFilter(request, response);
62      }
63  }
