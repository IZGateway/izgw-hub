3   /**
4    * Thrown by {@link ApiKeyPrincipalProvider} when a request presents a Bearer-scheme
5    * Authorization header but the token fails authentication for any reason (malformed,
6    * unverifiable signature, expired, revoked, unknown, or a dependency failure that prevents
7    * validation from completing).
8    *
9    * <p>{@link gov.cdc.izgateway.hub.service.HubPrincipalService} uses this to distinguish
10   * "no API key was presented" (still eligible for certificate fallback) from "an API key was
11   * presented but rejected" (must not fall back to certificate authentication).
12   */
13  public class ApiKeyAuthenticationException extends RuntimeException {
14      /**
15       * Create the exception.
16       * @param message	The reason that the API key failed authentication
17       */
18      public ApiKeyAuthenticationException(String message) {
19          super(message);
20      }
21  }
