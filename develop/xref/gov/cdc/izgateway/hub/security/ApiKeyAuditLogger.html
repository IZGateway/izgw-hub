1   package gov.cdc.izgateway.hub.security;
2   
3   import gov.cdc.izgateway.logging.markers.Markers2;
4   import gov.cdc.izgateway.utils.SystemUtils;
5   import lombok.extern.slf4j.Slf4j;
6   import org.springframework.stereotype.Component;
7   
8   import java.time.Instant;
9   
10  /**
11   * Emits structured audit events for API-key credential lifecycle actions.
12   *
13   * <p>Events are written as structured Logstash fields via {@link Markers2} so they flow through
14   * Hub's existing JSON logging pipeline alongside the rest of the application's security logging.
15   * This is a dedicated, injectable component so the audit behaviour can be unit-tested independently
16   * of the code paths that trigger it.</p>
17   *
18   * <p>Currently this emits the {@code API_KEY_REVOKED} and {@code API_KEY_EXPIRED} events used by the
19   * grace-period revocation job (IGDD-2711, IGDD-3167). Authentication audit events
20   * ({@code API_KEY_USED} / {@code API_KEY_AUTH_FAILED}) are introduced by IGDD-2704, wired into
21   * {@link ApiKeyPrincipalProvider} in that change.</p>
22   */
23  @Component
24  @Slf4j
25  public class ApiKeyAuditLogger {
26  
27      /**
28       * Create the audit logger. Spring creates this component.
29       */
30      public ApiKeyAuditLogger() {
31          // Nothing to initialize.
32      }
33  
34      /** Audit event type emitted when an API key is revoked (cut off before its own expiry). */
35      public static final String API_KEY_REVOKED = "API_KEY_REVOKED";
36  
37      /** Audit event type emitted when an API key is marked expired (reached its own expiry). */
38      public static final String API_KEY_EXPIRED = "API_KEY_EXPIRED";
39  
40      /**
41       * {@code revokedBy} value used when a credential is revoked automatically by the
42       * grace-period revocation scheduled job (IGDD-2711), as opposed to a manual,
43       * operator-driven revocation through Config Console.
44       */
45      public static final String SYSTEM_GRACE_REVOCATION = "system:grace-revocation";
46  
47      /**
48       * {@code expiredBy} value used when a credential is marked expired automatically by the
49       * grace-period revocation scheduled job because it reached its own {@code expiresAt} on or
50       * before its {@code graceExpiresAt} (IGDD-3167).
51       */
52      public static final String SYSTEM_GRACE_EXPIRATION = "system:grace-expiration";
53  
54      /**
55       * Emit an {@code API_KEY_REVOKED} audit event for a credential revocation.
56       *
57       * <p>Used by the grace-period revocation scheduled job (IGDD-2711) when a superseded
58       * credential is revoked after its grace period expires. The event carries no token or
59       * secret material.</p>
60       *
61       * @param keyId          the revoked API key identifier (JWT {@code jti})
62       * @param jurisdictionId the jurisdiction the credential belonged to
63       * @param revokedBy      the identity performing the revocation (e.g. {@link #SYSTEM_GRACE_REVOCATION})
64       * @param supersededBy   the {@code jti} of the renewed credential that superseded this one; may be {@code null}
65       */
66      public void apiKeyRevoked(String keyId, String jurisdictionId, String revokedBy, String supersededBy) {
67          log.info(Markers2.append(
68                  "eventType", API_KEY_REVOKED,
69                  "keyId", keyId,
70                  "jurisdictionId", jurisdictionId,
71                  "revokedBy", revokedBy,
72                  "supersededBy", supersededBy,
73                  // Which Hub performed this. The grace-period sweep is environment-agnostic and the
74                  // DynamoDB table is shared across environments, so without these the record cannot be
75                  // attributed to an instance (IGDD-3257 review).
76                  "environment", SystemUtils.getDestTag(),
77                  "serverName", SystemUtils.getHostname()
78          ).and(Markers2.append("timestamp", Instant.now().toString())),
79                  "API key revoked for jurisdiction {} (keyId={}, revokedBy={})", jurisdictionId, keyId, revokedBy);
80      }
81  
82      /**
83       * Emit an {@code API_KEY_EXPIRED} audit event for a credential that reached its own expiry.
84       *
85       * <p>Used by the grace-period revocation scheduled job (IGDD-3167) when a superseded
86       * credential's own {@code expiresAt} was on or before its {@code graceExpiresAt} — the JWT
87       * {@code exp} capped its validity before the grace window would have. The event carries no
88       * token or secret material.</p>
89       *
90       * @param keyId          the expired API key identifier (JWT {@code jti})
91       * @param jurisdictionId the jurisdiction the credential belonged to
92       * @param expiredBy      the identity recording the expiry (e.g. {@link #SYSTEM_GRACE_EXPIRATION})
93       * @param supersededBy   the {@code jti} of the renewed credential that superseded this one; may be {@code null}
94       */
95      public void apiKeyExpired(String keyId, String jurisdictionId, String expiredBy, String supersededBy) {
96          log.info(Markers2.append(
97                  "eventType", API_KEY_EXPIRED,
98                  "keyId", keyId,
99                  "jurisdictionId", jurisdictionId,
100                 "expiredBy", expiredBy,
101                 "supersededBy", supersededBy,
102                 // See apiKeyRevoked: identifies the Hub instance that performed the termination.
103                 "environment", SystemUtils.getDestTag(),
104                 "serverName", SystemUtils.getHostname()
105         ).and(Markers2.append("timestamp", Instant.now().toString())),
106                 "API key expired for jurisdiction {} (keyId={}, expiredBy={})", jurisdictionId, keyId, expiredBy);
107     }
108 }
