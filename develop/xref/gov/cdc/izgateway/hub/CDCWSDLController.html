25  /**
26   * Handles SOAP requests at {@code /izgw} that use the CDC IIS 2011 WSDL.
27   */
28  @RestController
29  @RolesAllowed({Roles.SOAP, Roles.ADMIN})
30  @RequestMapping("/izgw")
31  @Lazy(false)
32  public class CDCWSDLController extends BaseGatewayController {
33  
34  	/**
35  	 * Create the controller. Spring supplies the parameters.
36  	 * @param mshService	The service that maps message header values to sources
37  	 * @param destinationService	The destination service used to obtain routing information
38  	 * @param endpointStatusService	The service that gets and stores endpoint status
39  	 * @param messageSender	The sender for outbound messages
40  	 * @param adsController	The ADS controller, used for DEX status checks
41  	 * @param registry	The access control registry that this controller registers with
42  	 * @param accessControlService	The service that checks access to destinations
43  	 * @param hubConfig	The sender configuration that supplies the maximum message size
44  	 */
45  	@Autowired
46  	public CDCWSDLController(
47  		IMessageHeaderService mshService,
48  		DestinationService destinationService,
49  		EndpointStatusService endpointStatusService,
50  		MessageSender messageSender,
51  		ADSController adsController,
52  		AccessControlRegistry registry,
53  		AccessControlService accessControlService,
54  		SenderConfig hubConfig
55  	) {
56  		super(mshService, SoapMessage.IIS2011_NS, "cdc-iis-2011.wsdl", Arrays.asList(SoapMessage.IIS2014_NS),
57  			destinationService, endpointStatusService, messageSender, adsController, registry, accessControlService, hubConfig);
58  	}
59  
60      @Override
61      protected boolean isHubWsdl() {
62          // IIS Controller is never to be considered Hub
63          return false;
64      }
65  
66  	@Override
67  	protected void validateHubHeader(SubmitSingleMessageRequest submitSingleMessage) throws Fault {
68  		if (submitSingleMessage.getHubHeader() != null && !submitSingleMessage.getHubHeader().isEmpty()) {
69              throw new UnsupportedOperationFault("IZGW-specific HubHeader is not allowed in CDC WSDL requests.", null);
70  		}
71  	}
72  
73  	@Override
74  	protected void customizeResponse(SubmitSingleMessageResponse response, IDestination dest) {
75          // No customization needed as this is not a Hub response (no hub headers to be set)
76  	}
77  
78  	/**
79  	 * Handle SOAP requests with destinationId as a path parameter
80  	 * This allows requests to be made to /izgw/{destinationId}
81  	 * @param soapMessage The SOAP message
82  	 * @param destinationId The destination ID
83  	 * @return The response entity
84  	 * @throws Fault If an error occurs
85  	 */
86  	@PostMapping(value = "/{destinationId}", produces = {
87  		"application/soap+xml",
88  		"application/soap",
89  		MediaType.APPLICATION_XML_VALUE,
90  		MediaType.TEXT_XML_VALUE,
91  		MediaType.TEXT_PLAIN_VALUE,
92  		MediaType.TEXT_HTML_VALUE
93  	})
94  	public ResponseEntity<?> submitSoapRequestWithDestination(
95  		@RequestBody SoapMessage soapMessage,
96  		@PathVariable String destinationId
97  	) throws Fault {
98  		if (soapMessage.getWsaHeaders() != null) {
99  			soapMessage.getWsaHeaders().setTo(destinationId);
100 		}
101 
102 		return submitSoapRequest(soapMessage, null);
103 	}
104 }
