84      /**
85       * Create the status checker service. Spring supplies the parameters.
86       * @param config	The status checker configuration
87       * @param messageSender	The sender for connectivity tests
88       * @param destinationService	The service that finds destinations
89       * @param endpointStatusService	The service that stores endpoint status
90       * @param clientTlsSupport	The client TLS support that supplies the client certificate
91       * @param adsChecker	The checker for ADS destinations
92       * @param app	The application properties that supply the scheduler
93       */
94      @Autowired
95      public StatusCheckerService(
96      	StatusCheckerConfiguration config, 
97      	MessageSender messageSender, 
98      	IDestinationService destinationService, 
99      	EndpointStatusService endpointStatusService, 
100     	ClientTlsSupport clientTlsSupport,
101     	ADSChecker adsChecker,
102     	AppProperties app
103     ) {
104     	this.scheduler = app.getScheduler();
105         this.config = config;
106         this.messageSender = messageSender;
107         this.destinationService = destinationService;
108         this.clientTlsSupport = clientTlsSupport;
109         this.endpointStatusService =  endpointStatusService;
110         this.adsChecker = adsChecker;
111         setCommonName(X500Utils.getCommonName(clientTlsSupport.getCertificate()));
112         setStatusCheckerEventId(new TransactionData().getEventId());
113         messageSender.setStatusChecker(this);
114     }

120     public List<String> getHosts() {
121     	List<String> l = new ArrayList<>();
122 		l.add(SystemUtils.getHostname());

124     		if (!l.contains(s.getStatusBy())) {

155     /**
156      * Check the status of a destination now and save the result.
157      * @param dest	The destination to check
158      * @return The new status of the destination
159      */
160     public IEndpointStatus updateDestinationStatus(IDestination dest) {
161         EndpointStatus s = new EndpointStatus(dest);
162         if (dest.isDex() || dest.isAzure()) {
163             doAdsStatusCheck(dest, s);
164         } else {
165             doHubStatusCheck(s);
166         }
167         endpointStatusService.save(s);
168         return s;
169     }

171 	private void doAdsStatusCheck(IDestination dest, IEndpointStatus s) {
172 		try {
173 		    adsChecker.check(dest.getDestId());
174 		    s.connected();
175 		} catch (Exception ex) {
176 		    this.reportFault(s, ex);
177 		}
178 	}
179 	private String doHubStatusCheck(EndpointStatus d) {
180         ConnectivityTestRequest ct = new ConnectivityTestRequest();
181         String message = String.format("Wishing %s (%s) an Audacious Hello at %tC", d.getDestUri(), d.getDestId(),
182                 new Date());
183         ct.setEchoBack(message);
184 
185         String status = null;
186         try {
187         	IDestination destination = destinationService.findByDestId(d.getDestId());
188         	if (destination == null) {
189         		return null; // Nobody to send to.
190         	}
191             ConnectivityTestResponse cr = messageSender.sendConnectivityTest(destination, ct);
192             if (cr.getEchoBack() == null) {
193                 status = "No Echo";
194             } else {
195             	// We are not verifying that the echoback response matches b/c some IIS fail at this.
196                 status = d.connected();
197             } 
198         } catch (Exception e) {
199             status = reportFault(d, e);
200         }
201         d.setStatus(status);
202         return status;
203     }

205     private String reportFault(IEndpointStatus s, Exception e) {
206         String status = null;
207         if (e instanceof FaultSupport f) {
208         	status = f.getSummary();
209             s.setStatus(status);
210             s.setDetail(f.getDetail());
211             s.setDiagnostics(f.getDiagnostics());
212             s.setRetryStrategy(f.getRetry().toString());
213         } else {
214         	status = e.getMessage();
215             s.setStatus(status);
216             s.setDetail("Unexpected Exception");
217             s.setDiagnostics("IZ Gateway threw an unexpected exception.");
218             s.setRetryStrategy(RetryStrategy.CONTACT_SUPPORT.toString());
219 		}
220 
221         log.error(Markers2.append(e), "Status check failure on {}", s.getDestId());
222 
223         return status;
224 	}

226     /**
227      * Check the status of a destination as an internal request, with logging. If the destination
228      * is under maintenance, return its current status and do not check it.
229      * @param dest	The destination to check
230      * @return The status of the destination, or null if dest is null
231      */
232     public IEndpointStatus checkDestination(IDestination dest) {
233         if (dest == null) {
234             return null;
235         }
236 
237         if (dest.isUnderMaintenance()) {
238             // Destination is under maintenance, report status, and do nothing further
239             return endpointStatusService.getEndpointStatus(dest);
240         }
241 
242         Map<String, String> map = setupMDC();
243         TransactionData tDataOriginal = RequestContext.getTransactionData();
244         TransactionData tData = new TransactionData(new Date(), statusCheckerEventId);
245         tData.setServiceType("Gateway");
246         RequestContext.setTransactionData(tData);
247         setDestinationInfoFromDestination(RequestContext.getDestinationInfo(), dest);
248         tData.setMessageType(MessageType.CONNECTIVITY_TEST);
249 
250         try {
251 
252             String requestUri = "/rest/status/" + dest.getDestId();
253             MDC.put(LoggingValve.REQUEST_URI, requestUri);
254 
255             // Override source type, this is an internal call.
256             logInternalSource(tData);
257 
258             return updateDestinationStatus(dest);
259         } finally {
260             tData.logIt();
261             RequestContext.setTransactionData(tDataOriginal);
262             restoreMDC(map);
263         }
264     }

281     private void logInternalSource(TransactionData tData) {
282         SourceInfo source = tData.getSource();
283 
284         tData.setCipherSuite("INTERNAL_JAVA_CALL");
285         source.setCipherSuite("INTERNAL_JAVA_CALL");
286         source.setCertificate(clientTlsSupport.getCertificate());
287         source.setFacilityId("IZGW");
288         source.setHost(SystemUtils.getHostname());
289         source.setId("izgw");
290         source.setIpAddress(HostInfo.LOCALHOST_IP4);
291         source.setType(SourceInfo.SOURCE_TYPE_INTERNAL);
292     }

294     private void checkStatus(IDestination dest, int failureCount) {
295         try {
296         	if (checkDestination(dest).isConnected()) {
297         		return;
298         	}
299         } catch (Exception ex) {
300             // Ignore failures.
301         }
302 
303         // We did not succeed, set checks for the future if there are any to be scheduled.
304     	lookForReset(dest, Math.max(failureCount, CHECK_INTERVALS.length - 1));
305     }
306     
307 	private void lookForReset(IDestination dest, int count) {
308         scheduler.schedule(
309             () -> checkStatus(dest, count), StatusCheckerService.CHECK_INTERVALS[count], TimeUnit.SECONDS
310         );
311 	}

313 	public void lookForReset(IDestination dest) {

322 	/**
323 	 * Update status after successful or failed message send. This keeps status fresh and avoids
324 	 * unnecessary status checks.
325 	 * @param status	Current status
326 	 * @param dest		Destination (needed on failure states to look for a reset of the circuit breaker)
327 	 * @param reason	The reason for the failure
328 	 */
329 	@Override
330 	public void updateStatus(IEndpointStatus status, IDestination dest, Fault reason) {
331 		boolean wasCircuitBreakerThrown = status.isCircuitBreakerThrown();
332 		if (reason == null) {
333 			status.connected();
334 			if (wasCircuitBreakerThrown) {
335 				logCircuitBreakerReset(status);
336 			}
337 		} else {
338 			status = new EndpointStatus(dest).fromFault(reason);
339 			if (!wasCircuitBreakerThrown && reason.shouldBreakCircuit()) {
340 				status.setStatus(IEndpointStatus.CIRCUIT_BREAKER_THROWN);
341 				logCircuitBreakerThrown(status, reason);
342 				lookForReset(dest);
343 			}
344 		}
345 		endpointStatusService.save(status);
346 	}

349 	@Override
350     public void logCircuitBreakerReset(IEndpointStatus status) {
351         // Ensure that there is an event id if not set.
352         Map<String, String> mdcValues = setupMDC();
353         // This was previously a broken connection.  We've reset the circuit breaker,
354         // log that event.
355         log.info(Markers2.append("status", status),
356                 "Circuit Breaker Reset for {} ({}) in {}",
357                 status.getDestId(), status.getDestUri(), SystemUtils.getDestTypeAsString());
358         restoreMDC(mdcValues);
359     }
360 
361 	@Override
362     public void logCircuitBreakerThrown(IEndpointStatus status, Throwable why) {
363         Map<String, String> mdcValues = setupMDC();
364         log.info(Markers2.append(why, "status", status),
365                 "Circuit Breaker Thrown for {} ({}) in {}: {}", status.getDestId(),
366                 status.getDestUri(), SystemUtils.getDestTypeAsString(),
367                 status.getStatus());
368         restoreMDC(mdcValues);
369     }

371     /**
372      * Set the MDC logging values for an internal request, if they are not already set.
373      * @return The MDC values before the change, for use with {@link #restoreMDC(Map)}
374      */
375     public static Map<String, String> setupMDC() {
376         // Set various MDC values that are expected to appear in logs to reasonable values that would tell folks about
377         // where this request came from.
378         Map<String, String> m = new HashMap<>();
379         LoggingValve.MDC_EVENTS.forEach(s -> m.put(s,  MDC.get(s)));
380         // Don't mess with MDC if values are already present.
381         if (MDC.get(LoggingValve.EVENT_ID) != null && MDC.get(LoggingValve.REQUEST_URI) != null) {
382         	return m;
383         }
384         MDC.put(LoggingValve.EVENT_ID, statusCheckerEventId);
385         MDC.put(LoggingValve.REQUEST_URI, Thread.currentThread().getName());
386         MDC.put(LoggingValve.METHOD, "INTERNAL");
387         MDC.put(LoggingValve.IP_ADDRESS, HostInfo.LOCALHOST_IP4);
388         MDC.put(LoggingValve.COMMON_NAME, commonName);
389 
390         return m;
391     }
392 
393     /**
394      * Restore the MDC logging values that {@link #setupMDC()} returned.
395      * @param m	The MDC values to restore
396      */
397     public static void restoreMDC(Map<String, String> m) {
398         for (Map.Entry<String, String> e: m.entrySet()) {
399             if (e.getValue() == null) {
400                 MDC.remove(e.getKey());
401             } else {
402                 MDC.put(e.getKey(), e.getValue());
403             }
404         }
405     }
