23  @Slf4j
24  class OldModelHelper implements AccessControlModelHelper {
25  	/** The access control service this helper is associated with. */
26  	private final AccessControlService accessControlService;
27  	/** The server's own access control entry. */
28  	private AccessControl serverAccess; 
29  
30  	/**
31  	 * @param accessControlService
32  	 */
33  	OldModelHelper(AccessControlService accessControlService) {
34  		this.accessControlService = accessControlService;
35  		serverAccess = new AccessControl(
36  			IAccessControlService.GROUP_CATEGORY,
37  			Roles.INTERNAL,
38  			this.accessControlService.getServerName(), 
39  			SystemUtils.getDestType()
40  		);
41  	}

43  	private Map<String, Set<IAccessControl>> allowedUsersByGroup = Collections.emptyMap();
44  	private Map<String, Map<String, Boolean>> allowedRoutesByEvent = Collections.emptyMap();
45  	
46  	@Override
47  	public void refresh() {
48  		Map<String, Set<IAccessControl>> newAllowedUsersByGroup = new TreeMap<>();
49          Map<String, Map<String, Boolean>> newAllowedRoutesByEvent = new TreeMap<>();
50      	getControls(newAllowedUsersByGroup, newAllowedRoutesByEvent);
51          // Add the server itself to the internal group.
52          newAllowedUsersByGroup.computeIfAbsent(Roles.INTERNAL, k -> new LinkedHashSet<>()).add(serverAccess);
53          allowedUsersByGroup = newAllowedUsersByGroup;
54          allowedRoutesByEvent = newAllowedRoutesByEvent;
55          getUserRoles();
56  	}
57  	
58  	@Override
59  	public Map<String, TreeSet<String>> getUserRoles() {
60  		if (allowedUsersByGroup.isEmpty()) {
61  			refresh();
62  		}
63  		Set<String> users = new TreeSet<>();
64  		Set<String> groups = new TreeSet<>();
65  		updateUsersAndGroups(users, groups);
66  		return updateRoles(users, groups);
67  	}
68  	
69  	private void updateUsersAndGroups(Set<String> users, Set<String> groups) { 
70  		for (Entry<String, Set<IAccessControl>> entry : allowedUsersByGroup.entrySet()) {
71  			users.add(entry.getKey());
72  			for (IAccessControl ac: entry.getValue()) {
73  				String member = ac.getMember();
74  				if (!ac.isAllowed() || member == null) {
75  					continue;
76  				}
77  				if (!IAccessControl.isGroup(member)) {
78  					users.add(member);
79  				} else {
80  					groups.add(member);
81  				}
82  			}
83  		}
84  		users.add("*");
85  	}
86  	
87  	private Map<String, TreeSet<String>> updateRoles(Set<String> users, Set<String> groups) {
88  		Map <String, TreeSet<String>> result = new TreeMap<>();
89  
90  		// Do the crosswalk of users and roles, and return membership based on implementation of this class. 
91  		for (String user: users) {
92  			for (String group: groups) {
93  				if (isUserInGroup(user, group)) {
94  					Set<String> roles = result.computeIfAbsent(user, k -> new TreeSet<String>());
95  					roles.add(group);
96  				}
97  			}
98  		}
99  		
100 		return result;
101 	}
102 	
103 	private void getControls(
104 		Map<String, Set<IAccessControl>> newAllowedUsersByGroup,
105 		Map<String, Map<String, Boolean>> newAllowedRoutesByEvent
106 	) {
107 		List<IAccessControl> controls = new ArrayList<>(accessControlService.accessControlRepository.findAllForEnvironment());
108         for (IAccessControl control: controls) {
109         	switch (control.getCategory()) {
110         	case IAccessControlService.GROUP_CATEGORY:
111         		newAllowedUsersByGroup
112 					.computeIfAbsent(
113 						control.getName(),
114 	        			k -> new LinkedHashSet<>()
115 	        		)
116 	        		.add(control);
117         		break;
118         	case IAccessControlService.ROUTE_CATEGORY:
119         		newAllowedRoutesByEvent.computeIfAbsent(
120         			control.getName(),
121 		        	k -> new LinkedHashMap<>()
122 		        ).put(control.getMember(), control.isAllowed());
123         		break;
124         	default:
125         		log.error("Unrecognized category {} in AccessControls", control.getCategory());
126         		continue;
127         	}
128 		}
129 	}

131 	@Override
132 	public Map<String, Object> getGroups() {
133 		if (allowedUsersByGroup.isEmpty()) {
134 			refresh();
135 		}
136 		Map<String, Object> results = new TreeMap<>();
137 		allowedUsersByGroup.values().forEach(s -> s.forEach(ac -> updateAccessGroup(ac, results)));
138     	return results;
139 	}
140 
141 	private Object updateAccessGroup(IAccessControl ac, Map<String, Object> results) {
142 		String groupName = ac.getName();
143 		AccessGroup group = (AccessGroup)results.computeIfAbsent(groupName, cn -> createAccessGroupFromControl(ac));
144 		if (Roles.values().contains(group.getGroupName())) {
145 			group.getRoles().add(group.getGroupName());
146 		}
147 		String member = ac.getMember();
148 		if (IAccessControl.isGroup(member)) {
149 			group.getGroups().add(member);
150 		} else {
151 			group.getUsers().add(member);
152 		}
153 		return group;
154 	}
155 
156 	private AccessGroup createAccessGroupFromControl(IAccessControl acGroup) {
157 		AccessGroup group = new AccessGroup();
158 		group.setGroupName(acGroup.getName());
159 		group.setEnvironment(SystemUtils.getDestType());
160 		return group;
161 	}

163 	@Override
164 	public boolean isUserInRole(String user, String role) {
165 		if (!Roles.values().contains(role)) {
166 			return false;
167 		}
168 		// Roles are groups in the old model
169 		return isUserInGroup(user, role);
170 	}
171 	
172 	@Override
173 	public boolean isUserInGroup(String user, String group) {
174 		
175 		Set<IAccessControl> set = allowedUsersByGroup.get(group);
176 		if (set == null) {
177 			return false;
178 		}
179 		return set.stream().anyMatch(ac -> {
180 			String member = ac.getMember();
181 			return IAccessControl.isGroup(member) ? isUserInGroup(user, member) : AccessControlModelHelper.commonNameMatches(user, member);
182 		});
183 	}
184 
185 	@Override
186 	public boolean isUserDenied(String user) {
187 		return accessControlService.blacklistEnabled && isUserInRole(user, IAccessControlRegistry.BLACKLIST_ROLE);
188 	}
189 
190 	@Override
191 	public Set<String> getEventTypes() {
192 		return allowedRoutesByEvent.keySet();
193 	}
194 
195 	@Override
196 	public IAccessControl unblock(String user) {
197 		return this.accessControlService.accessControlRepository.removeUserFromGroup(user, Roles.BLACKLIST);
198 	}
199 
200 	@Override
201 	public IAccessControl block(String user, String reason) {
202 		return this.accessControlService.accessControlRepository.addUserToGroup(user, Roles.BLACKLIST);
203 	}
204 
205 	@Override
206 	public Set<String> getDenyList() {
207 		Set<IAccessControl> denyList = allowedUsersByGroup.get(IAccessControlRegistry.BLACKLIST_ROLE);
208 		Set<String> result = new TreeSet<>();
209 		denyList.stream().filter(e -> e.isAllowed()).forEach(e -> result.add(e.getMember()));
210 		return result;
211 	}

214 	public boolean canAccessDestination(String user, String destId) {
215 		// The old model does not have destination access controls
216 		return true;
217 	}
