52  	@Override
53  	public void refresh() {
54      	accessGroupCache = refreshCache(accessControlService.accessGroupRepository, ag -> ag.getGroupName());
55      	denyListRecordCache = refreshCache(accessControlService.denyListRecordRepository, dr -> dr.getPrincipal());
56      	// Defensively trim fileTypeName: it's hand-entered in the configuration console/DynamoDB,
57      	// and a stray leading/trailing space (e.g. a copy-paste typo) breaks the three-tier
58      	// reportType match in getFileType() below as well as every downstream use of the
59      	// canonical name (meta_ext_event, data_stream_id, ...). Mutate the entity itself, not
60      	// just the cache key, since getFileType() callers read fileType.getFileTypeName() directly.
61      	fileTypeCache = refreshCache(accessControlService.fileTypeRepository, ft -> {
62      		String trimmed = StringUtils.trim(ft.getFileTypeName());
63      		ft.setFileTypeName(trimmed);
64      		return trimmed;
65      	});
66      	sourceAttackExceptionCache = refreshCache(accessControlService.sourceAttackExceptionRepository, SourceAttackExceptionRecord::getSender);
67      	Map<String, Set<AllowedUser>> newAllowedUserCache = new TreeMap<>();
68      	for (AllowedUser user : refreshCache(accessControlService.allowedUserRepository, au -> au.getDestinationId()).values()) {
69      		newAllowedUserCache.computeIfAbsent(
70      			user.getDestinationId(), 
71      			k -> new TreeSet<>()
72      		).add(user);
73      	}
74      	allowedUserCache = newAllowedUserCache;
75  	}

77  	<T> Map<String, T> refreshCache(IRepository<T> repo, Function<T, String> nameFunction) {
78  		TreeMap<String, T> cache = new TreeMap<>();
79  		repo.findAllForEnvironment().stream().forEach(r -> cache.put(nameFunction.apply(r), r));
80  		return cache;
81  	}

83  	@Override
84  	public Map<String, TreeSet<String>> getUserRoles() {
85  		
86  		Map <String, TreeSet<String>> result = new TreeMap<>();
87  		if (accessGroupCache.isEmpty()) {
88  			refresh();
89  		}
90  		for (IAccessGroup group : accessGroupCache.values()) {
91  			Set<String> groupRoles = new TreeSet<>(group.getRoles());
92  			for (String user : group.getUsers()) {
93  				Set<String> roles = result.computeIfAbsent(user, k -> new TreeSet<String>());
94  				roles.addAll(groupRoles);
95  			}
96  			for (String memberGroup : group.getGroups()) {
97  				IAccessGroup ag = accessGroupCache.get(memberGroup);
98  				if (ag != null) {
99  					for (String user : ag.getUsers()) {
100 						Set<String> roles = result.computeIfAbsent(user, k -> new TreeSet<String>());
101 						roles.addAll(groupRoles);
102 					}
103 				}
104 			}
105 		}
106 		return result;
107 	}

122 	/**
123 	 * Look up a FileType by report type name using a three-tier match:
124 	 * <ol>
125 	 *   <li>Exact match against the cached {@code fileTypeName}.</li>
126 	 *   <li>Case-insensitive match (handles casing variants such as
127 	 *       {@code "ROUTINEIMMUNIZATION"} → {@code "routineImmunization"}).</li>
128 	 *   <li>Noise-word stripped match — removes {@code "vaccination"},
129 	 *       {@code "immunization"}, and {@code "prevention"} from both the
130 	 *       submitted value and each registry key before comparing.  This
131 	 *       allows legacy submission values such as {@code "farmerFlu"} to
132 	 *       match the canonical entry {@code "farmerFluVaccination"} and
133 	 *       preserves backward compatibility.</li>
134 	 * </ol>
135 	 *
136 	 * @param reportType the report type name to look up
137 	 * @return the matching FileType, or {@code null} if not found or input is blank
138 	 */
139 	IFileType getFileType(String reportType) {
140 		if (reportType == null || reportType.isBlank()) {
141 			return null;
142 		}
143 		if (fileTypeCache.isEmpty()) {
144 			refresh();
145 		}
146 		// Tier 1: exact match
147 		FileType exact = fileTypeCache.get(reportType);
148 		if (exact != null) {
149 			return exact;
150 		}
151 		// Tier 2: case-insensitive scan
152 		FileType caseInsensitive = fileTypeCache.entrySet().stream()
153 				.filter(e -> e.getKey().equalsIgnoreCase(reportType))
154 				.map(Map.Entry::getValue)
155 				.findFirst()
156 				.orElse(null);
157 		if (caseInsensitive != null) {
158 			return caseInsensitive;
159 		}
160 		// Tier 3: noise-word stripped match for backward compatibility
161 		// (e.g. "farmerFlu" matches "farmerFluVaccination")
162 		String strippedInput = ADSUtils.stripNoiseWords(reportType);
163 		return fileTypeCache.entrySet().stream()
164 				.filter(e -> ADSUtils.stripNoiseWords(e.getKey()).equals(strippedInput))
165 				.map(Map.Entry::getValue)
166 				.findFirst()
167 				.orElse(null);
168 	}

179 		IAccessGroup accessGroup = accessGroupCache.get(group);
180 		if (accessGroup == null) {
181 			log.warn("Group {} does not exist", group);
182 			return false;
183 		}
184 		return isUserInGroup(user, accessGroup);
185 	}

187 	private boolean isUserInGroup(String user, IAccessGroup accessGroup) {
188 		Set<String> users = accessGroup.getUsers();
189 		if (users.contains(user) || users.contains("*")) {
190 			return true;
191 		}
192 		
193 		return accessGroup.getGroups().stream().anyMatch(g -> isUserInGroup(user, g));
194 	}

201 	@Override
202 	public boolean canAccessDestination(String user, String destId) {
203 		if (allowedUserCache.isEmpty()) {
204 			refresh();
205 		}
206 		Set<AllowedUser> users = allowedUserCache.get(destId); 
207 		if (users == null) {
208 			return true;  // There is no restriction on this destination.
209 		}
210 		// There is an access control list for this destination, so check the sender and verify that
211 		// they have not been disabled.
212 		return users.stream().anyMatch(au -> AccessControlModelHelper.commonNameMatches(user, au.getPrincipal()) && au.isEnabled());
213 	}

216 	public IDenyListRecord unblock(String user) {
217 		if (denyListRecordCache.isEmpty()) {
218 			refresh();
219 		}
220 		DenyListRecord dlr = denyListRecordCache.get(user);
221 		if (dlr != null) {
222 			dlr.setUpdated();
223 			accessControlService.denyListRecordRepository.delete(dlr);
224 		}
225 		denyListRecordCache.remove(user);
226 		return dlr;
227 	}

234 	@Override
235 	public IDenyListRecord block(String user, String reason, String createdBy) {
236 		if (denyListRecordCache.isEmpty()) {
237 			refresh();
238 		}
239 		DenyListRecord dlr = denyListRecordCache.get(user);
240 		if (dlr != null) {
241 			// Already blocked
242 			return dlr;
243 		}
244 		dlr = accessControlService.denyListRecordRepository.createEntity();
245 		dlr.setPrincipal(user);
246 		dlr.setEnvironment(SystemUtils.getDestType());
247 		dlr.setReason(reason);
248 		dlr.setCreatedBy(createdBy);
249 		dlr = accessControlService.denyListRecordRepository.store(dlr);
250 		denyListRecordCache.put(user, dlr);
251 		return dlr;
252 	}

267 	/**
268 	 * Create (or replace) a source-attack exception record for a sender (IGDD-2805).
269 	 * @param sender	The sender's common name
270 	 * @param reason	Operator justification for the exception
271 	 * @return	The stored exception record
272 	 */
273 	SourceAttackExceptionRecord addSourceAttackException(String sender, String reason) {
274 		if (sourceAttackExceptionCache.isEmpty()) {
275 			refresh();
276 		}
277 		SourceAttackExceptionRecord record = accessControlService.sourceAttackExceptionRepository.createEntity();
278 		record.setSender(sender);
279 		record.setReason(reason);
280 		record.setEnvironment(SystemUtils.getDestType());
281 		record.setCreatedBy(RequestContext.getPrincipal().getName());
282 		record = accessControlService.sourceAttackExceptionRepository.store(record);
283 		sourceAttackExceptionCache.put(sender, record);
284 		return record;
285 	}

289 	 * @param sender	The sender's common name

292 		if (sourceAttackExceptionCache.isEmpty()) {
293 			refresh();
294 		}
295 		SourceAttackExceptionRecord record = sourceAttackExceptionCache.get(sender);
296 		if (record != null) {
297 			accessControlService.sourceAttackExceptionRepository.delete(record);
298 		}
299 		sourceAttackExceptionCache.remove(sender);
300 	}
