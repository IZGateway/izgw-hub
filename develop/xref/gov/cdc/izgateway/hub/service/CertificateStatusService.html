12  import java.security.cert.X509Certificate;
13  import java.util.ArrayList;
14  import java.util.List;
15  
16  
17  /**
18   * Service for managing certificate statuses for Revocation checking.
19   * @author Audacious Inquiry
20   *
21   */
22  @Service
23  public class CertificateStatusService implements ICertificateStatusService<ICertificateStatus> {
24      private final ICertificateStatusRepository<CertificateStatus> certificateStatusRepository;
25      
26      /**
27       * Constructor
28       * @param factory	The RepositoryFactory to use to create repositories.
29       */
30      @Autowired
31      public CertificateStatusService(RepositoryFactory factory) {
32          this.certificateStatusRepository = factory.certificateStatusRepository();
33      }
34      
35      @Override
36  	public List<ICertificateStatus> getAllCertificates() {
37          return new ArrayList<>(certificateStatusRepository.findAll());
38      }
39  
40      @Override
41  	public CertificateStatus save(ICertificateStatus certificateStatus){
42      	CertificateStatus cs;
43      	if (certificateStatus instanceof CertificateStatus cs2) {
44  			cs = cs2;
45  		} else {
46  			cs = new CertificateStatus(certificateStatus);
47  		}
48  		return certificateStatusRepository.store(cs);
49      }
50      
51      @Override
52  	public CertificateStatus findByCertificateId(String certificateId) {
53      	return certificateStatusRepository.findByCertificateId(certificateId);
54      }
55  
56  	@Override
57  	public void refresh() {
58  		// There is no refresh needed for this implementation.
59  	}
60  
61  	@Override
62  	public CertificateStatus create(X509Certificate cert) {
63  		return new CertificateStatus(cert);
64  	}
65  }
