109     @Override
110     protected ResponseEntity<?> connectivityTest(ConnectivityTestRequest connectivityTest, String destinationId) throws Fault {
111         if (StringUtils.isEmpty(destinationId)) {
112             return super.connectivityTest(connectivityTest, destinationId);
113         }
114         IDestination dest = getDestination(destinationId);
115 
116         logDestination(dest);
117 
118         IEndpointStatus s = endpointStatusService.getEndpointStatus(dest);
119         ResponseEntity<?> result = null;
120         if (dest.isDex()) {
121             // Do a status check on DEX Endpoint.
122             adsController.getDestinationStatus(connectivityTest.getWsaHeaders().getMessageID(), null, null, destinationId);
123             result = super.connectivityTest(connectivityTest, destinationId);
124         } else {
125             ConnectivityTestResponse response = messageSender.sendConnectivityTest(dest, connectivityTest);
126             if (connectivityTest.is2014Message()) {
127                 response.setSchema(SoapMessage.HUB_NS);  // Shift from client to Hub Schema
128             }
129             result = checkResponseEntitySize(new ResponseEntity<>(response, HttpStatus.OK));
130         }
131 
132         // We got a good result, update status
133         messageSender.getStatusChecker().updateStatus(s, dest, null);
134         return result;
135     }

137     /**
138      * Checks to see if a user can send to the destination.
139      * The method allows access controls to be set on a destination to allow or prohibit users
140      * access to a specific destination.  If no access controls have been set up, then users
141      * are simply expected to follow policy, but there is no other enforcement.
142      *
143      * Checks to see an access control group exists for the specified destination.
144      * 1. If none does, simply returns.
145      *
146      * 2. If the user has the ADMIN role, also returns.
147      *
148      * 3. Otherwise, checks to see if the user is a member of the group given by the destination name, and if
149      * they are, returns.
150      *
151      * 4. If they are not a member, this method throws a security fault notifying the user that they
152      * cannot send messages to the specified group.
153      *
154      * @param destId	The destination identifier
155      * @throws SecurityFault if the user is not permitted to access the destination.
156      */
157     protected void checkAccess(String destId) throws SecurityFault {
158         if (Application.isAdministrator()) {
159             return;
160         }
161         accessControlService.checkAccessToDestination(destId);
162     }

164     @Override
165     protected ResponseEntity<?> submitSingleMessage(SubmitSingleMessageRequest submitSingleMessage, String destinationId) throws Fault {
166         // Validate HubHeader if needed (subclasses can override this)
167         validateHubHeader(submitSingleMessage);
168 
169         IDestination dest = getDestination(destinationId);
170         logDestination(dest);
171 
172         checkAccess(destinationId);
173         IEndpointStatus s = endpointStatusService.getEndpointStatus(dest);
174         checkMessage(submitSingleMessage);
175         SubmitSingleMessageResponse response = messageSender.sendSubmitSingleMessage(dest, submitSingleMessage);
176         response.updateAction(isHubWsdl());
177 
178         // Allow subclasses to customize response processing
179         customizeResponse(response, dest);
180 
181         ResponseEntity<?> result = checkResponseEntitySize(new ResponseEntity<>(response, HttpStatus.OK));
182         // A good result updates the status.
183         messageSender.getStatusChecker().updateStatus(s, dest, null);
184         return result;
185     }

205     @Override
206     protected IDestination getDestination(String destinationId) throws UnknownDestinationFault {
207         if ("".equals(destinationId)) {
208             throw UnknownDestinationFault.invalidDestination(destinationId, "Request has no destination value in the DestinationId element");
209         } else if (null == destinationId) {
210             throw UnknownDestinationFault.missingDestination("Request is missing the DestinationId element");
211         } else if (!destinationId.matches(IDestination.ID_PATTERN)) {
212             RequestContext.getDestinationInfo().setId(destinationId);
213             throw UnknownDestinationFault.invalidDestination(destinationId, "Destination " + destinationId + " is invalid.");
214         }
215         IDestination dest = destinationService.findByDestId(destinationId);
216         if (dest == null) {
217             RequestContext.getDestinationInfo().setId(destinationId);
218             throw UnknownDestinationFault.unknownDestination(destinationId, destinationId, null);
219         }
220         logDestination(dest);
221         return dest;
222     }

224     /**
225      * Record the destination in the destination information for logging.
226      * @param routing	The destination
227      */
228     protected void logDestination(IDestination routing) {
229         DestinationInfo destination = RequestContext.getDestinationInfo();
230         destination.setId(routing.getDestId());
231 
232         String wsdlVersion = routing.getDestVersion();
233         if (!StringUtils.isBlank(wsdlVersion)) {
234             destination.setProtocol(wsdlVersion);
235         }
236 
237         String url = routing.getDestUri();
238         // Destination logging uses the local url for diagnostic use.
239         // This is only visible in logs.
240         destination.setUrl(destinationService.localUrl(url));
241         if (routing.getDestUri().startsWith("/")) {
242             destination.setIpAddress(HostInfo.LOCALHOST_IP4);
243             destination.setHost(HostInfo.LOCALHOST);
244             return;
245         }
246 
247         String host = destinationService.serverOf(url);
248         destination.setHost(host);
249         InetAddress address;
250         try {
251             address = InetAddress.getByName(host);
252             destination.setIpAddress(address.getHostAddress());
253         } catch (UnknownHostException e) {
254             destination.setAddressUnknown();
255         }
256     }

264     /**
265      * The fault code for {@code SecurityFault.sourceAttack(...)} — see izgw-core's
266      * {@code SecurityFault.MESSAGE_TEMPLATES[1]}, code {@code "61"}. Not a public constant in
267      * izgw-core, so it's mirrored here (IGDD-2805); a unit test asserts it against a real
268      * {@code SecurityFault.sourceAttack(...)} instance so a future izgw-core change to that literal
269      * fails loudly instead of silently disabling the lockout below.
270      */
271     private static final String SOURCE_ATTACK_CODE = "61";
272 
273     /**
274      * Handle a detected source attack (IGDD-2805) by adding the sender to the deny list, subject to
275      * the {@code hub.source-attack-lockout.enabled} flag and any configured exception, before
276      * delegating to the normal fault-handling behavior. The {@code sf.getEndpoint() == null} guard
277      * excludes the outbound/destination-response source-attack scan (see {@code MessageSender}),
278      * which reaches this same method with a non-null endpoint — without it, a malicious response
279      * from a destination IIS would wrongfully deny-list the original (innocent) sender.
280      */
281     @Override
282     protected ResponseEntity<FaultMessage> handleFault(Fault fault) {
283         if (fault instanceof SecurityFault sf
284                 && SOURCE_ATTACK_CODE.equals(sf.getCode())
285                 && sf.getEndpoint() == null) {
286             accessControlService.handleSourceAttack(RequestContext.getSourceInfo().getCommonName(), sf.getDiagnostics());
287         }
288         return super.handleFault(fault);
289     }
