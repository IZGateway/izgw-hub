5   import java.util.Arrays;
6   import java.util.Iterator;

9   /**
10   * A {@link Histogram} that keeps its buckets in an array.
11   * Each array row has the low value, the high value and the count of a bucket.
12   */
13  public class ArrayHistogram implements Histogram {
14      private int[][] data;
15      /* Constants for Histogram bucket positions */
16      private static final int BUCKET_LOW = 0;
17      private static final int BUCKET_HIGH = 1;
18      private static final int BUCKET_COUNT = 2;
19  
20      /**
21       * A {@link Histogram.Row} that reads one row of the array.
22       */
23      public class ArrayRow implements Row {
24          private final int index;
25          ArrayRow(int index) {
26              this.index = index;
27          }
28          @Override
29          public int getLow() {
30              return data[index][BUCKET_LOW];
31          }
32  
33          @Override
34          public int getHigh() {
35              return data[index][BUCKET_HIGH];
36          }
37  
38          @Override
39          public int getCount() {
40              return data[index][BUCKET_COUNT];
41          }
42      }
43      private class RowList extends AbstractList<Row> {
44          @Override
45          public Row get(int index) {
46              return new ArrayRow(index);
47          }
48  
49          @Override
50          public int size() {
51              return data.length;
52          }
53      }
54  
55      /**
56       * Make a histogram from a copy of an array of buckets.
57       * @param data The buckets. Each row has the low value, the high value and the count.
58       */
59      public ArrayHistogram(int[][] data) {
60          // Deep copy the array.
61          this.data = Arrays.copyOf(data, data.length);
62          for (int i = 0; i < data.length; i++) {
63              this.data[i] = Arrays.copyOf(this.data[i], this.data[i].length);
64          }
65      }
66  
67      /**
68       * Make a histogram from a list of counts with buckets of the same size.
69       * @param it    The counts, one for each bucket.
70       * @param start The low value of the first bucket.
71       * @param unit  The size of each bucket.
72       * @param isCdf If true, the counts are cumulative and each count includes the counts before it.
73       */
74      public ArrayHistogram(Iterable<Integer> it, int start, int unit, boolean isCdf) {
75          this(it.iterator(), start, unit, isCdf);
76      }
77  
78      /**
79       * Make a histogram from a list of counts with buckets of the same size.
80       * @param it    The counts, one for each bucket.
81       * @param start The low value of the first bucket.
82       * @param unit  The size of each bucket.
83       * @param isCdf If true, the counts are cumulative and each count includes the counts before it.
84       */
85      public ArrayHistogram(Iterator<Integer> it, int start, int unit, boolean isCdf) {
86          List<int[]> l = new ArrayList<>();
87          int lastCount = 0;
88          while (it.hasNext()) {
89              int count = it.next();
90              int[] values = new int[3];
91              values[BUCKET_LOW] = start;
92              start += unit;
93              values[BUCKET_HIGH] = start;
94              values[BUCKET_COUNT] = count - lastCount;
95              l.add(values);
96              if (isCdf) {
97                  lastCount = count;
98              }
99          }
100         data = l.toArray(new int[0][0]);
101     }
102 
103     /**
104      * Generate a histogram using the buckets in buckets, from a set of actual values.
105      * @param values    The values to generate a histogram for.
106      * @param buckets   The buckets to use.
107      */
108     public ArrayHistogram(Iterable<Integer> values, Histogram buckets) {
109         List<Row> bucketRows = buckets.getRows();
110         data = new int[bucketRows.size()][];
111 
112         for (int i = 0; i < data.length; i++) {
113             Row bucket = bucketRows.get(i);
114             data[i] = new int[3];
115             data[i][BUCKET_LOW] = bucket.getLow();
116             data[i][BUCKET_HIGH] = bucket.getHigh();
117         }
118 
119         for (int value: values) {
120             if (value < data[0][BUCKET_LOW]) {
121                 // Out of range low, don't count it.
122                 continue;
123             }
124             // For those who object to linear search, these histograms are highly skewed.
125             // Binary search will likely not perform much better than twice as well as simple
126             // linear search.
127             for (int i = 0; i < data.length; i++) {
128                 if (value < data[i][BUCKET_HIGH]) {
129                     data[i][BUCKET_COUNT]++;
130                     break;
131                 }
132             }
133             // Again, if it didn't fit, we don't count it
134         }
135     }
136 
137     @Override
138     public List<Row> getRows() {
139         return new RowList();
140     }

145         return _toString();
