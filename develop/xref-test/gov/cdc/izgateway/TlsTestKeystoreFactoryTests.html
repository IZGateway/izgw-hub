15  /**
16   * Standalone test of {@link TlsTestKeystoreFactory} -- no Spring context, no AWS/Elastic
17   * dependency. Verifies the ephemeral keystore generation this IGDD-3582 verification relies on
18   * actually works before building the full {@code @SpringBootTest} on top of it.
19   */
20  class TlsTestKeystoreFactoryTests {
21  
22  	@Test
23  	void generatedServerKeyStoreRoundTripsAndCanBeRotated(@TempDir File dir) throws Exception {
24  		char[] password = "password".toCharArray();
25  
26  		KeyPair pair1 = TlsTestKeystoreFactory.generateKeyPair();
27  		X509Certificate cert1 = TlsTestKeystoreFactory.selfSign(pair1, TlsTestKeystoreFactory.serverDn(), 30);
28  		KeyStore ks1 = TlsTestKeystoreFactory.newServerKeyStore("dev.izgateway.org", pair1, cert1, password);
29  
30  		KeyPair pair2 = TlsTestKeystoreFactory.generateKeyPair();
31  		X509Certificate cert2 = TlsTestKeystoreFactory.selfSign(pair2, TlsTestKeystoreFactory.serverDn(), 30);
32  		KeyStore ks2 = TlsTestKeystoreFactory.newServerKeyStore("dev.izgateway.org", pair2, cert2, password);
33  
34  		// Same DN, different serials -- this is the "rotation" the hot-reload test depends on.
35  		assertEquals(cert1.getSubjectX500Principal(), cert2.getSubjectX500Principal());
36  		assertNotEquals(cert1.getSerialNumber(), cert2.getSerialNumber());
37  
38  		KeyPair clientPair = TlsTestKeystoreFactory.generateKeyPair();
39  		X509Certificate clientCert = TlsTestKeystoreFactory.selfSign(clientPair, TlsTestKeystoreFactory.clientDn(), 30);
40  		assertTrue(clientCert.getSubjectX500Principal().getName().contains("tls-hot-reload-test.testing.izgateway.org"));
41  
42  		TlsTestKeystoreFactory.addTrustedCert(ks1, "admin-client", clientCert);
43  		TlsTestKeystoreFactory.addTrustedCert(ks2, "admin-client", clientCert);
44  
45  		File file1 = new File(dir, "awsdev_keystore.bcfks");
46  		File file2 = new File(dir, "awsdev_keystore_rotated.bcfks");
47  		TlsTestKeystoreFactory.writeKeyStore(ks1, file1, password);
48  		TlsTestKeystoreFactory.writeKeyStore(ks2, file2, password);
49  
50  		// Round-trip: read each file back exactly as the application would.
51  		KeyStore reloaded1 = KeyStore.getInstance("BCFKS", TlsTestKeystoreFactory.BCFIPS);
52  		try (var in = new java.io.FileInputStream(file1)) {
53  			reloaded1.load(in, password);
54  		}
55  		assertEquals(2, java.util.Collections.list(reloaded1.aliases()).size());
56  		X509Certificate reloadedServerCert = (X509Certificate) reloaded1.getCertificate("dev.izgateway.org");
57  		assertEquals(cert1.getSerialNumber(), reloadedServerCert.getSerialNumber());
58  		X509Certificate reloadedClientCert = (X509Certificate) reloaded1.getCertificate("admin-client");
59  		assertEquals(clientCert.getSerialNumber(), reloadedClientCert.getSerialNumber());
60  
61  		KeyStore reloaded2 = KeyStore.getInstance("BCFKS", TlsTestKeystoreFactory.BCFIPS);
62  		try (var in = new java.io.FileInputStream(file2)) {
63  			reloaded2.load(in, password);
64  		}
65  		X509Certificate reloadedRotatedCert = (X509Certificate) reloaded2.getCertificate("dev.izgateway.org");
66  		assertEquals(cert2.getSerialNumber(), reloadedRotatedCert.getSerialNumber());
67  		assertNotEquals(reloadedServerCert.getSerialNumber(), reloadedRotatedCert.getSerialNumber());
68  	}
69  }
