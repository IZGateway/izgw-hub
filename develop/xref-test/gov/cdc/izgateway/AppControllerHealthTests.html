81  	private static Set<String> keysOf(IHealth body) {
82  		JsonNode json = MAPPER.valueToTree(body);
83  		Set<String> keys = new TreeSet<>();
84  		json.fieldNames().forEachRemaining(keys::add);
85  		return keys;
86  	}

147 	/**
148 	 * The status code is what load balancer and container probes act on, so it must not depend
149 	 * on who is asking.
150 	 */
151 	@Test
152 	void isHealthyStatusCodeDoesNotDependOnTheCaller() {
153 		HealthService.setHealthy(false, "Database unreachable");
154 		try {
155 			RequestContext.setPrincipal(new UnauthenticatedPrincipal());
156 			HttpServletResponse anonymous = mock(HttpServletResponse.class);
157 			controller.isHealthy(anonymous);
158 			verify(anonymous).setStatus(HttpServletResponse.SC_SERVICE_UNAVAILABLE);
159 
160 			RequestContext.setPrincipal(new AuthenticatedTestPrincipal());
161 			HttpServletResponse authenticated = mock(HttpServletResponse.class);
162 			controller.isHealthy(authenticated);
163 			verify(authenticated).setStatus(HttpServletResponse.SC_SERVICE_UNAVAILABLE);
164 		} finally {
165 			HealthService.setHealthy(true, "Recovered");
166 		}
167 	}

169 	/**
170 	 * The controller tests above assert on the returned object. These two go through Spring's own
171 	 * JSON serialization, because the declared return type is now the {@link IHealth} interface --
172 	 * if Spring ever pinned the writer to the declared type instead of the runtime type, every
173 	 * response would serialize as an empty object and none of the assertions above would notice.
174 	 * <p>
175 	 * {@code standaloneSetup} rather than {@code @SpringBootTest}: a full context here needs a real
176 	 * DynamoDB. See {@code HubWSDLControllerSourceAttackTests} for the same approach.
177 	 */
178 	@Test
179 	void theSerializedResponseWithholdsDetailFromAnUnauthenticatedCaller() throws Exception {
180 		RequestContext.setPrincipal(new UnauthenticatedPrincipal());
181 
182 		String json = MockMvcBuilders.standaloneSetup(controller).build()
183 				.perform(get("/rest/health"))
184 				.andExpect(status().isOk())
185 				.andReturn().getResponse().getContentAsString();
186 
187 		JsonNode body = MAPPER.readTree(json);
188 		Set<String> keys = new TreeSet<>();
189 		body.fieldNames().forEachRemaining(keys::add);
190 
191 		assertEquals(new TreeSet<>(Set.of("healthy", "statusAt")), keys,
192 				"Serialized body was: " + json);
193 	}
194 
195 	@Test
196 	void theSerializedResponseCarriesDetailForAnAuthenticatedCaller() throws Exception {
