26  /**
27   * Generates ephemeral, in-memory-only self-signed certificates and BCFKS keystores for the
28   * TLS certificate hot-reload verification test (IGDD-3582). Nothing produced here is ever
29   * checked into the repository -- every run generates fresh key material so the test fixtures
30   * can never expire, unlike a committed keystore file would.
31   */
32  final class TlsTestKeystoreFactory {
33  	static final String BCFIPS = "BCFIPS";
34  
35  	static {
36  		if (Security.getProvider(BCFIPS) == null) {
37  			Security.addProvider(new BouncyCastleFipsProvider());
38  		}
39  	}
40  
41  	private TlsTestKeystoreFactory() {
42  		// Utility class -- no instances.
43  	}
44  
45  	static KeyPair generateKeyPair() throws GeneralSecurityException {
46  		KeyPairGenerator kpg = KeyPairGenerator.getInstance("RSA");
47  		kpg.initialize(2048);
48  		return kpg.generateKeyPair();
49  	}
50  
51  	/**
52  	 * Build the X500Name used for the server identity certificate, matching the real
53  	 * {@code awsdev_keystore.bcfks} fixture's DN (minus the email attribute, which is not
54  	 * behaviorally relevant to anything under test).
55  	 */
56  	static X500Name serverDn() {
57  		return new X500NameBuilder(BCStyle.INSTANCE)
58  				.addRDN(BCStyle.CN, "dev.izgateway.org")
59  				.addRDN(BCStyle.OU, "Federal and Strategic Advisory")
60  				.addRDN(BCStyle.O, "Audacious Inquiry")
61  				.addRDN(BCStyle.L, "Baltimore")
62  				.addRDN(BCStyle.ST, "Maryland")
63  				.addRDN(BCStyle.C, "US")
64  				.build();
65  	}
66  
67  	/**
68  	 * Build the X500Name for the admin test client certificate. The CN matches the
69  	 * {@code *.testing.izgateway.org} wildcard seeded into the ADMIN group in
70  	 * {@code src/test/resources/data.sql}, so this identity is granted ADMIN access under
71  	 * the JPA-backed test access-control data with no additional role wiring.
72  	 */
73  	static X500Name clientDn() {
74  		return new X500NameBuilder(BCStyle.INSTANCE)
75  				.addRDN(BCStyle.CN, "tls-hot-reload-test.testing.izgateway.org")
76  				.addRDN(BCStyle.OU, "Testing")
77  				.addRDN(BCStyle.O, "IZ Gateway")
78  				.addRDN(BCStyle.L, "Baltimore")
79  				.addRDN(BCStyle.ST, "Maryland")
80  				.addRDN(BCStyle.C, "US")
81  				.build();
82  	}

84  	static X509Certificate selfSign(KeyPair keyPair, X500Name dn, int validityDays) throws Exception {
85  		Date notBefore = new Date();
86  		Date notAfter = new Date(notBefore.getTime() + validityDays * 86_400_000L);
87  		BigInteger serial = BigInteger.valueOf(System.nanoTime());
88  		X509v3CertificateBuilder builder = new JcaX509v3CertificateBuilder(
89  				dn, serial, notBefore, notAfter, dn, keyPair.getPublic());
90  		ContentSigner signer = new JcaContentSignerBuilder("SHA256withRSA").build(keyPair.getPrivate());
91  		return new JcaX509CertificateConverter().setProvider(BCFIPS).getCertificate(builder.build(signer));
92  	}
93  
94  	/**
95  	 * Create a BCFKS keystore containing the given server identity as a single
96  	 * {@code PrivateKeyEntry}.
97  	 */
98  	static KeyStore newServerKeyStore(String alias, KeyPair keyPair, X509Certificate cert, char[] password)
99  			throws GeneralSecurityException {
100 		KeyStore ks = KeyStore.getInstance("BCFKS", BCFIPS);
101 		try {
102 			ks.load(null, null);
103 		} catch (Exception e) {
104 			throw new GeneralSecurityException(e);
105 		}
106 		ks.setKeyEntry(alias, keyPair.getPrivate(), password, new Certificate[] { cert });
107 		return ks;
108 	}
109 
110 	/** Add a trusted certificate entry (e.g. the admin client cert) to an existing keystore. */
111 	static void addTrustedCert(KeyStore ks, String alias, X509Certificate cert) throws GeneralSecurityException {
112 		ks.setCertificateEntry(alias, cert);
113 	}
114 
115 	static void writeKeyStore(KeyStore ks, File file, char[] password) throws Exception {
116 		try (OutputStream out = new FileOutputStream(file)) {
117 			ks.store(out, password);
118 		}
119 	}
