1   package gov.cdc.izgateway.hub.security;
2   
3   import gov.cdc.izgateway.security.IzgPrincipal;
4   import org.junit.jupiter.api.Test;
5   
6   import java.security.Principal;
7   
8   import static org.assertj.core.api.Assertions.assertThat;
9   
10  class ApiKeyPrincipalTests {
11  
12      private static final String SUB = "TEST_ORG";
13      private static final String JTI = "0d0fa1d2-3c5d-4e2a-9a5c-1f7b4d2e9c10";
14      private static final String UPN = "test.example.gov";
15      private static final String ISSUER = "http://localhost:3000";
16  
17      private ApiKeyPrincipal principal() {
18          return new ApiKeyPrincipal(SUB, JTI, UPN, ISSUER);
19      }
20  
21      @Test
22      void rolesAreAlwaysEmpty() {
23          assertThat(principal().getRoles()).isEmpty();
24      }
25  
26      @Test
27      void subjectIsExposedAsOrganization() {
28          assertThat(principal().getOrganization()).isEqualTo(SUB);
29      }
30  
31      @Test
32      void upnIsCarriedAsName() {
33          assertThat(principal().getName()).isEqualTo(UPN);
34      }
35  
36      @Test
37      void jtiFieldIsSet() {
38          assertThat(principal().getJti()).isEqualTo(JTI);
39      }
40  
41      @Test
42      void upnFieldIsSet() {
43          assertThat(principal().getUpn()).isEqualTo(UPN);
44      }
45  
46      @Test
47      void issuerFieldIsSet() {
48          assertThat(principal().getIssuer()).isEqualTo(ISSUER);
49      }
50  
51      @Test
52      void isInstanceOfIzgPrincipalAndJavaPrincipal() {
53          assertThat(principal()).isInstanceOf(IzgPrincipal.class);
54          assertThat(principal()).isInstanceOf(Principal.class);
55      }
56  
57      @Test
58      void serialNumberHexReturnsJti() {
59          // ApiKeyPrincipal uses jti as the serial number equivalent (differs from cert-backed principals)
60          assertThat(principal().getSerialNumberHex()).isEqualTo(JTI);
61      }
62  
63      @Test
64      void nullSubjectIsTolerated() {
65          ApiKeyPrincipal p = new ApiKeyPrincipal(null, JTI, UPN, ISSUER);
66  
67          assertThat(p.getOrganization()).isNull();
68      }
69  }
