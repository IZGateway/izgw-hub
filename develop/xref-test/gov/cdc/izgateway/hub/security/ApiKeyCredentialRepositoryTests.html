10  import java.util.Optional;

12  import static org.assertj.core.api.Assertions.assertThat;
13  import static org.mockito.Mockito.when;
14  
15  @ExtendWith(MockitoExtension.class)
16  class ApiKeyCredentialRepositoryTests {
17  
18      @Mock
19      private ApiKeyCredentialRepository repository;
20  
21      @Test
22      void findByJti_missingRecord_returnsEmpty() {
23          when(repository.findByJti("unknown-jti")).thenReturn(Optional.empty());
24  
25          Optional<ApiKeyCredential> result = repository.findByJti("unknown-jti");
26  
27          assertThat(result).isEmpty();
28      }
29  
30      @Test
31      void findByJti_existingRecord_returnsPresent() {
32          ApiKeyCredential cred = new ApiKeyCredential();
33          cred.setStatus("active");
34          when(repository.findByJti("test-jti")).thenReturn(Optional.of(cred));
35  
36          Optional<ApiKeyCredential> result = repository.findByJti("test-jti");
37  
38          assertThat(result).isPresent();
39          assertThat(result.get().getStatus()).isEqualTo("active");
40      }
41  
42      @Test
43      void sortKeyFormat_isJtiAlone() {
44          // The sort key is the jti alone — no environment prefix (IGDD-3140).
45          String jti = "018f4e2a-5678-7abc-8def-000000000002";
46          ApiKeyCredential cred = new ApiKeyCredential();
47          cred.setJti(jti);
48          assertThat(cred.getPrimaryId()).isEqualTo(jti);
49          assertThat(cred.getSortKey()).isEqualTo(jti);
50      }
51  }
