62      /** An active credential valid for this Hub's target environment. */
63      private static ApiKeyCredential activeCredForThisEnv() {
64          ApiKeyCredential cred = new ApiKeyCredential();
65          cred.setStatus("active");
66          cred.setEnvironments(Set.of(TEST_ENV_ID));

68          return cred;

71      private String buildToken(String alg, String issuer, String jti, Date exp) throws Exception {
72          return buildToken(alg, issuer, jti, exp, TEST_UPN);
73      }

75      private String buildToken(String alg, String issuer, String jti, Date exp, String upn) throws Exception {
76          JWSAlgorithm jwsAlg = "HS256".equals(alg) ? JWSAlgorithm.HS256 : JWSAlgorithm.RS256;
77          JWSHeader header = new JWSHeader.Builder(jwsAlg).keyID(TEST_KID).build();
78          // No `env` claim: environment authorization is a server-side property of the credential (IGDD-3140).
79          JWTClaimsSet.Builder claimsBuilder = new JWTClaimsSet.Builder()
80                  .issuer(issuer)
81                  .subject("TEST_ORG")
82                  .jwtID(jti)
83                  .expirationTime(exp != null ? exp : Date.from(Instant.now().plus(Duration.ofDays(365))));
84          if (upn != null) {
85              claimsBuilder.claim("upn", upn);
86          }
87          JWTClaimsSet claims = claimsBuilder.build();
88          if (!"HS256".equals(alg)) {
89              return "eyJhbGciOiJSUzI1NiJ9.eyJpc3MiOiJvdGhlciJ9.signature";
90          }
91          SignedJWT jwt = new SignedJWT(header, claims);
92          jwt.sign(new MACSigner(TEST_SECRET.getBytes(StandardCharsets.UTF_8)));
93          return jwt.serialize();
94      }

96      @Test
97      void happyPath_validJwtActiveCredential_returnsApiKeyPrincipal() throws Exception {
98          String token = buildToken("HS256", TEST_ISSUER, TEST_JTI, null);
99          when(jwtTokenExtractor.extractToken(request)).thenReturn(token);
100 
101         when(credentialRepository.findByJti(TEST_JTI)).thenReturn(Optional.of(activeCredForThisEnv()));
102 
103         IzgPrincipal principal = provider.getPrincipal(request);
104 
105         assertThat(principal).isInstanceOf(ApiKeyPrincipal.class);
106         ApiKeyPrincipal apiKey = (ApiKeyPrincipal) principal;
107         assertThat(apiKey.getName()).isEqualTo(TEST_UPN);
108         assertThat(apiKey.getUpn()).isEqualTo(TEST_UPN);
109         assertThat(apiKey.getJti()).isEqualTo(TEST_JTI);
110         assertThat(apiKey.getOrganization()).isEqualTo("TEST_ORG");
111         assertThat(apiKey.getRoles()).isEmpty();
112         // useTypes is read from the credential record (never the token) for the routing-time
113         // intersection check against the destination jurisdiction's allowedUseTypes (IGDD-3257).
114         assertThat(apiKey.getUseTypes()).containsExactly("PROVIDER");
115     }
