19  /**
20   * Tests {@link SourceAttackExceptionController} (IGDD-2805): request delegation and role gating.
21   */
22  class SourceAttackExceptionControllerTests {
23  
24      private static final String PATH = "/rest/sourceAttackExceptions";
25  
26      private AccessControlService accessControlService;
27      private SourceAttackExceptionController controller;
28  
29      @BeforeEach
30      void setUp() {
31          accessControlService = mock(AccessControlService.class);
32          controller = new SourceAttackExceptionController(mock(AccessControlRegistry.class), accessControlService);
33      }
34  
35      @Test
36      @DisplayName("list() delegates to AccessControlService")
37      void list_delegatesToService() {
38          List<SourceAttackExceptionRecord> expected = List.of(new SourceAttackExceptionRecord());
39          when(accessControlService.listSourceAttackExceptions()).thenReturn(expected);
40  
41          assertEquals(expected, controller.list());
42      }
43  
44      @Test
45      @DisplayName("create() delegates to AccessControlService with the request's sender and reason")
46      void create_delegatesToService() {
47          SourceAttackExceptionRecord created = new SourceAttackExceptionRecord();
48          when(accessControlService.createSourceAttackException("VHA.example.gov", "known false positive")).thenReturn(created);
49  
50          SourceAttackExceptionController.CreateExceptionRequest request = new SourceAttackExceptionController.CreateExceptionRequest();
51          request.setSender("VHA.example.gov");
52          request.setReason("known false positive");
53  
54          assertEquals(created, controller.create(request));
55      }
56  
57      @Test
58      @DisplayName("delete() delegates to AccessControlService")
59      void delete_delegatesToService() {
60          controller.delete("VHA.example.gov");
61  
62          verify(accessControlService).deleteSourceAttackException("VHA.example.gov");
63      }
64  
65      @Test
66      @DisplayName("Only the ADMIN role is registered for this controller's routes")
67      void onlyAdminRoleIsAllowed() {
68          AccessControlRegistry registry = new AccessControlRegistry();
69          registry.register(SourceAttackExceptionController.class, PATH);
70  
71          assertEquals(List.of(Roles.ADMIN), registry.getAllowedRoles(RequestMethod.POST, PATH));
72          assertEquals(List.of(Roles.ADMIN), registry.getAllowedRoles(RequestMethod.GET, PATH));
73          assertEquals(List.of(Roles.ADMIN), registry.getAllowedRoles(RequestMethod.DELETE, PATH + "/VHA.example.gov"));
74      }
75  }
