16  /**
17   * Tests {@code AccessControlService.useTypeViolation} — the routing-time use-type decision for API-key
18   * callers (IGDD-3257). A non-null return always rejects: the calling method has no warn/permissive mode,
19   * so a returned fault is always thrown.
20   */
21  class UseTypeAccessControlTests {
22  
23      private static final String DEST_ID = "az";
24  
25      private static ApiKeyPrincipal principal(Set<String> useTypes) {
26          return new ApiKeyPrincipal("42", "018f4e2a-5678-7abc-8def-000000000002",
27                  "test.example.gov", "http://localhost:3000", useTypes);
28      }
29  
30      @Test
31      @DisplayName("Intersecting use types produce no violation")
32      void intersectingUseTypes_noViolation() {
33          assertNull(AccessControlService.useTypeViolation(
34                  principal(Set.of("PROVIDER")), DEST_ID, Set.of("PROVIDER", "PATIENT")));
35      }
36  
37      @Test
38      @DisplayName("Disjoint use types produce a violation")
39      void disjointUseTypes_produceViolation() {
40          SecurityFault fault = AccessControlService.useTypeViolation(
41                  principal(Set.of("PATIENT")), DEST_ID, Set.of("PROVIDER"));
42          assertNotNull(fault);
43      }
44  
45      @Test
46      @DisplayName("A jurisdiction with no allowedUseTypes denies every API-key sender")
47      void emptyAllowedUseTypes_deniesAll() {
48          assertNotNull(AccessControlService.useTypeViolation(
49                  principal(Set.of("PROVIDER")), DEST_ID, Set.of()));
50          assertNotNull(AccessControlService.useTypeViolation(
51                  principal(Set.of("PROVIDER")), DEST_ID, null));
52      }
53  
54      @Test
55      @DisplayName("A credential with no useTypes is denied")
56      void credentialWithoutUseTypes_isDenied() {
57          assertNotNull(AccessControlService.useTypeViolation(principal(null), DEST_ID, Set.of("PROVIDER")));
58          assertNotNull(AccessControlService.useTypeViolation(principal(Set.of()), DEST_ID, Set.of("PROVIDER")));
59      }
60  
61      @Test
62      @DisplayName("The fault identifies the credential, destination, and both use-type sets")
63      void faultMessageIsDiagnostic() {
64          SecurityFault fault = AccessControlService.useTypeViolation(
65                  principal(Set.of("PATIENT")), DEST_ID, Set.of("PROVIDER"));
66          assertNotNull(fault);
67          String detail = fault.getMessage() + " " + fault.getDetail();
68          assertTrue(detail.contains("018f4e2a-5678-7abc-8def-000000000002"), "should name the jti: " + detail);
69          assertTrue(detail.contains(DEST_ID), "should name the destination: " + detail);
70          assertTrue(detail.contains("PATIENT"), "should report the credential useTypes: " + detail);
71          assertTrue(detail.contains("PROVIDER"), "should report the allowedUseTypes: " + detail);
72      }
73  
74      @Test
75      @DisplayName("The fault reports a 4xx client error, not a server error")
76      void faultIsClientError() {
77          SecurityFault fault = AccessControlService.useTypeViolation(
78                  principal(Set.of("PATIENT")), DEST_ID, Set.of("PROVIDER"));
79          assertNotNull(fault);
80          // A use-type denial is the caller's problem, so it must not surface as a 5xx. The REST/ADS path
81          // derives its status from the fault's RetryStrategy (see ExceptionHandling#handleFault), and
82          // CORRECT_MESSAGE maps to 400. Note the SOAP path hardcodes 500 for every fault in izgw-core's
83          // SoapControllerBase#handleFault — that is a separate, cross-repo concern.
84          assertEquals(RetryStrategy.CORRECT_MESSAGE, fault.getRetry());
85          assertTrue(fault.getRetry().getStatus().is4xxClientError(),
86                  "expected a 4xx status, got " + fault.getRetry().getStatus());
87      }
88  
89      @Test
90      @DisplayName("The fault carries no token or secret material")
91      void faultCarriesNoSecrets() {
92          SecurityFault fault = AccessControlService.useTypeViolation(
93                  principal(Set.of("PATIENT")), DEST_ID, Set.of("PROVIDER"));
94          String detail = fault.getMessage() + " " + fault.getDetail();
95          assertTrue(!detail.contains("Bearer") && !detail.contains("eyJ"), "must not contain token material");
96      }
97  }
