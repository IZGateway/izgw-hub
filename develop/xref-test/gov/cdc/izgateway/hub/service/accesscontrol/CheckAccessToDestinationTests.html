34  /**
35   * Tests how {@code AccessControlService.checkAccessToDestination} composes its two independent checks:
36   * the source/destination rule (configurable warn-or-deny via {@code hub.access-control.action}) and the
37   * API-key use-type rule (always deny, IGDD-3257).
38   *
39   * <p>The case that motivated these tests: with the default {@code action=warn}, a source violation used
40   * to {@code return} early and skip the use-type check entirely, so the two policies were coupled by
41   * control flow. Warn mode must suppress its own rule only, not the ones after it.</p>
42   *
43   * @author Audacious Inquiry
44   */
45  class CheckAccessToDestinationTests {
46  
47      private static final String DEST_ID = "az";
48      private static final int JURISDICTION_ID = 42;
49      private static final String JTI = "018f4e2a-5678-7abc-8def-000000000002";
50      private static final String SENDER_CN = "sender.example.gov";
51  
52      private IDestinationService destinationService;
53      private IJurisdictionService jurisdictionService;
54      private AccessControlService service;
55  
56      @BeforeAll
57      static void bootstrapAppProperties() {
58          // Constructing a TransactionData (via RequestContext.init) reads the AppProperties singleton to
59          // decide PHI masking. Outside a Spring context nothing has published it, so publish a default —
60          // the constructor registers itself as the instance.
61          if (AppProperties.getInstance() == null) {
62              new AppProperties();
63          }
64      }
65  
66      @BeforeEach
67      void setUp() {
68          destinationService = mock(IDestinationService.class);
69          jurisdictionService = mock(IJurisdictionService.class);
70          // A spy so canAccessDestination can be stubbed: the real implementation needs the model helpers
71          // built by afterPropertiesSet(), which is out of scope here — the composition is what matters.
72          service = spy(new AccessControlService(
73                  mock(RepositoryFactory.class),
74                  mock(IAccessControlRegistry.class),
75                  mock(AccessControlMigrator.class),
76                  destinationService,
77                  jurisdictionService));
78          // RequestContext supplies the sender common name, the principal, and the TransactionData that
79          // records the process error.
80          RequestContext.init();
81          RequestContext.getSourceInfo().setCommonName(SENDER_CN);
82      }
83  
84      @AfterEach
85      void tearDown() {
86          RequestContext.clear();
87      }

89      /** Stub the source/destination rule's outcome. */
90      private void sourceAccess(boolean permitted) {
91          // any() rather than anyString(): a caller with no resolved common name passes null here, which
92          // anyString() would not match — the stub would silently fall through to the real implementation.
93          doReturn(permitted).when(service).canAccessDestination(any(), any());
94      }

96      /** Put an API-key caller with the given useTypes on the request context. */
97      private void apiKeyCaller(Set<String> useTypes) {
98          RequestContext.setPrincipal(
99                  new ApiKeyPrincipal("42", JTI, "test.example.gov", "http://localhost:3000", useTypes));
100     }
101 
102     /** Point DEST_ID at a jurisdiction with the given policy. */
103     private void destinationPolicy(Set<String> allowedUseTypes) {
104         IDestination dest = mock(IDestination.class);
105         when(dest.getJurisdictionId()).thenReturn(JURISDICTION_ID);
106         when(destinationService.findByDestId(DEST_ID)).thenReturn(dest);
107         Jurisdiction jurisdiction = new Jurisdiction();
108         jurisdiction.setJurisdictionId(JURISDICTION_ID);
109         jurisdiction.setAllowedUseTypes(allowedUseTypes);
110         when(jurisdictionService.getJurisdiction(anyInt())).thenReturn(jurisdiction);
111     }

113     @Test
114     @DisplayName("A warn-only source violation does not suppress the use-type check")
115     void sourceViolationInWarnMode_stillEnforcesUseType() {
116         service.accessControlAction = "warn";
117         sourceAccess(false);
118         apiKeyCaller(Set.of("PATIENT"));
119         destinationPolicy(Set.of("PROVIDER"));
120 
121         SecurityFault fault = assertThrows(SecurityFault.class,
122                 () -> service.checkAccessToDestination(DEST_ID));
123 
124         assertEquals("Use Type Not Allowed", fault.getSummary());
125         // The rejection is recorded on the transaction so it shows up in the request log, not just the throw.
126         assertTrue(RequestContext.getTransactionData().getHasProcessError());
127         assertTrue(RequestContext.getTransactionData().getProcessError().startsWith("Use Type Not Allowed"),
128                 "process error should record the use-type denial: "
129                         + RequestContext.getTransactionData().getProcessError());
130     }
131 
132     @Test
133     @DisplayName("A warn-only source violation still allows a message whose use types are permitted")
134     void sourceViolationInWarnMode_useTypeSatisfied_allows() {
135         service.accessControlAction = "warn";
136         sourceAccess(false);
137         apiKeyCaller(Set.of("PROVIDER"));
138         destinationPolicy(Set.of("PROVIDER", "PATIENT"));
139 
140         assertDoesNotThrow(() -> service.checkAccessToDestination(DEST_ID));
141     }
142 
143     @Test
144     @DisplayName("A denied source check throws its own fault and short-circuits the use-type check")
145     void sourceDenied_shortCircuits() {
146         service.accessControlAction = "deny";
147         sourceAccess(false);
148         apiKeyCaller(Set.of("PATIENT"));
149 
150         SecurityFault fault = assertThrows(SecurityFault.class,
151                 () -> service.checkAccessToDestination(DEST_ID));
152 
153         assertEquals("Source Not Allowed", fault.getSummary());
154         assertTrue(fault.getDetail().contains(SENDER_CN), "should name the sender: " + fault.getDetail());
155         // The message is already rejected, so the destination is never resolved for a use-type decision.
156         verifyNoInteractions(destinationService);
157     }
158 
159     @Test
160     @DisplayName("A permitted source with disjoint use types is still rejected")
161     void sourcePermitted_useTypeViolation_throws() {
162         service.accessControlAction = "deny";
163         sourceAccess(true);
164         apiKeyCaller(Set.of("PATIENT"));
165         destinationPolicy(Set.of("PROVIDER"));
166 
167         SecurityFault fault = assertThrows(SecurityFault.class,
168                 () -> service.checkAccessToDestination(DEST_ID));
169 
170         assertEquals("Use Type Not Allowed", fault.getSummary());
171     }

173     @Test
174     @DisplayName("A certificate caller is unaffected by the use-type rule, even in warn mode")
175     void certificateCaller_skipsUseTypeCheck() {
176         service.accessControlAction = "warn";
177         sourceAccess(false);
178         RequestContext.setPrincipal(new CertificatePrincipal());
179 
180         assertDoesNotThrow(() -> service.checkAccessToDestination(DEST_ID));
181         // useTypes is a property of an ApiKeyCredential; a cert caller has none, so the rule cannot apply
182         // and the destination is never resolved for it.
183         verifyNoInteractions(destinationService);
184     }

186     @Test
187     @DisplayName("An API-key caller to an unknown destination is left to the caller's UnknownDestinationFault")
188     void unknownDestination_doesNotThrowUseTypeFault() {
189         service.accessControlAction = "warn";
190         sourceAccess(true);
191         apiKeyCaller(Set.of("PATIENT"));
192         when(destinationService.findByDestId(DEST_ID)).thenReturn(null);
193
