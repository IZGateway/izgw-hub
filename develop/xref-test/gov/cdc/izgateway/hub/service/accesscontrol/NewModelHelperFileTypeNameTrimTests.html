25  /**
26   * Regression test for a real incident on IGDD-3743: a {@code FileType.fileTypeName}
27   * value hand-entered in the configuration console/DynamoDB with a trailing space
28   * (e.g. {@code "riAnnualAggregate "}) broke the three-tier {@code reportType} match
29   * in {@link NewModelHelper#getFileType(String)}, since neither exact nor
30   * case-insensitive comparison tolerates whitespace. Confirmed live against
31   * {@code dev.izgateway.org} via the CI {@code verify} job on PR #209.
32   * <p>
33   * {@link NewModelHelper#refresh()} now trims {@code fileTypeName} for both the cache
34   * key and the entity's own field, since {@link NewModelHelper#getFileType(String)}
35   * callers read {@code fileType.getFileTypeName()} directly off the returned entity,
36   * not just the cache key.
37   * </p>
38   */
39  @SuppressWarnings("unchecked")
40  class NewModelHelperFileTypeNameTrimTests {
41  
42      private AccessControlService service;
43  
44      @BeforeEach
45      void setUp() {
46          RepositoryFactory factory = mock(RepositoryFactory.class);
47          // NewModelHelper.refresh() touches all five caches unconditionally, so every
48          // repository the factory can return must be a real mock (not null) or refresh() NPEs.
49          when(factory.accessGroupRepository()).thenReturn(mock(IAccessGroupRepository.class));
50          when(factory.denyListRecordRepository()).thenReturn(mock(IDenyListRecordRepository.class));
51          IFileTypeRepository<FileType> fileTypeRepository = mock(IFileTypeRepository.class);
52          when(factory.fileTypeRepository()).thenReturn(fileTypeRepository);
53          when(factory.allowedUserRepository()).thenReturn(mock(IAllowedUserRepository.class));
54          when(factory.sourceAttackExceptionRepository()).thenReturn(mock(ISourceAttackExceptionRepository.class));
55  
56          FileType untrimmed = new FileType();
57          untrimmed.setFileTypeName("riAnnualAggregate "); // trailing space, as actually seen in dev
58          when(fileTypeRepository.findAllForEnvironment()).thenReturn(List.of(untrimmed));
59  
60          service = new AccessControlService(
61                  factory,
62                  mock(IAccessControlRegistry.class),
63                  mock(AccessControlMigrator.class),
64                  mock(IDestinationService.class),
65                  mock(IJurisdictionService.class));
66          service.newModelHelper = new NewModelHelper(service);
67      }
68  
69      @Test
70      void getFileType_trimsTrailingSpaceFromDynamoDbName_exactMatchSucceeds() {
71          IFileType ft = service.newModelHelper.getFileType("riAnnualAggregate");
72  
73          assertNotNull(ft, "clean 'riAnnualAggregate' should exact-match despite the trailing space in DynamoDB");
74          assertEquals("riAnnualAggregate", ft.getFileTypeName(),
75                  "the returned entity's own fileTypeName must also be trimmed, "
76                  + "since callers (e.g. MetadataBuilder.setReportType) read it directly");
77      }
78  
79      @Test
80      void getEventTypes_doesNotLeakUntrimmedName() {
81          // getFileType() triggers refresh() as a side effect when the cache is empty.
82          service.newModelHelper.getFileType("riAnnualAggregate");
83  
84          assertEquals(List.of("riAnnualAggregate"), List.copyOf(service.newModelHelper.getEventTypes()));
85      }
86  }
