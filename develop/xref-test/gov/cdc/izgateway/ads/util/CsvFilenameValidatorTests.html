21      // 1. testMonthlyRSV_XXA_2022SEP.csv  – contains "test", otherwise valid
22      // -------------------------------------------------------------------------
23  
24      @Test
25      void parseFilename_testPrefix_strippedAndFlagSet() {
26          CsvFilenameComponents c = CsvFilenameValidator.parseFilename("testMonthlyRSV_XXA_2022SEP.csv");
27  
28          assertNotNull(c, "Should parse successfully after stripping 'test'");
29          assertTrue(c.isTestFile(),           "isTestFile should be true");
30          assertEquals("Monthly",   c.getFrequency());
31          assertEquals("RSV",       c.getReportTypeAbbrev());
32          assertEquals("XXA",       c.getEntityId());
33          assertEquals("2022SEP",   c.getDateString());
34          assertEquals("2022",      c.getYear());
35          assertEquals("SEP",       c.getMonth());
36          assertNull(c.getQuarter(),           "Quarter should be null for a monthly filename");
37          assertEquals("csv",       c.getExtension());
38          assertTrue(c.isMonthly());
39          assertFalse(c.isQuarterly());
40      }
41  
42      @Test
43      void validate_testPrefix_passesWithPeriodAndEntity() {
44          CsvFilenameComponents result =
45              CsvFilenameValidator.validate("testMonthlyRSV_XXA_2022SEP.csv", "XXA", "2022-SEP");
46  
47          assertTrue(result.isValid(), () -> "Expected valid but got errors: " + result.getErrors());
48          assertTrue(result.isTestFile());
49      }
