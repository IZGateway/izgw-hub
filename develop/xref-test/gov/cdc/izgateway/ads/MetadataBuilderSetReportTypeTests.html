20  /**
21   * Integration tests for {@link MetadataBuilder#setReportType(String)} focusing on
22   * the interaction with an injected {@link IAccessControlService}.
23   *
24   * <ul>
25   *   <li>Verifies that metadata fields ({@code extEvent}, {@code extEventType},
26   *       {@code dataStreamId}) are set correctly for every known report type.</li>
27   *   <li>Verifies warning-only behavior when a report type is not in the registry
28   *       (processing must continue, no error added).</li>
29   *   <li>Verifies that no registry lookup occurs when no service is injected.</li>
30   * </ul>
31   *
32   * <p>Covers Task 10 of the ads-metadata-management change request.</p>
33   */
34  class MetadataBuilderSetReportTypeTests {
35  
36      private IAccessControlService mockService;
37      private IFileType mockFileType;
38  
39      @BeforeEach
40      void setUp() {
41          mockService  = mock(IAccessControlService.class);
42          mockFileType = mock(IFileType.class);
43          // Default: getFileType(x) returns a mock whose getFileTypeName() returns x (identity —
44          // canonical inputs already have the right casing, so normalization is a no-op for them).
45          when(mockService.getFileType(anyString())).thenAnswer(inv -> {
46              String name = inv.getArgument(0);
47              IFileType ft = mock(IFileType.class);
48              when(ft.getFileTypeName()).thenReturn(name);
49              return ft;
50          });
51          // farmerFlu is a legacy alias; the registry maps it to the canonical farmerFluVaccination.
52          IFileType farmerFluType = mock(IFileType.class);
53          when(farmerFluType.getFileTypeName()).thenReturn("farmerFluVaccination");
54          when(mockService.getFileType("farmerFlu")).thenReturn(farmerFluType);
55      }

61      @ParameterizedTest(name = "{0}: extEvent={1}, dataStreamId={2}")
62      @CsvSource({
63          "routineImmunization,        routineImmunization,        routine-immunization",
64          "influenzaVaccination,        influenzaVaccination,        influenza-vaccination",
65          "rsvPrevention,               rsvPrevention,               rsv-prevention",
66          "measlesVaccination,          measlesVaccination,          measles-vaccination",
67          "covidAllMonthlyVaccination,  covidAllMonthlyVaccination,  covid-all-monthly-vaccination",
68          "covidBridgeVaccination,      covidBridgeVaccination,      covid-bridge-vaccination",
69          "genericImmunization,         genericImmunization,         generic-immunization",
70          // farmerFlu: special extEvent mapping drives the correct dataStreamId
71          "farmerFlu,                   farmerFluVaccination,        farmer-flu-vaccination",
72          "farmerFluVaccination,        farmerFluVaccination,        farmer-flu-vaccination",
73      })
74      void setReportType_setsAllDerivedFields(
75              String reportType, String expectedExtEvent, String expectedDataStreamId) throws MetadataFault {
76  
77          MetadataBuilder builder = new MetadataBuilder(mockService);
78          builder.setMetadataValidationEnabled(false);
79          builder.setReportType(reportType.trim());
80  
81          MetadataImpl meta = builder.build();
82          assertEquals(expectedExtEvent.trim(),       meta.getExtEvent(),      "extEvent");
83          assertEquals(expectedExtEvent.trim(),       meta.getExtEventType(),  "extEventType (canonical when registry match found)");
84          assertEquals(expectedDataStreamId.trim(),   meta.getDataStreamId(),  "dataStreamId");
85          assertTrue(builder.getErrors().isEmpty(),   "no errors expected");
86      }

100     @Test
101     void setReportType_unknownType_logsWarnButNoError() throws MetadataFault {
102         // Registry returns null → unknown type
103         when(mockService.getFileType("unknownType")).thenReturn(null);
104 
105         MetadataBuilder builder = new MetadataBuilder(mockService);
106         builder.setMetadataValidationEnabled(false);
107         builder.setReportType("unknownType");
108 
109         // Processing must continue — no validation error should be added
110         assertTrue(builder.getErrors().isEmpty(),
111                 "unknown report type should log a warning but not add an error");
112         // Fields are still computed from the name
113         MetadataImpl meta = builder.build();
114         assertEquals("unknown-type", meta.getDataStreamId());
115         assertEquals("unknownType",  meta.getExtEventType());
116     }

178     @ParameterizedTest(name = "case-variant: setReportType({0}) → dataStreamId={2}, extEventType={1}")
179     @CsvSource({
180         // input,                   canonical name,               expected dataStreamId
181         "covidall,                  covidAllMonthlyVaccination,   covid-all-monthly-vaccination",
182         "COVIDALLMONTHLYVACCINATION,covidAllMonthlyVaccination,   covid-all-monthly-vaccination",
183         "farmerflu,                 farmerFluVaccination,         farmer-flu-vaccination",
184         "FARMERFLU,                 farmerFluVaccination,         farmer-flu-vaccination",
185         "ROUTINEIMMUNIZATION,       routineImmunization,          routine-immunization",
186     })
187     void setReportType_caseVariant_normalizesAndComputesCorrectDataStreamId(
188             String input, String canonicalName, String expectedDataStreamId) throws MetadataFault {
189 
190         // Override the default stub for this specific case-variant input so that
191         // getFileTypeName() returns the canonical registry name, not the raw input.
192         IFileType canonicalFileType = mock(IFileType.class);
193         when(canonicalFileType.getFileTypeName()).thenReturn(canonicalName.trim());
194         when(mockService.getFileType(input.trim())).thenReturn(canonicalFileType);
195 
196         MetadataBuilder builder = new MetadataBuilder(mockService);
197         builder.setMetadataValidationEnabled(false);
198         builder.setReportType(input.trim());
199 
200         MetadataImpl meta = builder.build();
201         assertEquals(expectedDataStreamId.trim(), meta.getDataStreamId(), "dataStreamId");
202         assertEquals(canonicalName.trim(),        meta.getExtEventType(), "extEventType normalized to canonical");
203         assertTrue(builder.getErrors().isEmpty(), "no errors expected");
204     }
