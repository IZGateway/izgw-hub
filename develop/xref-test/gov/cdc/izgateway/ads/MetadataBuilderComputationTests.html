3   import static org.junit.jupiter.api.Assertions.assertEquals;

5   import org.junit.jupiter.api.Test;

9   /**
10   * Unit tests for the static computation methods on {@link MetadataBuilder}.
11   * <p>
12   * These methods are package-private ({@code static}) so they can be tested directly
13   * without constructing a full Spring context. Each method is tested against all known
14   * production file type names plus edge cases.
15   * </p>
16   *
17   * <p>Covers Task 7 of the ads-metadata-management change request.</p>
18   */
19  class MetadataBuilderComputationTests {
20  
21      // -------------------------------------------------------------------------
22      // computeMetaExtEvent  (package-private — accessed via setReportType side-effects
23      //  or tested indirectly via MetadataBuilder; exposed here for completeness
24      //  by calling setReportType on a no-arg builder and inspecting getErrors/build)
25      // -------------------------------------------------------------------------
26  
27      /**
28       * Verify that setReportType stores the expected extEvent on the built metadata.
29       * farmerFlu is the only name that maps to a different extEvent value.
30       */
31      @ParameterizedTest(name = "setReportType({0}).extEvent = {1}")
32      @CsvSource({
33          "routineImmunization,        routineImmunization",
34          "influenzaVaccination,        influenzaVaccination",
35          "farmerFlu,                   farmerFluVaccination",
36          "farmerFluVaccination,        farmerFluVaccination",
37          "rsvPrevention,               rsvPrevention",
38          "measlesVaccination,          measlesVaccination",
39          "covidAllMonthlyVaccination,  covidAllMonthlyVaccination",
40          "genericImmunization,         genericImmunization",
41      })
42      void setReportType_setsCorrectExtEvent(String reportType, String expectedExtEvent) throws MetadataFault {
43          MetadataBuilder builder = new MetadataBuilder();
44          builder.setMetadataValidationEnabled(false);
45          builder.setReportType(reportType.trim());
46  
47          MetadataImpl meta = builder.build();
48          assertEquals(expectedExtEvent.trim(), meta.getExtEvent());
49      }
50  
51      /**
52       * setReportType always stores the raw report type name as extEventType,
53       * regardless of any mapping applied to extEvent.
54       */
55      @ParameterizedTest(name = "setReportType({0}).extEventType = {0}")
56      @CsvSource({
57          "routineImmunization",
58          "influenzaVaccination",
59          "farmerFlu",
60          "rsvPrevention",
61          "measlesVaccination",
62          "genericImmunization",
63      })
64      void setReportType_extEventTypeIsAlwaysRawInput(String reportType) throws MetadataFault {
65          MetadataBuilder builder = new MetadataBuilder();
66          builder.setMetadataValidationEnabled(false);
67          builder.setReportType(reportType);
68  
69          MetadataImpl meta = builder.build();
70          assertEquals(reportType, meta.getExtEventType());
71      }
72  
73      /**
74       * genericImmunization forces ExtSourceVersion to DEX_VERSION2.
75       */
76      @Test
77      void setReportType_genericImmunization_forcesVersion2() throws MetadataFault {
78          MetadataBuilder builder = new MetadataBuilder();
79          builder.setMetadataValidationEnabled(false);
80          builder.setReportType("genericImmunization");
81  
82          MetadataImpl meta = builder.build();
83          assertEquals(Metadata.DEX_VERSION2, meta.getExtSourceVersion());
84      }
85  
86      /**
87       * A blank reportType should add a validation error and not throw.
88       */
89      @Test
90      void setReportType_blank_addsError() {
91          MetadataBuilder builder = new MetadataBuilder();
92          builder.setReportType("   ");
93          assertEquals(1, builder.getErrors().size());
94      }
95  
96      @Test
97      void setReportType_null_addsError() {
98          MetadataBuilder builder = new MetadataBuilder();
99          builder.setReportType(null);
100         assertEquals(1, builder.getErrors().size());
101     }
102 }
