32  /**
33   * IGDD-3582 regression test: proves the Hub's embedded HTTPS listener picks up a rotated
34   * TLS certificate without a process restart.
35   *
36   * <p>The fix is {@code ServerSslBundleReloader} (izgw-core), scheduled by izgw-hub's
37   * {@code ServerSslBundleReloadScheduler}: it polls the server's {@code SslBundle} keystore file's
38   * modification time and, on a change, rebuilds the bundle and pushes it via
39   * {@code SslBundleRegistry.updateBundle(...)}. This is deliberately <em>not</em> Spring Boot's own
40   * {@code reload-on-update} (disabled in {@code application.yml}): that mechanism is backed by
41   * {@code java.nio.file.WatchService} (inotify on Linux), which does not detect a change written
42   * by a different client to a network filesystem -- and this application's keystore lives on an
43   * AWS EFS (NFSv4.1) volume in every real deployment. Polling file mtime is the same mechanism
44   * already proven correct on EFS for the outbound client TLS path.
45   *
46   * <p>This test forces an immediate reload via {@code GET /rest/trust?reload=true} (mTLS, admin
47   * client cert) rather than waiting out the scheduler's own poll interval -- that endpoint calls
48   * the exact same {@code ServerSslBundleReloader.reloadIfChanged()} method the scheduler does, just
49   * synchronously and on demand. Observing the result requires a *separate*, fresh connection: the
50   * triggering request's own TLS handshake (and therefore the certificate
51   * {@code HttpsURLConnection.getServerCertificates()} would report on it) completes before the
52   * server processes the request and reloads, so that connection can never see its own effect --
53   * confirmed the hard way in CI (Tomcat's own connector log showed the reload succeeding with the
54   * correct rotated serial, while the triggering connection's handshake still reported the old one).
55   * Before this fix, this test fails: the certificate served after rotation still matches the
56   * pre-rotation serial.
57   *
58   * <p>All key material is generated fresh by {@link TlsTestKeystoreFactory} for this run only --
59   * nothing is checked into the repository, so there is nothing here that can expire.
60   *
61   * <p>Requires the same AWS DynamoDB / Elastic access {@link ApplicationTests} already requires
62   * to boot the real application context -- this is not a new environmental requirement introduced
63   * by this test.
64   */
65  @TestInstance(TestInstance.Lifecycle.PER_CLASS)

67  @ComponentScan("gov.cdc.izgateway")

96  	@DynamicPropertySource
97  	static void registerSslProperties(DynamicPropertyRegistry registry) throws Exception {
98  		sslDir = Files.createTempDirectory("tls-hot-reload-test-").toFile();
99  		sslDir.deleteOnExit();
100 		keystoreFile = new File(sslDir, "awsdev_keystore.bcfks");
101 		// File.deleteOnExit() on a directory only succeeds if it is empty at shutdown, so the
102 		// keystore file living inside sslDir needs its own registration too, or the directory
103 		// (and this file) leak on every run.
104 		keystoreFile.deleteOnExit();
105 
106 		clientKeyPair = TlsTestKeystoreFactory.generateKeyPair();
107 		clientCert = TlsTestKeystoreFactory.selfSign(clientKeyPair, TlsTestKeystoreFactory.clientDn(), 30);
108 
109 		initialKeyPair = TlsTestKeystoreFactory.generateKeyPair();
110 		initialCert = TlsTestKeystoreFactory.selfSign(initialKeyPair, TlsTestKeystoreFactory.serverDn(), 30);
111 		rotatedKeyPair = TlsTestKeystoreFactory.generateKeyPair();
112 		rotatedCert = TlsTestKeystoreFactory.selfSign(rotatedKeyPair, TlsTestKeystoreFactory.serverDn(), 30);
113 
114 		KeyStore initial = TlsTestKeystoreFactory.newServerKeyStore(SERVER_ALIAS, initialKeyPair, initialCert, STORE_PASSWORD);
115 		TlsTestKeystoreFactory.addTrustedCert(initial, CLIENT_ALIAS, clientCert);
116 		TlsTestKeystoreFactory.writeKeyStore(initial, keystoreFile, STORE_PASSWORD);
117 
118 		String path = keystoreFile.getAbsolutePath();
119 		String password = new String(STORE_PASSWORD);
120 		// Point the izgw-server SslBundle (type/provider already set to bcfks/BCFIPS in
121 		// application.yml) at this test's generated, ephemeral keystore file. This is the
122 		// SERVER's own identity AND the truststore deciding whether the generated admin client
123 		// cert is accepted during the mTLS handshake -- both live in the same bcfks file here.
124 		registry.add("spring.ssl.bundle.jks.izgw-server.keystore.location", () -> "file:" + path);
125 		registry.add("spring.ssl.bundle.jks.izgw-server.keystore.password", () -> password);
126 		registry.add("spring.ssl.bundle.jks.izgw-server.truststore.location", () -> "file:" + path);
127 		registry.add("spring.ssl.bundle.jks.izgw-server.truststore.password", () -> password);
128 		registry.add("spring.ssl.bundle.jks.izgw-server.key.alias", () -> SERVER_ALIAS);
129 		// Disable the fixed-port (9081) management connector: Application.java only adds it
130 		// when additionalPort >= 1. WebEnvironment.RANDOM_PORT only randomizes server.port,
131 		// not this second connector, so with Surefire's reuseForks=true, this test's
132 		// separately-configured context collides with ApplicationTests' already-bound 9081
133 		// in the same JVM (BindException: Address already in use). Not needed for this test.
134 		registry.add("server.local-port", () -> "0");

137 	@Test
138 	void certificateRotationIsPickedUpWithoutRestart() throws Exception {
139 		// Built once and reused throughout: none of this client-side material changes, only the
140 		// server's presented certificate does.
141 		SSLSocketFactory clientFactory = clientSslSocketFactory();
142 
143 		BigInteger baselineServing = capturePresentedCertSerial(clientFactory);
144 		assertEquals(initialCert.getSerialNumber(), baselineServing,
145 				"Server should initially present the generated baseline certificate");
146 
147 		// ServerSslBundleReloader detects a change by comparing file mtime, not content, to the
148 		// mtime it saw when the bundle was primed at context startup. Some filesystems (confirmed
149 		// on CI) only have whole-second mtime resolution, so rotating immediately after startup
150 		// can land in the same tick and look unchanged even though the content genuinely differs.
151 		// Sleep past the worst-case 1s granularity before rotating -- this is about the test's own
152 		// timing assumptions, not the reload mechanism's production latency.
153 		Thread.sleep(1100); // NOSONAR java:S2925 -- see comment above
154 
155 		// Rotate: atomically replace the live keystore file with the "renewed" one -- a plain
156 		// local file operation, exactly as design.md Decision 5 describes.
157 		KeyStore rotated = TlsTestKeystoreFactory.newServerKeyStore(SERVER_ALIAS, rotatedKeyPair, rotatedCert, STORE_PASSWORD);
158 		TlsTestKeystoreFactory.addTrustedCert(rotated, CLIENT_ALIAS, clientCert);
159 		File tempFile = new File(sslDir, "awsdev_keystore.bcfks.new");
160 		TlsTestKeystoreFactory.writeKeyStore(rotated, tempFile, STORE_PASSWORD);
161 		Files.move(tempFile.toPath(), keystoreFile.toPath(), StandardCopyOption.ATOMIC_MOVE, StandardCopyOption.REPLACE_EXISTING);
162 
163 		// Force an immediate reload via the same endpoint an operator would use, rather than
164 		// waiting out ServerSslBundleReloadScheduler's own poll interval. This call's own TLS
165 		// handshake -- and therefore the certificate it observes -- completes before the server
166 		// processes the request and reloads, so it cannot see the new certificate on itself; it
167 		// only needs to succeed. A separate, fresh connection is what actually observes the result.
168 		triggerReload(clientFactory);
169 
170 		BigInteger postRotationServing = capturePresentedCertSerial(clientFactory);
171 		assertNotEquals(baselineServing, postRotationServing,
172 				"Server is still presenting the pre-rotation certificate -- hot reload did not happen");
173 		assertEquals(rotatedCert.getSerialNumber(), postRotationServing,
174 				"Server should now present the rotated certificate's exact serial");
175 	}

177 	/**
178 	 * Calls {@code GET /rest/trust?reload=true} (mTLS, admin client cert) and asserts it
179 	 * succeeded. {@code HttpsURLConnection.connect()} only establishes the TLS handshake -- it
180 	 * does not force the HTTP request itself to be sent. {@code getResponseCode()} does, and that
181 	 * is the whole point of this call: without it, {@code TrustController.getTrust()} never runs.
182 	 */
183 	private void triggerReload(SSLSocketFactory clientFactory) throws Exception {
184 		HttpsURLConnection conx = openAdminConnection(clientFactory, true);
185 		try {
186 			conx.connect();
187 			int status = conx.getResponseCode();
188 			if (status != 200) {
189 				throw new AssertionError("GET /rest/trust?reload=true returned HTTP " + status);
190 			}
191 		} finally {
192 			conx.disconnect();
193 		}
194 	}

196 	/** Connects over TLS (a fresh handshake) and returns the serial number of the certificate the server presents. */
197 	private BigInteger capturePresentedCertSerial(SSLSocketFactory clientFactory) throws Exception {
198 		HttpsURLConnection conx = openAdminConnection(clientFactory, false);
199 		try {
200 			conx.connect();
201 			Certificate[] serverCerts = conx.getServerCertificates();
202 			return ((X509Certificate) serverCerts[0]).getSerialNumber();
203 		} finally {
204 			conx.disconnect();
205 		}
206 	}

208 	private HttpsURLConnection openAdminConnection(SSLSocketFactory clientFactory, boolean reload) throws Exception {
209 		URI uri = URI.create("https://localhost:" + port + "/rest/trust?reload=" + reload);
210 		HttpsURLConnection conx = (HttpsURLConnection) uri.toURL().openConnection();
211 		conx.setRequestMethod("GET");
212 		conx.setSSLSocketFactory(clientFactory);
213 		// Hostname verification is deliberately disabled: the generated test certs use the
214 		// real server DN (CN=dev.izgateway.org, matching design.md Decision 5) while this test
215 		// connects via loopback ("localhost") to exercise the embedded listener directly -- the
216 		// hostname mismatch is expected and intentional, not a validation gap. Which certificate
217 		// is presented is validated against a pinned trust store (see clientSslSocketFactory()),
218 		// not left open to "any" certificate. CodeQL's inline `lgtm[rule-id]` suppression comment
219 		// does not take effect under this repo's CodeQL setup (confirmed: the alert stayed open
220 		// with the comment in place) -- dismissed via the code-scanning alerts API instead
221 		// (reason: "used in tests"), not via a comment here.
222 		conx.setHostnameVerifier((hostname, session) -> true);
223 		// Disabling keep-alive is necessary but not sufficient on its own: Java's
224 		// HttpURLConnection connection pool can reuse an already-established TLS connection
225 		// (and its already-negotiated certificate) for a subsequent request to the same
226 		// host:port even with a brand-new SSLSocketFactory -- pooling happens at the socket
227 		// level, below where the factory comes into play.
228 		conx.setRequestProperty("Connection", "close");
229 		return conx;
230 	}

232 	/**
233 	 * Builds an in-memory (BCFKS, never written to disk) client SSLContext presenting the
234 	 * generated admin test cert, and trusting exactly the two certificates this run generated
235 	 * (baseline + rotated) -- pinned, not a blanket trust-all TrustManager, since the point of
236 	 * this test is to observe which of those two specific certificates the server presents.
237 	 * Uses BCFKS/BCFIPS and explicitly pins the BCJSSE provider, matching this project's
238 	 * {@code ClientTlsSupport} and the rest of this diff's key material -- not just FIPS-policy
239 	 * consistency, but because the real {@code Application} this test boots sets
240 	 * {@code org.bouncycastle.fips.approved_only=true} process-wide, so an unpinned
241 	 * {@code getInstance(...)} call could otherwise silently resolve through a FIPS-restricted
242 	 * provider depending on this JVM's provider registration order.
243 	 */
244 	private SSLSocketFactory clientSslSocketFactory() throws Exception {
245 		KeyStore clientKs = KeyStore.getInstance("BCFKS", TlsTestKeystoreFactory.BCFIPS);
246 		clientKs.load(null, null);
247 		clientKs.setKeyEntry("client", clientKeyPair.getPrivate(), STORE_PASSWORD, new Certificate[] { clientCert });
248 		KeyManagerFactory kmf = KeyManagerFactory.getInstance("PKIX", "BCJSSE");
249 		kmf.init(clientKs, STORE_PASSWORD);
250 
251 		KeyStore trustKs = KeyStore.getInstance("BCFKS", TlsTestKeystoreFactory.BCFIPS);
252 		trustKs.load(null, null);
253 		trustKs.setCertificateEntry("initial", initialCert);
254 		trustKs.setCertificateEntry("rotated", rotatedCert);
255 		TrustManagerFactory tmf = TrustManagerFactory.getInstance("PKIX", "BCJSSE");
256 		tmf.init(trustKs);
257 
258 		SSLContext sslContext = SSLContext.getInstance("TLSv1.2", "BCJSSE");
259 		sslContext.init(kmf.getKeyManagers(), tmf.getTrustManagers(), new SecureRandom());
260 		return sslContext.getSocketFactory();
261 	}
