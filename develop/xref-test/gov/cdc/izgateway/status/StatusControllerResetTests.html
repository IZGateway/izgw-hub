107 	private void distributeYields(Map<String, String> remoteOutcomes) {
108 		when(refreshQueueService.distribute(any(RefreshRequest.class), any()))
109 				.thenAnswer(call -> {
110 					Map<String, String> results = call.getArgument(1);
111 					results.putAll(remoteOutcomes);
112 					return results;
113 				});
114 	}

146 	@Test
147 	void resetAllAppliesTheResetOnTheServingInstance() {
148 		// distribute() skips the caller, so the serving instance only resets if the
149 		// controller applies the command itself. Without this the endpoint resets every
150 		// instance except the one that answered, and still reports it as OK (Local).
151 		doReturn(List.of()).when(endpointStatusService).findAll();
152 		distributeYields(Map.of());
153 
154 		controller.resetCircuitBreakers();
155 
156 		ArgumentCaptor<RefreshRequest> applied = ArgumentCaptor.forClass(RefreshRequest.class);
157 		verify(refreshQueueService).applyLocally(applied.capture());
158 		assertTrue(applied.getValue().reset(), "the local command must reset");
159 		assertNull(applied.getValue().destId(), "a null destination means every destination");
160 		assertFalse(applied.getValue().refreshConfig(), "a reset must not also reload configuration");
161 	}

164 	void resetOneAppliesTheResetOnTheServingInstance() {
165 		when(destinationService.findByDestId("md")).thenReturn(new Destination());
166 		when(endpointStatusService.findById("md")).thenReturn(statusFor("md"));
167 		distributeYields(Map.of());
168 
169 		controller.resetCircuitBreakerById("md");
170 
171 		// The controller must not mutate directly. applyLocally is the only reset, so a
172 		// local failure is reported rather than thrown, and the body is read afterwards.
173 		verify(endpointStatusService, never()).resetCircuitBreakerById(anyString());
174 		ArgumentCaptor<RefreshRequest> applied = ArgumentCaptor.forClass(RefreshRequest.class);
175 		verify(refreshQueueService).applyLocally(applied.capture());
176 		assertEquals("md", applied.getValue().destId());

212 		// The Configuration Console reads destId and status from the top level. The
213 		// @JsonUnwrapped on the status member is what keeps them there.
214 		Map<String, String> hosts = new TreeMap<>();
215 		hosts.put("us-east-1:host-a", "OK (Local)");
216 		StatusController.ResetOneResponse response = new StatusController.ResetOneResponse(
217 				statusFor("md"), hosts, RefreshQueueService.summarizeByRegion(hosts));
218 
219 		String json = mapperWithoutJurisdictionLookup().writeValueAsString(response);
220 
221 		assertTrue(json.contains("\"destId\":\"md\""), "destId must stay at the top level: " + json);
222 		assertTrue(json.contains("\"hosts\""), "hosts must be present for an authorised caller");
223 		assertTrue(json.contains("\"regions\""), "regions must be present");
224 		assertFalse(json.contains("\"status\":{"), "the status object must not be nested under a status key");
225 	}

305 	@Test
306 	void bothResetEndpointsRequireTheAdminRole() throws NoSuchMethodException {
307 		// A reset now reaches every instance in every region, so losing this annotation
308 		// would widen the blast radius fleet-wide. These tests call the controller
309 		// directly and bypass Spring Security, so nothing else here would notice.
310 		for (Method m : List.of(
311 				StatusController.class.getMethod("resetCircuitBreakers"),
312 				StatusController.class.getMethod("resetCircuitBreakerById", String.class))) {
313 			RolesAllowed roles = m.getAnnotation(RolesAllowed.class);
314 			assertNotNull(roles, m.getName() + " must carry @RolesAllowed");
315 			assertTrue(List.of(roles.value()).contains(Roles.ADMIN),
316 					m.getName() + " must require the admin role, found " + List.of(roles.value()));
317 		}
318 	}
