70  	void testLoggingEvents(List<LogEvent> list) {
71  		boolean hasBuild = false;
72  		boolean hasHeartbeat = false;
73  		Health health = null;
74  		boolean isHealthy = true;
75  		for (LogEvent event : list) {
76  			if (Strings.CS.contains(event.getMessage(), "Build:")) {
77  				hasBuild = true;
78  			} else if (Strings.CS.contains(event.getMessage(), "Heartbeat")) {
79  				hasHeartbeat = true;
80  			}
81  			
82  			
83  			Health h = event.getHealth();
84  			if (h != null) {
85  				health = h;		// Health was reported.
86  				if (!h.isHealthy()) {
87  					isHealthy = false;	
88  				}
89  			}
90  		}
91  		assertTrue(hasBuild, "Build not found");
92  		assertTrue(hasHeartbeat, "Heartbeat not found");
93  		assertNotNull(health, "Health was reported");
94  		assertTrue(isHealthy, "Instance is not healthy");
95  	}

118 	@Test
119 	void testLogEvents() {
120 		int count = 0;
121 		for (LogEvent event: logController.getLogs(null)) {
122 			count++;
123 			if ("DEBUG".equals(event.getLevel()) || Strings.CS.startsWith(event.getLoggerName(), "gov.cdc.izgateway.db.RefreshQueueService"))  {
124 				// Skip these, we don't care about them right now.
125 				continue;
126 			}
127 			Object eventId = event.getEventId();
128 			try {
129 				assertNotNull(eventId);
130 				assertTrue(eventId instanceof String);
131 				assertTrue(((String)eventId).matches("\\d+\\.\\d+"));
132 			} catch (AssertionFailedError err) {
133 				System.out.println("Failing Event["+count+"]: " + event.toString());
134 				throw err;
135 			}
136 		}
