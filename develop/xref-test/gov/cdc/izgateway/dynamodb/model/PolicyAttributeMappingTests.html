1   package gov.cdc.izgateway.dynamodb.model;
2   
3   import org.junit.jupiter.api.DisplayName;
4   import org.junit.jupiter.api.Test;
5   import software.amazon.awssdk.enhanced.dynamodb.TableSchema;
6   import software.amazon.awssdk.services.dynamodb.model.AttributeValue;
7   
8   import java.util.Map;
9   import java.util.Set;
10  
11  import static org.assertj.core.api.Assertions.assertThat;
12  
13  /**
14   * Pins the DynamoDB attribute types of the API-key access-control attributes to the storage contract
15   * agreed with Config Console (IGDD-3140 / IGDD-3257):
16   *
17   * <ul>
18   *   <li>{@code ApiKeyCredential.environments} — Number Set ({@code NS})</li>
19   *   <li>{@code ApiKeyCredential.useTypes} — String Set ({@code SS})</li>
20   *   <li>{@code Jurisdiction.allowedUseTypes} — String Set ({@code SS})</li>
21   * </ul>
22   *
23   * <p>These attributes are written by Config Console and only read by Hub, so a type disagreement is
24   * invisible to every test that mocks the repository: the bean compiles, the unit tests pass, and the
25   * failure surfaces only against real data. Asserting the generated attribute type here is the cheap
26   * guard. If one of these assertions fails, the Hub bean and Config Console have diverged — fix the
27   * contract, do not relax the test.</p>
28   *
29   * <p>Also pins the empty-set behaviour: DynamoDB cannot store an empty set, so "none" must be
30   * represented by the attribute being absent. For {@code Jurisdiction.allowedUseTypes} that absence
31   * <em>is</em> the deny-all policy state.</p>
32   *
33   * @author Audacious Inquiry
34   */
35  class PolicyAttributeMappingTests {
36  
37      private static final TableSchema<ApiKeyCredential> CREDENTIAL_SCHEMA =
38              TableSchema.fromBean(ApiKeyCredential.class);
39      private static final TableSchema<Jurisdiction> JURISDICTION_SCHEMA =
40              TableSchema.fromBean(Jurisdiction.class);
41  
42      /** Marshals with ignoreNulls=true, the way DynamoDbRepository writes items. */
43      private static Map<String, AttributeValue> marshal(ApiKeyCredential credential) {
44          return CREDENTIAL_SCHEMA.itemToMap(credential, true);
45      }
46  
47      private static ApiKeyCredential credential() {
48          ApiKeyCredential credential = new ApiKeyCredential();
49          credential.setJti("018f4e2a-5678-7abc-8def-000000000002");
50          credential.setStatus("active");
51          return credential;
52      }
53  
54      @Test
55      @DisplayName("environments marshals to a DynamoDB Number Set (NS), not a List")
56      void environments_isNumberSet() {
57          ApiKeyCredential credential = credential();
58          credential.setEnvironments(Set.of(2, 5));
59  
60          AttributeValue environments = marshal(credential).get("environments");
61  
62          assertThat(environments.ns()).containsExactlyInAnyOrder("2", "5");
63          assertThat(environments.hasL()).isFalse();
64          assertThat(environments.hasSs()).isFalse();
65      }
66  
67      @Test
68      @DisplayName("A round trip through the table schema preserves environments")
69      void environments_roundTrip() {
70          ApiKeyCredential credential = credential();
71          credential.setEnvironments(Set.of(2, 5));
72  
73          ApiKeyCredential restored = CREDENTIAL_SCHEMA.mapToItem(marshal(credential));
74  
75          assertThat(restored.getEnvironments()).containsExactlyInAnyOrder(2, 5);
76      }
77  
78      @Test
79      @DisplayName("useTypes marshals to a DynamoDB String Set (SS)")
80      void useTypes_isStringSet() {
81          ApiKeyCredential credential = credential();
82          credential.setUseTypes(Set.of("PROVIDER", "PUBLIC_HEALTH"));
83  
84          AttributeValue useTypes = marshal(credential).get("useTypes");
85  
86          assertThat(useTypes.ss()).containsExactlyInAnyOrder("PROVIDER", "PUBLIC_HEALTH");
87          assertThat(useTypes.hasL()).isFalse();
88      }
89  
90      @Test
91      @DisplayName("Absent set attributes are omitted on write and read back as null")
92      void nullSets_areOmittedAndReadBackAsNull() {
93          Map<String, AttributeValue> item = marshal(credential());
94  
95          // DynamoDB rejects an empty set, so "none" is expressed by omitting the attribute entirely.
96          assertThat(item).doesNotContainKeys("environments", "useTypes");
97  
98          ApiKeyCredential restored = CREDENTIAL_SCHEMA.mapToItem(item);
99          assertThat(restored.getEnvironments()).isNull();
100         assertThat(restored.getUseTypes()).isNull();
101     }
102 
103     @Test
104     @DisplayName("Jurisdiction.allowedUseTypes marshals to a DynamoDB String Set (SS)")
105     void allowedUseTypes_isStringSet() {
106         Jurisdiction jurisdiction = new Jurisdiction();
107         jurisdiction.setJurisdictionId(42);
108         jurisdiction.setAllowedUseTypes(Set.of("PROVIDER"));
109 
110         AttributeValue allowedUseTypes =
111                 JURISDICTION_SCHEMA.itemToMap(jurisdiction, true).get("allowedUseTypes");
112 
113         assertThat(allowedUseTypes.ss()).containsExactly("PROVIDER");
114         assertThat(allowedUseTypes.hasL()).isFalse();
115     }
116 
117     @Test
118     @DisplayName("A jurisdiction with no policy omits allowedUseTypes — absence is the deny-all state")
119     void allowedUseTypes_absentMeansDenyAll() {
120         Jurisdiction jurisdiction = new Jurisdiction();
121         jurisdiction.setJurisdictionId(42);
122 
123         Map<String, AttributeValue> item = JURISDICTION_SCHEMA.itemToMap(jurisdiction, true);
124 
125         assertThat(item).doesNotContainKey("allowedUseTypes");
126         assertThat(JURISDICTION_SCHEMA.mapToItem(item).getAllowedUseTypes()).isNull();
127     }
128 }
