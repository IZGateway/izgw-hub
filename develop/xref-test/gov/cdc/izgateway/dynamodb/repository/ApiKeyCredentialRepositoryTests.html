83      @Test
84      void buildGraceTerminationRequest_revokedBranch_targetsCorrectKeyWithGracePeriodCondition() {
85          Instant terminatedAt = Instant.parse("2026-07-20T15:00:00Z");
86          UpdateItemRequest req = ApiKeyCredentialRepository.buildGraceTerminationRequest(
87                  "izgateway-dev-test", "jti-abc", "revoked", terminatedAt, "system:grace-revocation");
88  
89          assertThat(req.tableName()).isEqualTo("izgateway-dev-test");
90          // Targets the exact item: partition entityType=ApiKeyCredential, sort key {jti} (no env prefix).
91          assertThat(req.key().get("entityType").s()).isEqualTo("ApiKeyCredential");
92          assertThat(req.key().get("sortKey").s()).isEqualTo("jti-abc");
93          // Only writes when still grace_period → exactly-once revoke across concurrent instances.
94          assertThat(req.conditionExpression()).isEqualTo("#st = :grace");
95          assertThat(req.expressionAttributeNames()).containsEntry("#st", "status");
96          assertThat(req.expressionAttributeValues().get(":grace").s()).isEqualTo("grace_period");
97          assertThat(req.expressionAttributeValues().get(":terminal").s()).isEqualTo("revoked");
98          // Writes revokedAt/revokedBy for the revoked branch; expiredAt/expiredBy stay untouched.
99          assertThat(req.updateExpression()).contains("revokedAt = :ta", "revokedBy = :tb")
100                 .doesNotContain("expiredAt", "expiredBy");
101         // terminatedAt is the ISO-8601 'Z' Instant form; updatedOn uses the DynamoDbAudit Date form
102         // (millis + numeric +0000 offset) so it round-trips through the Enhanced Client's converter.
103         assertThat(req.expressionAttributeValues().get(":ta").s()).isEqualTo("2026-07-20T15:00:00Z");
104         assertThat(req.expressionAttributeValues().get(":tb").s()).isEqualTo("system:grace-revocation");
105         assertThat(req.updateExpression()).contains("updatedOn = :uo", "updatedBy = :tb");
106         assertThat(req.expressionAttributeValues().get(":uo").s()).isEqualTo("2026-07-20T15:00:00.000+0000");
107     }

152     private Map<String, AttributeValue> rawItem(String jti, String status, String graceExpiresAt) {
153         Map<String, AttributeValue> item = new HashMap<>();
154         item.put("entityType", AttributeValue.fromS("ApiKeyCredential"));
155         // "sortKey" is the derived DynamoDB key attribute (has a phantom no-op setter); "jti" is the
156         // real bean property. Both are present on an actual row.
157         item.put("sortKey", AttributeValue.fromS(jti));
158         item.put("jti", AttributeValue.fromS(jti));
159         item.put("status", AttributeValue.fromS(status));
160         item.put("issuedAt", AttributeValue.fromS("2026-01-01T00:00:00Z"));
161         item.put("expiresAt", AttributeValue.fromS("2026-12-31T00:00:00Z"));
162         if (graceExpiresAt != null) {
163             item.put("graceExpiresAt", AttributeValue.fromS(graceExpiresAt));
164         }
165         return item;
166     }

168     @Test
169     void sanitizeAndMap_allValidTimestamps_mapsCleanly() {
170         Map<String, AttributeValue> item = rawItem("jti-good", "grace_period", "2026-07-20T00:00:00Z");
171 
172         ApiKeyCredential c = ApiKeyCredentialRepository.sanitizeAndMap(item);
173 
174         assertThat(c.getJti()).isEqualTo("jti-good");
175         assertThat(c.getStatus()).isEqualTo("grace_period");
176         assertThat(c.getIssuedAt()).isEqualTo(Instant.parse("2026-01-01T00:00:00Z"));
177         assertThat(c.getExpiresAt()).isEqualTo(Instant.parse("2026-12-31T00:00:00Z"));
178         assertThat(c.getGraceExpiresAt()).isEqualTo(Instant.parse("2026-07-20T00:00:00Z"));
179     }

181     @Test
182     void sanitizeAndMap_malformedGraceExpiresAt_nullsOnlyThatFieldAndKeepsRestOfRecord() {
183         // Reproduces the reported bad legacy value: missing colon in the UTC offset (+0000 vs +00:00/Z).
184         Map<String, AttributeValue> item = rawItem("jti-bad-grace", "grace_period", "2025-06-23T13:28:00.000+0000");
185 
186         ApiKeyCredential c = ApiKeyCredentialRepository.sanitizeAndMap(item);
187 
188         assertThat(c.getJti()).isEqualTo("jti-bad-grace");
189         assertThat(c.getStatus()).isEqualTo("grace_period");
190         assertThat(c.getGraceExpiresAt()).isNull();
191         // Unrelated valid Instant fields on the same record are unaffected.
192         assertThat(c.getIssuedAt()).isEqualTo(Instant.parse("2026-01-01T00:00:00Z"));
193         assertThat(c.getExpiresAt()).isEqualTo(Instant.parse("2026-12-31T00:00:00Z"));
194     }

196     @Test
197     void sanitizeAndMap_malformedGraceExpiresAt_dropsOutOfGraceCandidateSelection() {
198         // End-to-end of the fix's intent: a record that would previously abort findAll() entirely now
199         // maps to a credential that selectGraceCandidates naturally excludes (graceExpiresAt == null),
200         // instead of the malformed value silently making it look like a non-expiring grace_period key.
201         Map<String, AttributeValue> item = rawItem("jti-bad-grace", "grace_period", "2025-06-23T13:28:00.000+0000");
202         ApiKeyCredential c = ApiKeyCredentialRepository.sanitizeAndMap(item);
203 
204         assertThat(ApiKeyCredentialRepository.selectGraceCandidates(List.of(c), NOW)).isEmpty();
205     }

207     @Test
208     void sanitizeAndMap_multipleMalformedFields_nullsEachIndependently() {
209         Map<String, AttributeValue> item = rawItem("jti-multi-bad", "grace_period", "2025-06-23T13:28:00.000+0000");
210         item.put("issuedAt", AttributeValue.fromS("not-a-timestamp"));
211 
212         ApiKeyCredential c = ApiKeyCredentialRepository.sanitizeAndMap(item);
213 
214         assertThat(c.getGraceExpiresAt()).isNull();
215         assertThat(c.getIssuedAt()).isNull();
216         // A field that parses fine is untouched even though sibling fields on the same item are bad.
217         assertThat(c.getExpiresAt()).isEqualTo(Instant.parse("2026-12-31T00:00:00Z"));
218     }
