34  /**
35   * Test the performance simulator.
36   * Performs some number of test runs against the DevIisService
37   * and see if the response patterns match the expected
38   * cumulative distribution function for:
39   * a) Size
40   * b) Time
41   * c) Exceptions thrown
42   */
43  @Slf4j
44  class TestPerformanceSimulator extends TestMockBase {
45      private static final String QBP = "MSH|^~\\&|IZGW|IZGW|TEST|PERF|20210402091512.000-0100||QBP^Q11^QBP_Q11|20210330093013AZQ231|P|2.5.1|||ER|AL|||||Z34^CDCPHINVS|IZGW";
46      private static final String VXU = "MSH|^~\\&|IZGW|IZGW|TEST|PERF|20210402091512.000-0100||VXU^V04^VXU_V04|20210330093013AZQ231|P|2.5.1|||ER|AL|||||Z22^CDCPHINVS|IZGW";

48      private static final int MAX_RUNS = 1000;  
49      private static final int MAX_THREADS_PER_CORE = 20;
50      private static final int MAX_WAIT = 300;
51      private static final double ALPHA = 0.05;
52  
53      private static class CollectedData {
54          List<Integer> timings = Collections.synchronizedList(new ArrayList<>(MAX_RUNS));
55          List<Integer> sizes = Collections.synchronizedList(new ArrayList<>(MAX_RUNS));
56          Map<Simulator, Integer> exceptions = Collections.synchronizedMap(new TreeMap<Simulator, Integer>());
57      };

59      private static class KS_Stats {
60          boolean pass;
61          double ksValue;
62          double ksCrit;
63          double scales[] = new double[2];
64          String message;
65      }

67      private int seed;
68      /**
69       * Testing behavior of Mock IIS through PerformanceSimulator
70       */
71      @Test
72      @Disabled("Disable unless making changes to the simulator b/c it takes a while to run")
73      void testSimulator() throws Exception {
74          // Run this test in paralell or we'll be here all day!  That's a lot of sleeping.
75          ExecutorService x = Executors.newFixedThreadPool(Runtime.getRuntime().availableProcessors() * MAX_THREADS_PER_CORE);
76          CollectedData qbp_data = new CollectedData();
77          CollectedData vxu_data = new CollectedData();
78          AbstractPerformanceSimulator.setSeed(seed);
79          
80          for (int i = 0; i < MAX_RUNS; i++) {
81              x.submit(() -> sendMessage(QBP, qbp_data));
82              x.submit(() -> sendMessage(VXU, vxu_data));
83              log.debug("Request {}", i);
84          }
85          boolean success = false;
86  
87          int maxWait = MAX_WAIT, interval = 3;
88  
89          // Perform normal termination of the execution queue
90          x.shutdown();
91          do {
92              try {
93                  success = x.awaitTermination(interval, TimeUnit.SECONDS);
94                  maxWait -= interval;
95              } catch (InterruptedException e) {
96                  Thread.currentThread().interrupt();
97              }
98          } while (!success && maxWait > 0);
99          log.info("Done with requests");
100         System.err.println("\nDone");
101 
102         Assertions.assertTrue(x.isTerminated(), "All tasks not completed");
103         // Now for the real work of this test
104         Histogram actualValues = new ArrayHistogram(qbp_data.timings, PerformanceSimulatorDailyFailures.DELAYS);
105         KS_Stats test = testResults(actualValues, PerformanceSimulatorDailyFailures.DELAYS);
106         if (!test.pass) {
107             System.out.printf("QBP Timing Actual:\n%s\nExpected:\n%s%n", actualValues, PerformanceSimulatorDailyFailures.DELAYS);
108             log.error("QBP Timing distribution: {}", test.message);
109         }
110         //Assertions.assertTrue(test.pass, "Timing distribution does not match expected value for QBP");
111 
112         actualValues = new ArrayHistogram(qbp_data.sizes, PerformanceSimulatorDailyFailures.QBP_SIZES);
113         test = testResults(actualValues, PerformanceSimulatorDailyFailures.QBP_SIZES);
114         if (!test.pass) {
115             System.out.printf("QBP Size Actual:\n%s\nExpected:\n%s\n", actualValues, PerformanceSimulatorDailyFailures.QBP_SIZES);
116             log.error("QBP Size distribution: {}", test.message);
117         }
118         Assertions.assertTrue(test.pass, "Size distribution does not match expected value for QBP");
119 
120         actualValues = new ArrayHistogram(vxu_data.timings, PerformanceSimulatorDailyFailures.DELAYS);
121         test = testResults(actualValues, PerformanceSimulatorDailyFailures.DELAYS);
122         if (!test.pass) {
123             System.out.printf("VXU Timing Actual:\n%s\nExpected:\n%s%n", actualValues, PerformanceSimulatorDailyFailures.DELAYS);
124             log.error("VXU Timing distribution: {}", test.message);
125         }
126         Assertions.assertTrue(test.pass, "Timing distribution does not match expected value for VXU");
127 
128         // We don't need to worry about VXU SIZES in the simulator
129         System.out.printf("QBP Exceptions: %s%n", qbp_data.exceptions);
130         System.out.printf("VXU Exceptions: %s%n", vxu_data.exceptions);
131 
132         long[] expected = new long[ErrorType.values().length];
133         long[] observed = new long[expected.length];
134         int i = 0;
135         for (ErrorType et: ErrorType.values()) {
136             observed[i] = qbp_data.exceptions.getOrDefault(et, Integer.valueOf(0));
137             observed[i] += vxu_data.exceptions.getOrDefault(et, Integer.valueOf(0));
138             expected[i] = et.getFrequency();
139             log.info("Expected: {}\tObserved: {}\t{}", expected[i], observed[i], et);
140             i++;
141         }
142         ChiSquareTest chiSqTest = new ChiSquareTest();
143         double chisq = chiSqTest.chiSquareDataSetsComparison(expected, observed);
144         double alpha = chiSqTest.chiSquareTestDataSetsComparison(expected, observed);
145         String message = String.format("Exception distribution does not match expected values: %f (%f)", alpha, chisq);
146         log.info("Exceptions chisq = {} alpha = {}", chisq, alpha);
147         Assertions.assertTrue(alpha > ALPHA, message);
148     }

150     private KS_Stats testResults(Histogram actualValues, Histogram expectedValues) {
151         KS_Stats stats = new KS_Stats();
152         stats.ksValue = actualValues.ksStatistic(expectedValues, stats.scales);
153         stats.ksCrit = actualValues.ksCriticalValue(ALPHA/2, stats.scales);
154         stats.message = String.format("%f <=> %f @ (%f, %f)", stats.ksValue, stats.ksCrit, stats.scales[0], stats.scales[1]);
155         log.info(stats.message);
156         stats.pass = stats.ksValue < stats.ksCrit;
157         return stats;
158     }

160     /**
161      * Call submitSingleMessage with a QBP or VXU formatted to trigger
162      * performance testing behavior.
163      * @param hl7Message The message to send
164      * @return The resulting string value
165      * @throws Exception Any Exception thrown by submitSingleMessage
166      */
167     private Object sendMessage(String hl7Message, CollectedData data) throws Exception {
168         SubmitSingleMessageRequest request = new SubmitSingleMessageRequest(hl7Message);
169 
170         SubmitSingleMessageResponse response = null;
171         long timer = 0;
172         try {
173             RequestContext.init();
174             RequestContext.setResponse(new MockHttpServletResponse());
175             timer = -System.currentTimeMillis();
176             ResponseEntity<?> r = service.submitSoapRequest(request, null);
177             assertNotNull(r, "Result from submitSoapRequest was null");
178             if (r.getBody() instanceof FaultMessage fm) {
179             	Throwable ex = fm.getFault();
180             	if (ex != null) {
181 	                log.debug("Exception: {}", ex.getMessage(), ex);
182                     ErrorType et = findErrorTypeInException(ex);
183                     synchronized (data.exceptions) {
184                         Integer value = data.exceptions.getOrDefault(et, Integer.valueOf(0));
185                         data.exceptions.put(et, ++value);
186                     }
187             	}
188             } else {
189             	response = (SubmitSingleMessageResponse)r.getBody();
190             }
191             // Any exception thrown at this point is unexpected and should just throw
192         } finally {
193             timer += System.currentTimeMillis() + PerformanceSimulatorDailyFailures.OVERHEAD;
194             data.timings.add((int) timer);  // difference will safely fit into an int
195             if (response != null) {
196                 data.sizes.add(response.getHl7Message().length());
197             }
198             RequestContext.clear();
199         }
200         return null;
201     }

203 	private ErrorType findErrorTypeInException(Throwable inEx) {
204 		for (Throwable ex = inEx; ex != null; ex = ex.getCause()) {
205 			for (StackTraceElement trace : ex.getStackTrace()) {
206 				String methodName = trace.getMethodName();
207 				
208 				if (trace.getClassName().equals(AbstractPerformanceSimulator.class.getName()) &&
209 					methodName.startsWith("simulate")
210 				) {
211 					for (ErrorType et: ErrorType.values()) {
212 						if (et.name().replace("_", "").equalsIgnoreCase(methodName.substring("simulate".length()))) {
213 							return et;
214 						}
215 					}
216 				}
217 			}
218 		}
219 		log.error(Markers2.append(inEx), "Fault generating method not recognized in exception");
220 		return ErrorType.OTHER_FAULT; 
221 	}
